Private Function So(ByVal nm As String) As Double
    So = Val(Replace(Trim$(Me.Controls(nm).Text), ",", "."))
End Function

Private Function SoLe() As Long
    Dim s As String
    s = Trim$(Me.Controls("cboLe").Text)
    If InStr(s, ".") > 0 Then SoLe = Len(s) - InStr(s, ".") Else SoLe = 0
End Function
