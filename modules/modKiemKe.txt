Private Function TieuDe() As String
    TieuDe = "Xu" & ChrW(7845) & "t Polygon khoanh " & ChrW(273) & ChrW(7845) & "t"
End Function

Public Sub DC_XuatPolygonKhoanh()
    Dim csdl As String, vungs As Collection, s As String, lv As Level
    csdl = CsdlMacDinh()
    If Len(Dir$(csdl)) > 0 Then Set vungs = MoCSDL(csdl)
    If vungs Is Nothing Then Set vungs = New Collection
    If vungs.Count = 0 Then
        DC_ThongBao "Ch" & ChrW(432) & "a c" & ChrW(243) & " d" & ChrW(7919) & " li" & ChrW(7879) & "u khoanh " & ChrW(273) & ChrW(7845) & "t (CSDL) cho b" & ChrW(7843) & "n v" & ChrW(7869) & " n" & ChrW(224) & "y:" & vbCrLf & csdl & vbCrLf & vbCrLf & _
                    "Ch" & ChrW(7841) & "y Ki" & ChrW(7875) & "m k" & ChrW(234) & " > T" & ChrW(7841) & "o l" & ChrW(7853) & "p b" & ChrW(7843) & "n " & ChrW(273) & ChrW(7891) & " ki" & ChrW(7875) & "m k" & ChrW(234) & " > T" & ChrW(7841) & "o Topology b" & ChrW(7843) & "n " & ChrW(273) & ChrW(7891) & " ki" & ChrW(7875) & "m k" & ChrW(234) & " tr" & ChrW(432) & ChrW(7899) & "c.", TieuDe, vbExclamation
        Exit Sub
    End If
    s = InputBox("Co " & vungs.Count & " khoanh dat." & vbCrLf & _
                 "Level dat polygon (TT 08 - khoanh dat: 5):", "Xuat Polygon", DC_DocCauHinh("KK_LVPOLY", "5"))
    If StrPtr(s) = 0 Or Len(Trim$(s)) = 0 Then Exit Sub
    If IsNumeric(Trim$(s)) Then Set lv = DC_LevelTheoSo(CLng(Val(s))) Else Set lv = DC_TimLevel(s)
    If lv Is Nothing Then
        DC_ThongBao "Kh" & ChrW(244) & "ng t" & ChrW(236) & "m th" & ChrW(7845) & "y / kh" & ChrW(244) & "ng t" & ChrW(7841) & "o " & ChrW(273) & ChrW(432) & ChrW(7907) & "c level: " & s, TieuDe, vbExclamation
        Exit Sub
    End If
    DC_GhiCauHinh "KK_LVPOLY", Trim$(s)

    Dim i As Long, kv As Long, v As clsVung, rg() As Double, n As Long, dem As Long, bo As Long
    Dim gg As Long, o As Object
    On Error Resume Next
    CadInputQueue.SendKeyin "mark"
    Set o = ActiveDesignFile
    For i = 1 To vungs.Count
        Set v = vungs.Item(i)
        If v.HasRing Then
            gg = 0
            If v.Dao.Count > 0 Then Err.Clear: gg = o.GetNextGraphicGroup: If Err.Number <> 0 Then gg = 0
            For kv = 0 To v.Dao.Count
                If kv = 0 Then rg = v.ring Else rg = v.Dao.Item(kv)
                If VeShape(rg, lv, gg) Then
                    If kv = 0 Then dem = dem + 1
                Else
                    If kv = 0 Then bo = bo + 1
                End If
            Next kv
        End If
    Next i
    On Error GoTo 0
    RedrawAllViews
    DC_ThongBao ChrW(272) & ChrW(227) & " xu" & ChrW(7845) & "t " & dem & " polygon khoanh " & ChrW(273) & ChrW(7845) & "t l" & ChrW(234) & "n level " & lv.Name & "." & _
                IIf(bo > 0, vbCrLf & "B" & ChrW(7887) & " qua " & bo & " khoanh (qu" & ChrW(225) & " nhi" & ChrW(7873) & "u " & ChrW(273) & ChrW(7881) & "nh ho" & ChrW(7863) & "c l" & ChrW(7895) & "i h" & ChrW(236) & "nh h" & ChrW(7885) & "c).", "") & vbCrLf & _
                "(Sai: key-in  undo mark)", TieuDe
End Sub

Private Function VeShape(rg() As Double, ByVal lv As Level, ByVal gg As Long) As Boolean
    Dim n As Long, i As Long, pts() As Point3d, sh As ShapeElement
    On Error GoTo Loi
    n = UBound(rg, 1)                     ' rg(n) = rg(0) (vong kin)
    If n < 3 Or n > 4999 Then Exit Function
    ReDim pts(0 To n)
    For i = 0 To n
        pts(i).X = rg(i, 0): pts(i).Y = rg(i, 1): pts(i).Z = 0#
    Next i
    Set sh = CreateShapeElement1(Nothing, pts, msdFillModeNotFilled)
    If sh Is Nothing Then Exit Function
    Set sh.Level = lv
    On Error Resume Next
    sh.LineWeight = 0
    If gg > 0 Then sh.GraphicGroup = gg
    On Error GoTo Loi
    ActiveModelReference.AddElement sh
    VeShape = True
    Exit Function
Loi:
    VeShape = False
End Function

' Ma -> Array(soMau, rgb)
Private Function DocBangMau() As Object
    Dim d As Object, f As String, a() As String, i As Long, p() As String, n As Integer, s As String, q() As String
    Set d = CreateObject("Scripting.Dictionary")
    d.CompareMode = 1
    f = DC_THUMUC & "\MauHienTrang_TT08.txt"
    On Error Resume Next
    If Len(Dir$(f)) = 0 Then
        n = FreeFile
        Open f For Output As #n
        Print #n, "' Mau loai dat ban do hien trang - TT 08/2024/TT-BTNMT Phu luc V (9.7). Dong: Ma|SoMau|R|G|B"
        a = Split(MauMacDinh(), ",")
        For i = 0 To UBound(a)
            q = Split(a(i), ":"): p = Split(q(1), ".")
            Print #n, q(0) & "|" & p(0) & "|" & p(1) & "|" & p(2) & "|" & p(3)
        Next i
        Close #n
    End If
    n = FreeFile
    Open f For Input As #n
    Do While Not EOF(n)
        Line Input #n, s
        s = Trim$(s)
        If Len(s) > 0 And Left$(s, 1) <> "'" Then
            p = Split(s, "|")
            If UBound(p) >= 4 Then d(Trim$(p(0))) = Array(CLng(Val(p(1))), RGB(CLng(Val(p(2))), CLng(Val(p(3))), CLng(Val(p(4)))))
        End If
    Loop
    Close #n
    If d.Count = 0 Then
        a = Split(MauMacDinh(), ",")
        For i = 0 To UBound(a)
            q = Split(a(i), ":"): p = Split(q(1), ".")
            d(q(0)) = Array(CLng(p(0)), RGB(CLng(p(1)), CLng(p(2)), CLng(p(3))))
        Next i
    End If
    Set DocBangMau = d
End Function

        End If
    Next i
    If rgbMode And Not ct Is Nothing Then
        On Error Resume Next
        ActiveDesignFile.AttachColorTable ct
        On Error GoTo 0
    End If

' "ONT+CLN" -> "ONT" ; bo khoang trang
Private Function MaChinh(ByVal s As String) As String
    Dim p As Long
    s = UCase$(Trim$(s))
    p = InStr(s, "+"): If p > 0 Then s = Left$(s, p - 1)
    p = InStr(s, ","): If p > 0 Then s = Left$(s, p - 1)
    MaChinh = Trim$(s)
End Function

Private Function ToShape(rg() As Double, ByVal lv As Level, ByVal mau As Long) As Boolean
    Dim n As Long, i As Long, pts() As Point3d, sh As ShapeElement, o As Object
    On Error GoTo Loi
    n = UBound(rg, 1)
    If n < 3 Or n > 4999 Then Exit Function
    ReDim pts(0 To n)
    For i = 0 To n
        pts(i).X = rg(i, 0): pts(i).Y = rg(i, 1): pts(i).Z = 0#
    Next i
    Set sh = CreateShapeElement1(Nothing, pts, msdFillModeFilled)
    If sh Is Nothing Then Exit Function
    Set sh.Level = lv
    On Error Resume Next
    sh.Color = mau
    sh.LineWeight = 0
    Set o = sh
    o.FillColor = mau
    On Error GoTo Loi
    ActiveModelReference.AddElement sh
    ToShape = True
    Exit Function
Loi:
    ToShape = False
End Function
