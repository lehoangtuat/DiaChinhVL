Private Function NapMau(ByVal path As String) As Boolean
    NapMau = False
    gTuDgn = False
    Dim st As Object
    On Error Resume Next
    Set st = CreateObject("ADODB.Stream")
    On Error GoTo 0
    If st Is Nothing Then Exit Function

    st.Type = 2
    st.Charset = "utf-8"
    st.Open
    On Error Resume Next
    st.LoadFromFile path
    If Err.Number <> 0 Then st.Close: Exit Function
    On Error GoTo 0
    Dim txt As String
    txt = st.ReadText(-1)
    st.Close

    Dim raw() As String
    raw = Split(txt, vbLf)

    ' Dem
    Dim i As Long, nL As Long, nT As Long
    For i = LBound(raw) To UBound(raw)
        If Len(raw(i)) > 0 Then
            Select Case Left$(raw(i), 1)
                Case "L": nL = nL + 1
                Case "T": nT = nT + 1
            End Select
        End If
    Next i
    ReDim gLineLevel(1 To nL + 1)
    ReDim gLineCoords(1 To nL + 1)
    ReDim gTL(1 To nT + 1)
    ReDim gTKind(1 To nT + 1)
    ReDim gTCao(1 To nT + 1)
    ReDim gTX(1 To nT + 1)
    ReDim gTY(1 To nT + 1)
    ReDim gTS(1 To nT + 1)
    gNL = 0: gNT = 0

    Dim parts() As String, k As Long, m As Long
    Dim co() As Double
    For i = LBound(raw) To UBound(raw)
        If Len(raw(i)) = 0 Then GoTo tiep
        parts = Split(raw(i), vbTab)
        Select Case parts(0)
            Case "H"
                gUor = ValD(parts(1))
                If UBound(parts) >= 6 Then gTuDgn = (Trim$(parts(6)) = "DGN")
                gScDgn = 0: gDgnPath = ""
                If gTuDgn And UBound(parts) >= 8 Then
                    gScDgn = ValD(parts(7)): gDgnPath = Trim$(parts(8))
                End If
                gXmin = ValD(parts(2)): gYmin = ValD(parts(3))
                gXmax = ValD(parts(4)): gYmax = ValD(parts(5))
            Case "L"
                gNL = gNL + 1
                gLineLevel(gNL) = CLng(ValD(parts(1)))
                m = UBound(parts) - 1        ' so gia tri toa do
                ReDim co(0 To m - 1)
                For k = 0 To m - 1
                    co(k) = ValD(parts(2 + k))
                Next k
                gLineCoords(gNL) = co
            Case "T"
                gNT = gNT + 1
                gTL(gNT) = CLng(ValD(parts(1)))
                gTKind(gNT) = parts(2)
                gTCao(gNT) = ValD(parts(3))
                gTX(gNT) = ValD(parts(4))
                gTY(gNT) = ValD(parts(5))
                If UBound(parts) >= 6 Then gTS(gNT) = parts(6) Else gTS(gNT) = ""
        End Select
tiep:
    Next i

    DatCotMacDinh
    If gTuDgn Then TinhCotTuLuoi
    gLoaded = True
    NapMau = True
End Function

' Tim o so do (level 24) -> tra ve tam & kich thuoc qua ByRef
Private Function TimBox(ByRef cxb As Double, ByRef cyb As Double, _
                        ByRef bw As Double, ByRef bh As Double) As Boolean
    Dim i As Long, j As Long, co() As Double, m As Long
    Dim x1 As Double, y1 As Double, x2 As Double, y2 As Double
    Dim bestA As Double, has As Boolean
    bestA = -1: has = False
    For i = 1 To gNL
        If gLineLevel(i) = 24 Then
            co = gLineCoords(i)
            m = (UBound(co) + 1) \ 2
            x1 = co(0): x2 = co(0): y1 = co(1): y2 = co(1)
            For j = 0 To m - 1
                If co(2 * j) < x1 Then x1 = co(2 * j)
                If co(2 * j) > x2 Then x2 = co(2 * j)
                If co(2 * j + 1) < y1 Then y1 = co(2 * j + 1)
                If co(2 * j + 1) > y2 Then y2 = co(2 * j + 1)
            Next j
            If (x2 - x1) * (y2 - y1) > bestA Then
                bestA = (x2 - x1) * (y2 - y1)
                cxb = (x1 + x2) / 2#: cyb = (y1 + y2) / 2#
                bw = x2 - x1: bh = y2 - y1
                has = True
            End If
        End If
    Next i
    TimBox = has
End Function

' Tra ve min(hHienTai, co chu lon nhat ma chuoi s van vua be rong wMax).
' Dung de tim MOT co chu chung cho ca bang thay vi thu nho tung o.
Private Function CaoVuaO(ByVal hHienTai As Double, ByVal s As String, _
                         ByVal hGoc As Double, ByVal wMax As Double) As Double
    CaoVuaO = hHienTai
    If wMax <= 0# Or Len(s) = 0 Then Exit Function
    Dim w As Double, h As Double
    w = RongChuoi(s, hGoc) * C_CAL
    h = hGoc
    If w > wMax * 0.9 Then h = hGoc * (wMax * 0.9) / w
    If h < hHienTai Then CaoVuaO = h
End Function

' Dien 1 cell bang. tx = TAM X cot (template), ty = TAM Y hang (template).
' SUA 05/08/2026: dat chu CANH GIUA O bang Justification CenterCenter -> chu
' luon nam giua cot (thang hang duoi tieu de), khong con lech phai nhu cu.
' wMax chi de THU NHO chu neu rong hon o.
Private Sub DienCell(ByVal tx As Double, ByVal ty As Double, ByVal s As String, _
                     ByVal hc As Double, ByVal cxb As Double, ByVal cyb As Double, _
                     ByVal k As Double, ByVal Cpx As Double, ByVal Cpy As Double, _
                     Optional ByVal wMax As Double = 0#)
    If Len(Trim$(s)) = 0 Then Exit Sub
    Dim mx As Double, my As Double, h As Double, w As Double
    mx = (tx - cxb) * k + Cpx           ' tam o (da map sang ban ve)
    my = (ty - cyb) * k + Cpy
    h = hc
    w = RongChuoi(s, h) * C_CAL
    If wMax > 0# And w > wMax * 0.9 Then h = h * (wMax * 0.9) / w

    On Error Resume Next
    ActiveSettings.TextStyle.Height = h
    ActiveSettings.TextStyle.Width = h * C_TILE_W
    ActiveSettings.TextStyle.Justification = msdTextJustificationCenterCenter
    Dim oT As TextElement
    Set oT = Nothing
    Set oT = CreateTextElement1(Nothing, s, Point3dFromXYZ(mx, my, 0#), Matrix3dIdentity)
    If Not oT Is Nothing Then oT.LineWeight = 0: ActiveModelReference.AddElement oT
    On Error GoTo 0
End Sub

' SUA 09/10/2026: dua chu bang toa do vao GIUA HANG: tim 2 duong ke NGANG
' gan nhat (tren/duoi) di qua cot x cua tag -> lay trung diem.
' Chi ap dung khi mau lay tu .dgn (mau .txt cu da tinh san tam hang).
Private Function TamHang(ByVal x As Double, ByVal y As Double) As Double
    TamHang = y
    If Not gTuDgn Then Exit Function
    Dim i As Long, j As Long, m As Long, co() As Double
    Dim ax As Double, ay As Double, bx As Double, by As Double, yy As Double
    Dim yt As Double, yd As Double
    yt = 1E+30: yd = -1E+30
    For i = 1 To gNL
        co = gLineCoords(i)
        m = (UBound(co) + 1) \ 2
        For j = 0 To m - 2
            ax = co(2 * j): ay = co(2 * j + 1): bx = co(2 * j + 2): by = co(2 * j + 3)
            If Abs(by - ay) <= 0.01 * Abs(bx - ax) And Abs(bx - ax) > 0.000001 Then
                If (x - ax) * (x - bx) <= 0# Then
                    yy = (ay + by) / 2#
                    If yy > y And yy < yt Then yt = yy
                    If yy <= y And yy > yd Then yd = yy
                End If
            End If
        Next j
    Next i
    If yt < 1E+29 And yd > -1E+29 Then
        If (yt - yd) < 100# Then TamHang = (yt + yd) / 2#
    End If
End Function

' Sap cell theo Y giam dan (tren xuong) - insertion sort song song
Private Sub SapGiamY(ByRef ax() As Double, ByRef ay() As Double, ByVal c As Long)
    Dim i As Long, j As Long, tx As Double, ty As Double
    For i = 2 To c
        tx = ax(i): ty = ay(i)
        j = i - 1
        Do While j >= 1
            If ay(j) < ty Then
                ax(j + 1) = ax(j): ay(j + 1) = ay(j): j = j - 1
            Else
                Exit Do
            End If
        Loop
        ax(j + 1) = tx: ay(j + 1) = ty
    Next i
End Sub

' Be rong (chieu=0) hoac cao (chieu=1) cua ring
Private Function found_bb(ByRef ring() As Double, ByVal chieu As Long) As Double
    Dim n As Long, i As Long, A As Double, b As Double
    n = UBound(ring, 1)
    A = ring(0, chieu): b = ring(0, chieu)
    For i = 1 To n - 1
        If ring(i, chieu) < A Then A = ring(i, chieu)
        If ring(i, chieu) > b Then b = ring(i, chieu)
    Next i
    found_bb = b - A
End Function


' Nguong x giu panel trai (bo panel phai neu mau co 2 panel canh nhau).
' Tra ve gXmax+1 neu chi 1 panel (giu het).
Private Function PanelTraiXlim() As Double
    Dim A As Double
    A = (gXmax - gXmin) * (gYmax - gYmin)
    Dim rminx() As Double, rmaxx() As Double, rarea() As Double, nr As Long
    ReDim rminx(1 To gNL + 1): ReDim rmaxx(1 To gNL + 1): ReDim rarea(1 To gNL + 1)
    nr = 0
    Dim i As Long, co() As Double, m As Long, j As Long
    Dim x1 As Double, x2 As Double, y1 As Double, y2 As Double
    For i = 1 To gNL
        co = gLineCoords(i)
        m = (UBound(co) + 1) \ 2
        If m = 4 Or m = 5 Then
            x1 = co(0): x2 = co(0): y1 = co(1): y2 = co(1)
            For j = 0 To m - 1
                If co(2 * j) < x1 Then x1 = co(2 * j)
                If co(2 * j) > x2 Then x2 = co(2 * j)
                If co(2 * j + 1) < y1 Then y1 = co(2 * j + 1)
                If co(2 * j + 1) > y2 Then y2 = co(2 * j + 1)
            Next j
            If (x2 - x1) * (y2 - y1) > 0.15 * A Then
                nr = nr + 1
                rminx(nr) = x1: rmaxx(nr) = x2: rarea(nr) = (x2 - x1) * (y2 - y1)
            End If
        End If
    Next i

    If nr < 2 Then PanelTraiXlim = gXmax + 1#: Exit Function

    ' Top-2 theo dien tich
    Dim i1 As Long, i2 As Long
    i1 = 1
    For i = 2 To nr
        If rarea(i) > rarea(i1) Then i1 = i
    Next i
    i2 = -1
    For i = 1 To nr
        If i <> i1 Then
            If i2 = -1 Then
                i2 = i
            ElseIf rarea(i) > rarea(i2) Then
                i2 = i
            End If
        End If
    Next i

    Dim lMax As Double, rMin As Double
    If rminx(i1) <= rminx(i2) Then
        lMax = rmaxx(i1): rMin = rminx(i2)
    Else
        lMax = rmaxx(i2): rMin = rminx(i1)
    End If

    If rMin > lMax - 1# Then
        PanelTraiXlim = lMax + (rMin - lMax) * 0.5
    Else
        PanelTraiXlim = gXmax + 1#
    End If
End Function


' Ve chu cho khung mau:
'  - Uu tien: dat NGUYEN chuoi, can le trai (LeftBottom) tai dung toa do mau.
'  - Neu that bai: lui ve cat doan <=48 ky tu, can giua theo be rong tinh tay.
Private Sub VeTextMau(ByVal x As Double, ByVal y As Double, _
                      ByVal s As String, ByVal hc As Double)
    If Len(s) = 0 Or hc <= 0 Then Exit Sub

    ' Thu 1: nguyen chuoi, can le trai
    If VeChuoiTrai(x, y, s, hc) Then Exit Sub

    ' Thu 2 (du phong): cat doan, can giua theo be rong tinh tay
    Dim startX As Double
    If gKieuCan = "G" Then
        startX = x - RongChuoi(s, hc) / 2#
    Else
        startX = x
    End If
    Const MAXLEN As Long = 48
    Dim p As Long, sub_s As String, w As Double, curX As Double
    curX = startX
    p = 1
    Do While p <= Len(s)
        sub_s = Mid$(s, p, MAXLEN)
        w = RongChuoi(sub_s, hc)
        On Error Resume Next
        ActiveSettings.TextStyle.Height = hc
        ActiveSettings.TextStyle.Width = hc * C_TILE_W
        ActiveSettings.TextStyle.Justification = msdTextJustificationCenterCenter
        Dim oT As TextElement
        Set oT = Nothing
        Set oT = CreateTextElement1(Nothing, sub_s, Point3dFromXYZ(curX + w / 2#, y, 0#), Matrix3dIdentity)
        If Not oT Is Nothing Then oT.LineWeight = 0: ActiveModelReference.AddElement oT
        On Error GoTo 0
        curX = curX + w
        p = p + MAXLEN
    Loop
End Sub

' Dat nguyen chuoi can le trai. Tra ve True neu thanh cong.
Private Function VeChuoiTrai(ByVal x As Double, ByVal y As Double, _
                             ByVal s As String, ByVal hc As Double) As Boolean
    VeChuoiTrai = False
    Dim px As Double
    If gKieuCan = "G" Then
        px = x - RongChuoi(s, hc) / 2#     ' che do giua: dich trai nua be rong
    Else
        px = x
    End If
    On Error Resume Next
    Err.Clear
    ActiveSettings.TextStyle.Height = hc
    ActiveSettings.TextStyle.Width = hc * C_TILE_W
    ActiveSettings.TextStyle.Justification = msdTextJustificationLeftBottom
    Dim oT As TextElement
    Set oT = Nothing
    Set oT = CreateTextElement1(Nothing, s, Point3dFromXYZ(px, y - hc * 0.5, 0#), Matrix3dIdentity)
    If (Not oT Is Nothing) And (Err.Number = 0) Then
        oT.LineWeight = 0
        ActiveModelReference.AddElement oT
        If Err.Number = 0 Then VeChuoiTrai = True
    End If
    On Error GoTo 0
End Function

' Uoc luong be rong chuoi theo tung ky tu (chinh xac hon cho dong nhieu dau cham)
Private Function RongChuoi(ByVal s As String, ByVal hc As Double) As Double
    Dim i As Long, c As String, w As Double
    For i = 1 To Len(s)
        c = Mid$(s, i, 1)
        Select Case c
            Case ".", ",", ";", ":", "'", "|", "!", "`"
                w = w + hc * 0.24
            Case " "
                w = w + hc * 0.26
            Case "i", "l", "I", "j", "t", "f", "r", "(", ")", "[", "]", "/", "-"
                w = w + hc * 0.32
            Case "m", "M", "W", "w"
                w = w + hc * 0.82
            Case Else
                w = w + hc * 0.5
        End Select
    Next i
    RongChuoi = w
End Function


'============================================================
' CHEN KHUNG PHU LUC 19 CHO 1 THUA CO SAN (goi tu dialog/CSDL)
'   found   : thua da co hinh hoc + thuoc tinh
'   mauPath : file mau ; kieuCan "T"/"G"
'============================================================
Public Sub ChenKhungTuThua(ByVal found As clsVung, ByVal mauPath As String, _
                           ByVal kieuCan As String, ByVal toBanDo As String, _
                           ByVal tenXa As String, ByVal tenHuyen As String, _
                           ByVal tenTinh As String, ByVal tyLe As Double)
    If found Is Nothing Then Exit Sub
    If Not found.HasRing Then MsgBox "Thua khong co hinh hoc.", vbExclamation: Exit Sub
    If Not NapMau(mauPath) Then
        MsgBox "Khong doc duoc file mau: " & mauPath, vbExclamation, "Trich luc": Exit Sub
    End If
    If UCase$(Trim$(kieuCan)) = "G" Then gKieuCan = "G" Else gKieuCan = "T"

    Dim cxb As Double, cyb As Double, bw As Double, bh As Double
    If Not TimBox(cxb, cyb, bw, bh) Then
        MsgBox "Khong tim thay o so do (level 24) trong mau.", vbExclamation, "Trich luc": Exit Sub
    End If

    ' Level ho so / level so do (form Trich luc, luu trong DiaChinhVL.ini)
    Dim lvCu As Level
    Set lvCu = ActiveSettings.Level
    ChuanBiLevel
    If Not gLvHSMot Is Nothing Then DatLv gLvHSMot

    Dim ring() As Double, n As Long, i As Long, j As Long
    ring = found.ring: n = UBound(ring, 1)
    Dim pw As Double, ph As Double
    pw = found_bb(ring, 0): ph = found_bb(ring, 1)

    ' ================= TY LE TO BAN DO =================
    ' SUA 05/08/2026: khung mau 19 phai CUNG CO GIAY du to ty le nao, nen k
    ' (ty le dat khung) phai TI LE THUAN voi ty le to:  k = tyLe / 20000.
    ' Kiem chung tu to 1:500 -> k=0.025 (cot X(m)=4.2203/168.8, chu=0.7/28).
    '   1:500  -> k=0.025    1:1000 -> k=0.05
    '   1:2000 -> k=0.10     1:5000 -> k=0.25
    ' Truoc day hardcode C_CANH_O=26.2 chi dung cho 1:500; to khac bi sai co.
    Dim mdDinh As Double
    mdDinh = gTyLeCu: If mdDinh <= 0 Then mdDinh = 500   ' mac dinh 500 lan dau
    If tyLe <= 0 Then tyLe = mdDinh          ' 0 = dung ty le lan truoc / mac dinh
    gTyLeCu = tyLe                           ' nho cho lan sau (form truyen xuong)

    Dim k As Double, kp As Double
    k = tyLe / 20000#                       ' ti le dat khung theo ty le to
    Dim bwm As Double, bhm As Double
    bwm = bw * k: bhm = bh * k

    ' SUA 05/08/2026: VE THUA THEO TY LE THAT + LAM TRON (giong gCadas
    ' TinhTyLe + DuTyLe). KHONG phong to nhet 82% nua.
    '   - kp = 1  -> thua ve dung ty le to ban do (nhu nhanh khong-template).
    '   - Neu thua LON hon o so do thi thu nho theo BUOC CHUAN (1/2, 1/2.5,
    '     1/5, 1/10...) = ty le so do van la boi so chuan cua ty le to.
    ' Kiem chung: thua 78 (13.4x12.7 m) vao o 24.7x26.2 m -> kp=1, chiem 54%,
    ' dung 1:500 nhu gCadas (truoc day ep 82% lam phong to 1.52 lan -> sai).
    Dim buoc As Variant, bi As Long
    buoc = Array(1#, 2#, 2.5, 5#, 10#, 20#, 25#, 50#, 100#)
    kp = 1#
    ' SUA 12/10/2026: ty le CHON TAY la ty le KHUNG; so do thua giu dung kich
    ' thuoc that (kp = 1) - thua lon tran o so do van ve dung. Chi thu nho khi
    ' ty le Tu dong ma thua qua lon (> 1:10000).
    If gTyLeTuDong Then
        For bi = 0 To UBound(buoc)
            kp = 1# / buoc(bi)
            If pw * kp <= bwm * 0.9 And ph * kp <= bhm * 0.9 Then Exit For
        Next bi
    End If
    If kp <= 0 Then kp = 1

    Dim Cpx As Double, Cpy As Double
    Cpx = found.cx: Cpy = found.cy
    ' SUA 09/10/2026: dat trich luc RA NGOAI ban do (ben phai pham vi ban ve) de
    ' o so do khong bi lan nhan/moc/chieu dai canh cua ban do (hoac lan xuat truoc).
    ' SUA 09/10/2026 (2): MAC DINH dat NGAY TAI THUA (theo yeu cau nguoi dung).
    ' Muon dat ra ngoai: DiaChinhVL.ini -> TL_DATNGOAI=1
    If DC_DocCauHinh("TL_DATNGOAI", "0") = "1" Then
        Dim rgM As Range3d, khoangTrai As Double
        On Error Resume Next
        rgM = ActiveModelReference.Range(True)
        If Err.Number = 0 And rgM.High.X > rgM.Low.X Then
            khoangTrai = (cxb - gXmin) * k              ' tu mep trai khung den tam o so do
            Cpx = rgM.High.X + (gXmax - gXmin) * k * 0.1 + khoangTrai
        End If
        Err.Clear
        On Error GoTo 0
    End If
    gTLCx = Cpx: gTLCy = Cpy

    ' Mau .dgn: chep nguyen phan tu (dung font/can le/vi tri nhu file mau)
    Dim chep As Boolean
    chep = (gTuDgn And gScDgn > 0 And Len(gDgnPath) > 0)
    If chep Then chep = (Len(Dir$(gDgnPath)) > 0)
    If chep Then
        DC_ChepMauDgn gDgnPath, gScDgn * k, Cpx - cxb * k, Cpy - cyb * k, found, _
                      toBanDo, tenXa, tenHuyen, tenTinh
    End If

    Dim slDT As Long, slTD As Long, slCD As Long
    slDT = 1: slTD = 2: slCD = 2
    Dim xlim As Double
    xlim = PanelTraiXlim()

    ' Lines cua khung (ti le co dinh k)
    Dim co() As Double, m As Long
    Dim ax As Double, ay As Double, bx As Double, by As Double, lminx As Double
    For i = 1 To gNL
        co = gLineCoords(i)
        m = (UBound(co) + 1) \ 2
        lminx = co(0)
        For j = 0 To m - 1
            If co(2 * j) < lminx Then lminx = co(2 * j)
        Next j
        If lminx >= xlim Then GoTo boLine2
        If chep Then GoTo boLine2                 ' da chep tu .dgn
        For j = 0 To m - 2
            ax = (co(2 * j) - cxb) * k + Cpx
            ay = (co(2 * j + 1) - cyb) * k + Cpy
            bx = (co(2 * j + 2) - cxb) * k + Cpx
            by = (co(2 * j + 3) - cyb) * k + Cpy
            If Abs(bx - ax) > 0.000001 Or Abs(by - ay) > 0.000001 Then VeLine ax, ay, bx, by, -1
        Next j
boLine2:
    Next i

    ReDim cellX_tt(1 To gNT) As Double: ReDim cellY_tt(1 To gNT) As Double: Dim c_tt As Long
    ReDim cellX_xx(1 To gNT) As Double: ReDim cellY_xx(1 To gNT) As Double: Dim c_xx As Long
    ReDim cellX_yy(1 To gNT) As Double: ReDim cellY_yy(1 To gNT) As Double: Dim c_yy As Long
    ReDim cellX_cd(1 To gNT) As Double: ReDim cellY_cd(1 To gNT) As Double: Dim c_cd As Long
    c_tt = 0: c_xx = 0: c_yy = 0: c_cd = 0

    Dim s As String, gt As String, mx As Double, my As Double, hc As Double
    For i = 1 To gNT
        If gTX(i) >= xlim Then GoTo boTxt2
        mx = (gTX(i) - cxb) * k + Cpx
        my = (gTY(i) - cyb) * k + Cpy
        hc = gTCao(i) * k
        If hc <= 0 Then GoTo boTxt2
        s = gTS(i)
        If gTKind(i) = "TEXT" Then
            If Not chep Then VeTextMau mx, my, s, hc
        Else
            Select Case s
                Case "tttl": c_tt = c_tt + 1: cellX_tt(c_tt) = gTX(i): cellY_tt(c_tt) = TamHang(gTX(i), gTY(i))
                Case "tamXtl": c_xx = c_xx + 1: cellX_xx(c_xx) = gTX(i): cellY_xx(c_xx) = TamHang(gTX(i), gTY(i))
                Case "tamYtl": c_yy = c_yy + 1: cellX_yy(c_yy) = gTX(i): cellY_yy(c_yy) = TamHang(gTX(i), gTY(i))
                Case "cdtl": c_cd = c_cd + 1: cellX_cd(c_cd) = gTX(i): cellY_cd(c_cd) = TamHang(gTX(i), gTY(i))
                Case Else
                    If Not chep Then
                        gt = GiaTriTag(s, found, toBanDo, tenXa, tenHuyen, tenTinh, slDT)
                        If Len(gt) > 0 Then VeTextMau mx, my, gt, hc
                    End If
            End Select
        End If
boTxt2:
    Next i

    Dim hcB As Double
    hcB = 28# * k
    DatLv mLvBang                     ' so lieu bang toa do
    SapGiamY cellX_tt, cellY_tt, c_tt
    SapGiamY cellX_xx, cellY_xx, c_xx
    SapGiamY cellX_yy, cellY_yy, c_yy
    SapGiamY cellX_cd, cellY_cd, c_cd

    ' Be rong kha dung cua tung cot, DO TU CHINH FILE MAU (khoang cach giua
    ' tag cot nay va tag cot ke tiep) -> thu nho chu cho vua o, khong tran.
    Dim wDinh As Double, wColX As Double, wColY As Double, wCanh As Double
    If c_tt >= 1 And c_yy >= 1 Then wDinh = Abs(cellX_yy(1) - cellX_tt(1))
    If c_yy >= 1 And c_xx >= 1 Then wColX = Abs(cellX_xx(1) - cellX_yy(1))
    If c_xx >= 1 And c_cd >= 1 Then wColY = Abs(cellX_cd(1) - cellX_xx(1))
    wCanh = wColY

    Dim dai As Double

    ' SUA 05/08/2026 (1): MOT co chu chung cho CA BANG.
    ' Truoc day moi o tu thu nho rieng -> so 10 chu so (cot X) bi thu nhieu
    ' hon so 9 chu so (cot Y) -> hai cot lech co chu nhau. Gio quet truoc
    ' toan bo o, lay co NHO NHAT, roi ve tat ca bang co do.
    ' SUA 11/10/2026: danh sach HANG cua bang = vong ngoai + cac vong dao (thua
    ' con nam lot trong). Moi vong: cac dinh, hang cuoi lap lai dinh dau (khep kin);
    ' so hieu dinh danh tiep nhau. rCanh(r) = chieu dai canh r -> r+1 (-1 = khong).
    Dim nR As Long, rSo() As String, rX() As Double, rY() As Double, rCanh() As Double
    TaoHangBang found, nR, rSo, rX, rY, rCanh

    Dim hB As Double
    hB = hcB
    Dim r As Long
    For r = 1 To nR
        If r <= c_tt Then
            hB = CaoVuaO(hB, rSo(r), hcB, CW_DINH * k)
            hB = CaoVuaO(hB, FNum(rX(r), slTD), hcB, CW_X * k)
            hB = CaoVuaO(hB, FNum(rY(r), slTD), hcB, CW_Y * k)
            If rCanh(r) >= 0# Then hB = CaoVuaO(hB, FNum(rCanh(r), slCD), hcB, CW_CANH * k)
        End If
    Next r

    ' SUA 11/10/2026: bang trang Trich luc chi co c_tt hang (mau tluc: 16).
    ' Hang 1..c_tt dien o day; hang c_tt+1 tro di (vd tu 17) dua sang bang
    ' trang "Trich luc ban do dia chinh" (ttkq/tamykq/tamxkq/cdkq), bang TRAI truoc.
    If nR > c_tt And c_tt > 0 Then
        If Not DienBangKQ(c_tt + 1, nR, rSo, rX, rY, rCanh, slTD, slCD, cxb, cyb, k, Cpx, Cpy) Then
            MsgBox "Thua co " & nR & " hang toa do, mau chi co " & c_tt & " hang (khong co bang ttkq).", vbExclamation, "Trich luc"
        End If
    End If
    If True Then
        DatLv mLvBang
        For r = 1 To nR
            If r <= c_tt Then
                ' Tat ca dat vao TAM cot (CX_*), hang y = cellY_tt(r) -> thang hang.
                DienCell CX_DINH, cellY_tt(r), rSo(r), hB, cxb, cyb, k, Cpx, Cpy, CW_DINH * k
                ' cot X(m) = toa do BAC ; cot Y(m) = toa do DONG
                DienCell CX_X, cellY_tt(r), FNum(rX(r), slTD), hB, cxb, cyb, k, Cpx, Cpy, CW_X * k
                DienCell CX_Y, cellY_tt(r), FNum(rY(r), slTD), hB, cxb, cyb, k, Cpx, Cpy, CW_Y * k
                ' canh r -> r+1 dat GIUA hai hang va GIUA cot canh
                If rCanh(r) >= 0# And (r + 1) <= c_tt Then _
                    DienCell CX_CANH, (cellY_tt(r) + cellY_tt(r + 1)) / 2#, FNum(rCanh(r), slCD), hB, cxb, cyb, k, Cpx, Cpy, CW_CANH * k
            End If
        Next r
    End If

    ' SUA 09/10/2026: bang phu o trang Ho so (tths/cdhs): so hieu dinh + chieu dai
    DienBangPhu "tths", "cdhs", nR, rSo, rCanh, cxb, cyb, k, Cpx, Cpy, rX, rY
    ' SUA 14/10/2026: bang ttkq NHO (vd trang "9. Toa do dinh thua") = ban sao bang
    ' tttl: dien tu hang 1 (truoc day chi dien bang ttkq dai cho hang 17+)
    DienBangKQ 1, nR, rSo, rX, rY, rCanh, slTD, slCD, cxb, cyb, k, Cpx, Cpy, True

    ' So do thua: THU NHO vua o (kp), tam o = tam khung (Cpx,Cpy)
    VeThuaTrongO found, Cpx, Cpy, kp, bwm * 0.03
    Set gLvHSMot = Nothing
    Set gLvHSMap = Nothing
    DatLv lvCu
End Sub

' Doc cau hinh level (LV_HS, LV_SD) -> level cho tung nhom phan tu.
'   Level ho so : 1 so  -> moi phan tu chep tu mau + so lieu bang ve level do
'                 nhieu -> giu level cua file mau (so lieu bang theo level tag bang)
'   Level so do : so thu 1 = canh thua, 2 = so hieu dinh, 3 = nhan thua
'                 (1 so -> ca so do ve 1 level)
Private Sub ChuanBiLevel()
    Dim ds() As Long, c As Long, i As Long
    Set mLvBang = Nothing: Set mLvRanh = Nothing
    Set mLvSo = Nothing: Set mLvNhan = Nothing
    Set gLvHSMot = Nothing
    Set gLvHSMap = Nothing
    If DC_DocCauHinh("LV_HS_ON", "1") = "1" Then
        c = DC_DsSoLevel(DC_DocCauHinh("LV_HS", DC_LV_HS_MD), ds)
        If c = 1 Then
            Set gLvHSMot = DC_LevelTheoSo(ds(0))
            Set mLvBang = gLvHSMot
        ElseIf c > 1 Then
            ' SUA 10/10/2026: NHIEU so -> moi phan tu ho so PHAI nam tren cac level
            ' da nhap (de "Luu file ho so" lay du). Cac level cua file mau (tang dan)
            ' lan luot ung voi cac so da nhap; mau nhieu level hon thi phan du ve so cuoi.
            ' Vd mau 24,56,60,61 + nhap 24,56,60,61 -> giu nguyen nhu mau.
            Dim lvMau() As Long, nM As Long, j As Long, k2 As Long, tmp As Long
            ReDim lvMau(1 To gNL + gNT + 1)
            For i = 1 To gNL
                ThemLvMau lvMau, nM, gLineLevel(i)
            Next i
            For i = 1 To gNT
                ThemLvMau lvMau, nM, gTL(i)
            Next i
            For i = 1 To nM - 1                 ' sap tang dan
                For j = i + 1 To nM
                    If lvMau(j) < lvMau(i) Then tmp = lvMau(i): lvMau(i) = lvMau(j): lvMau(j) = tmp
                Next j
            Next i
            Set gLvHSMap = CreateObject("Scripting.Dictionary")
            For i = 1 To nM
                k2 = i - 1: If k2 > c - 1 Then k2 = c - 1
                Set gLvHSMap(CStr(lvMau(i))) = DC_LevelTheoSo(ds(k2))
            Next i
            Set gLvHSMot = DC_LevelTheoSo(ds(c - 1))      ' level mau la -> so cuoi
            For i = 1 To gNT
                If gTS(i) = "tttl" Then
                    If gLvHSMap.Exists(CStr(gTL(i))) Then Set mLvBang = gLvHSMap(CStr(gTL(i)))
                    Exit For
                End If
            Next i
            If mLvBang Is Nothing Then Set mLvBang = gLvHSMot
        End If
    End If
    If DC_DocCauHinh("LV_SD_ON", "1") = "1" Then
        c = DC_DsSoLevel(DC_DocCauHinh("LV_SD", DC_LV_SD_MD), ds)
        If c >= 1 Then
            Set mLvRanh = DC_LevelTheoSo(ds(0))
            If c >= 2 Then Set mLvSo = DC_LevelTheoSo(ds(1)) Else Set mLvSo = mLvRanh
            If c >= 3 Then Set mLvNhan = DC_LevelTheoSo(ds(2)) Else Set mLvNhan = mLvSo
        End If
    End If
End Sub

' Bang phu (vd trang Phieu do dac / Ho so: "So hieu dinh thua | Chieu dai (m)").
' Chi dien 2 cot: so hieu dinh (tag tagTT, moi hang 1 dinh, hang cuoi lap dinh 1)
' va chieu dai canh (tag tagCD, canh thu i vao tag cd thu i). So hang theo thua.
Private Sub DienBangPhu(ByVal tagTT As String, ByVal tagCD As String, ByVal nR As Long, _
                        rSo() As String, rCanh() As Double, ByVal cxb As Double, ByVal cyb As Double, _
                        ByVal k As Double, ByVal Cpx As Double, ByVal Cpy As Double, _
                        rX() As Double, rY() As Double)
    Dim tx() As Double, ty() As Double, cx() As Double, cy() As Double
    Dim ct As Long, cc As Long, i As Long, hT As Double, hC As Double, lvT As Long
    ReDim tx(1 To gNT + 1): ReDim ty(1 To gNT + 1)
    ReDim cx(1 To gNT + 1): ReDim cy(1 To gNT + 1)
    For i = 1 To gNT
        If gTS(i) = tagTT Then
            ct = ct + 1: tx(ct) = gTX(i): ty(ct) = TamHang(gTX(i), gTY(i))
            hT = gTCao(i): lvT = gTL(i)
        ElseIf gTS(i) = tagCD Then
            cc = cc + 1: cx(cc) = gTX(i): cy(cc) = gTY(i): hC = gTCao(i)
        End If
    Next i
    If ct = 0 Then Exit Sub
    SapCotHang tx, ty, ct, hT * 2#
    If cc > 0 Then SapCotHang cx, cy, cc, hC * 2#
    DatLvTag lvT

    ' canh thu j (khong tinh hang khep vong) vao tag cd thu j
    Dim w As Double, xc As Double
    For i = 1 To nR
        If i <= ct Then
            xc = TamCot(tx(i), ty(i), w)
            DienCell xc, ty(i), rSo(i), hT * k, cxb, cyb, k, Cpx, Cpy, w * k
        End If
        ' o chieu dai thu i nam giua hang i va i+1 -> hang khep vong de trong
        If rCanh(i) >= 0# And i <= cc Then
            xc = TamCot(cx(i), cy(i), w)
            DienCell xc, cy(i), FNum(rCanh(i), 2), hC * k, cxb, cyb, k, Cpx, Cpy, w * k
        End If
    Next i
    ' SUA 14/10/2026: bang phu co cot toa do (tag tamxhs = X Bac, tamyhs = Y Dong)
    ' -> dien theo hang cua tag tths (chi lay tag cung hang, bo tag rac o cho khac)
    Dim m As Long, yRow As Double, best As Long, dBest As Double, dd As Double, tg As Variant, gTri As Double
    For Each tg In Array("tamxhs", "tamyhs")
        For i = 1 To nR
            If i > ct Then Exit For
            best = 0: dBest = hT * 0.8
            For m = 1 To gNT
                If gTS(m) = tg Then
                    dd = Abs(TamHang(gTX(m), gTY(m)) - ty(i))
                    If dd < dBest And gTX(m) > tx(i) Then dBest = dd: best = m
                End If
            Next m
            If best > 0 Then
                If tg = "tamxhs" Then gTri = rX(i) Else gTri = rY(i)
                xc = TamCot(gTX(best), gTY(best), w)
                DienCell xc, ty(i), FNum(gTri, 2), gTCao(best) * k, cxb, cyb, k, Cpx, Cpy, w * k
            End If
        Next i
    Next tg
End Sub

' Danh sach hang bang toa do: vong ngoai + cac vong dao
Private Sub TaoHangBang(ByVal v As clsVung, ByRef nR As Long, ByRef rSo() As String, _
                        ByRef rX() As Double, ByRef rY() As Double, ByRef rCanh() As Double)
    Dim vong As New Collection, rg() As Double, kv As Long, m As Long, i As Long, so As Long
    vong.Add v.ring
    For kv = 1 To v.Dao.Count
        vong.Add v.Dao.Item(kv)
    Next kv
    nR = 0
    For kv = 1 To vong.Count
        rg = vong.Item(kv): nR = nR + UBound(rg, 1) + 1
    Next kv
    ReDim rSo(1 To nR + 1): ReDim rX(1 To nR + 1): ReDim rY(1 To nR + 1): ReDim rCanh(1 To nR + 1)
    nR = 0: so = 0
    For kv = 1 To vong.Count
        rg = vong.Item(kv): m = UBound(rg, 1)        ' rg(m) = rg(0) (vong kin)
        For i = 0 To m - 1
            nR = nR + 1
            rSo(nR) = CStr(so + i + 1)
            rX(nR) = rg(i, 1): rY(nR) = rg(i, 0)      ' X = toa do BAC, Y = DONG
            rCanh(nR) = Sqr((rg(i + 1, 0) - rg(i, 0)) ^ 2 + (rg(i + 1, 1) - rg(i, 1)) ^ 2)
        Next i
        nR = nR + 1                                    ' hang khep vong
        rSo(nR) = CStr(so + 1): rX(nR) = rg(0, 1): rY(nR) = rg(0, 0): rCanh(nR) = -1#
        so = so + m
    Next kv
End Sub

' Bang tren trang "TRICH LUC BAN DO DIA CHINH" (ttkq / tamykq / tamxkq / cdkq).
' Hang sap theo cot (bang trai truoc, roi bang phai), trong moi bang tu tren xuong.
' Tra ve False neu mau khong co bang nay.
Private Function DienBangKQ(ByVal r0 As Long, ByVal nR As Long, rSo() As String, rX() As Double, rY() As Double, _
                            rCanh() As Double, ByVal slTD As Long, ByVal slCD As Long, _
                            ByVal cxb As Double, ByVal cyb As Double, ByVal k As Double, _
                            ByVal Cpx As Double, ByVal Cpy As Double, _
                            Optional ByVal bangNho As Boolean = False) As Boolean
    Dim tx() As Double, ty() As Double, ct As Long, i As Long, hT As Double, lvT As Long
    ReDim tx(1 To gNT + 1): ReDim ty(1 To gNT + 1)
    For i = 1 To gNT
        If gTS(i) = "ttkq" Then
            ct = ct + 1: tx(ct) = gTX(i): ty(ct) = TamHang(gTX(i), gTY(i))
            hT = gTCao(i): lvT = gTL(i)
        End If
    Next i
    If ct = 0 Then Exit Function
    ' Mau co the co nhieu bang ttkq: bang nho (16 hang) tren trang co so do va
    ' 2 bang DAI dat canh nhau. Chi lay cac cot co nhieu hang nhat (bang dai).
    Dim i2 As Long, dem As Long, demMax As Long, nCot() As Long
    ReDim nCot(1 To ct)
    For i = 1 To ct
        dem = 0
        For i2 = 1 To ct
            If Abs(tx(i2) - tx(i)) < hT * 2# Then dem = dem + 1
        Next i2
        nCot(i) = dem
        If dem > demMax Then demMax = dem
    Next i
    dem = 0
    For i = 1 To ct
        If (nCot(i) * 2 > demMax) Xor bangNho Then dem = dem + 1: tx(dem) = tx(i): ty(dem) = ty(i)
    Next i
    ct = dem
    If ct = 0 Then Exit Function
    SapCotHang tx, ty, ct, hT * 2#
    DatLvTag lvT

    Dim w As Double, xc As Double, xT As Double, r As Long, hc As Double, sl As Long
    hc = hT * k
    ' sl = o (slot) trong bang, r = hang du lieu (bat dau tu r0, vd 17)
    r = r0: sl = 0
    Do While r <= nR And sl < ct
        sl = sl + 1
        xc = TamCot(tx(sl), ty(sl), w)
        DienCell xc, ty(sl), rSo(r), hc, cxb, cyb, k, Cpx, Cpy, w * k
        xT = TagCanhPhai("tamykq", tx(sl), ty(sl), hT)
        If xT > -1E+29 Then xc = TamCot(xT, ty(sl), w): DienCell xc, ty(sl), FNum(rX(r), slTD), hc, cxb, cyb, k, Cpx, Cpy, w * k
        xT = TagCanhPhai("tamxkq", tx(sl), ty(sl), hT)
        If xT > -1E+29 Then xc = TamCot(xT, ty(sl), w): DienCell xc, ty(sl), FNum(rY(r), slTD), hc, cxb, cyb, k, Cpx, Cpy, w * k
        ' canh r -> r+1: giua 2 o, chi khi o ke tiep cung 1 bang (cung cot)
        If rCanh(r) >= 0# And r < nR And sl + 1 <= ct Then
            If Abs(tx(sl + 1) - tx(sl)) < hT * 2# Then
                xT = TagCanhPhai("cdkq", tx(sl), (ty(sl) + ty(sl + 1)) / 2#, hT * 2#)
                If xT > -1E+29 Then
                    xc = TamCot(xT, ty(sl), w)
                    DienCell xc, (ty(sl) + ty(sl + 1)) / 2#, FNum(rCanh(r), slCD), hc, cxb, cyb, k, Cpx, Cpy, w * k
                End If
            End If
        End If
        r = r + 1
    Loop
    If r <= nR And Not bangNho Then MsgBox "Thua co " & nR & " hang toa do, bang khong du cho (con " & (nR - r + 1) & " hang).", vbExclamation, "Trich luc"
    DienBangKQ = True
End Function

' X cua tag (ten) gan nhat ben PHAI x0, cung hang (|dy| <= dyMax). Khong co -> -1E+30
Private Function TagCanhPhai(ByVal ten As String, ByVal x0 As Double, ByVal y0 As Double, _
                             ByVal dyMax As Double) As Double
    Dim i As Long, best As Double, yy As Double
    TagCanhPhai = -1E+30: best = 1E+30
    If dyMax <= 0 Then dyMax = 1#
    For i = 1 To gNT
        If gTS(i) = ten Then
            If gTX(i) > x0 Then
                yy = TamHang(gTX(i), gTY(i))
                If Abs(yy - y0) <= dyMax * 1.5 Or Abs(gTY(i) - y0) <= dyMax * 1.5 Then
                    If gTX(i) - x0 < best Then best = gTX(i) - x0: TagCanhPhai = gTX(i)
                End If
            End If
        End If
    Next i
End Function

' Dat level cho so lieu bang theo level cua tag trong mau

    If Not gLvHSMap Is Nothing Then

' Sap tag bang: theo COT (trai -> phai, sai so tol), trong cot tu tren xuong
Private Sub SapCotHang(ByRef ax() As Double, ByRef ay() As Double, ByVal c As Long, ByVal tol As Double)
    Dim i As Long, j As Long, x As Double, y As Double, truoc As Boolean
    If tol <= 0 Then tol = 1#
    For i = 2 To c
        x = ax(i): y = ay(i): j = i - 1
        Do While j >= 1
            If Abs(ax(j) - x) > tol Then
                truoc = (ax(j) > x)
            Else
                truoc = (ay(j) < y)
            End If
            If Not truoc Then Exit Do
            ax(j + 1) = ax(j): ay(j + 1) = ay(j): j = j - 1
        Loop
        ax(j + 1) = x: ay(j + 1) = y
    Next i
End Sub

' Tam cot: 2 duong ke DOC gan nhat (trai/phai) cat qua hang y. w = be rong cot.
Private Function TamCot(ByVal x As Double, ByVal y As Double, ByRef w As Double) As Double
    Dim i As Long, j As Long, m As Long, co() As Double
    Dim ax As Double, ay As Double, bx As Double, by As Double, xx As Double
    Dim xl As Double, xr As Double
    TamCot = x: w = 0#
    xl = -1E+30: xr = 1E+30
    For i = 1 To gNL
        co = gLineCoords(i)
        m = (UBound(co) + 1) \ 2
        For j = 0 To m - 2
            ax = co(2 * j): ay = co(2 * j + 1): bx = co(2 * j + 2): by = co(2 * j + 3)
            If Abs(bx - ax) <= 0.01 * Abs(by - ay) And Abs(by - ay) > 0.000001 Then
                If (y - ay) * (y - by) <= 0# Then
                    xx = (ax + bx) / 2#
                    If xx <= x And xx > xl Then xl = xx
                    If xx > x And xx < xr Then xr = xx
                End If
            End If
        Next j
    Next i
    If xl > -1E+29 And xr < 1E+29 Then
        If (xr - xl) < 1000# Then TamCot = (xl + xr) / 2#: w = xr - xl
    End If
End Function

Private Sub VeThuaTrongO(ByVal v As clsVung, ByVal cx As Double, ByVal cy As Double, _
                         ByVal kp As Double, ByVal cao As Double)

    ring = v.ring: n = UBound(ring, 1)
    If n < 3 Then Exit Sub
    If cao <= 0 Then cao = 0.5
    Dim rMoc As Double: rMoc = cao * 0.5

    Dim mx As Double, my As Double, dai As Double

    DatLv mLvRanh                     ' canh thua
    For i = 0 To n - 1
        ax = cx + (ring(i, 0) - v.cx) * kp: ay = cy + (ring(i, 1) - v.cy) * kp

        For i = 0 To m - 1
            ax = cx + (dr(i, 0) - v.cx) * kp: ay = cy + (dr(i, 1) - v.cy) * kp
            bx = cx + (dr(i + 1, 0) - v.cx) * kp: by = cy + (dr(i + 1, 1) - v.cy) * kp

        Next i
    Next k2

    ' so do (chieu dai da co trong bang toa do). Chi giu SO HIEU DINH + nhan thua.

    ' SUA 12/10/2026: so hieu dinh = 80% co chu nhan thua (nhan = cao * 0.84)
    Dim caoS As Double: caoS = cao * 0.84 * 0.8
    ' SUA 15/10/2026: so hieu dinh dat RA NGOAI thua theo phan giac goc
    Dim hx As Double, hy As Double, kc As Double

            HuongNgoai dr, m, i, True, hx, hy
            kc = caoS * (0.75 + 0.3 * Len(CStr(soTiep + i + 1)))
            VeText ax + hx * kc, ay + hy * kc, CStr(soTiep + i + 1), caoS, -1
        Next i
        soTiep = soTiep + m
    Next k2

    ' SUA 05/08/2026: NHAN GIUA THUA (giong gCadas):  LoaiDat  SoThua/DienTich
    '   vd:  CLN 78/84,2   (dien tich tu dong = v.Dt, lam tron 1 so le)

    VeNhanPhanSo lefX + wLoai + gap + wFrac / 2#, cy, sSo, sDt, caoN, -1
End Sub

' Huong don vi tu dinh i ra NGOAI thua (vong dao: vao trong lo)
Private Sub HuongNgoai(r() As Double, ByVal n As Long, ByVal i As Long, ByVal dao As Boolean, dx As Double, dy As Double)
    Dim p As Long, q As Long, ux As Double, uy As Double, wx As Double, wy As Double, L As Double, e As Double
    p = i - 1: If p < 0 Then p = n - 1
    q = i + 1
    ux = r(i, 0) - r(p, 0): uy = r(i, 1) - r(p, 1): L = Sqr(ux * ux + uy * uy): If L > 0 Then ux = ux / L: uy = uy / L
    wx = r(i, 0) - r(q, 0): wy = r(i, 1) - r(q, 1): e = Sqr(wx * wx + wy * wy): If e > 0 Then wx = wx / e: wy = wy / e
    If e > L Then e = L
    dx = ux + wx: dy = uy + wy: L = Sqr(dx * dx + dy * dy)
    If L < 0.2 Then dx = -uy: dy = ux: L = 1
    dx = dx / L: dy = dy / L
    e = e * 0.05
    If DiemTrongPolygon(r, r(i, 0) + dx * e, r(i, 1) + dy * e) Xor dao Then dx = -dx: dy = -dy
End Sub

' Mau tu .dgn: moi loai tag (tttl, tamYtl, tamXtl, cdtl) nam trong 1 cot;
' cot = khoang giua 2 duong ke DOC gan nhat (trai/phai) cua bang.
' (tamYtl nam o cot X(m), tamXtl o cot Y(m) - giong quy uoc mau cu)
' Kiem chung: mau cu tinh ra dung 8 so trong DatCotMacDinh.
Private Sub TinhCotTuLuoi()
    Dim xT(0 To 3) As Double, t As Long, i As Long, j As Long
    Dim y1 As Double, y2 As Double, coY As Boolean
    For t = 0 To 3: xT(t) = 1E+30: Next t
    For i = 1 To gNT
        t = -1
        Select Case gTS(i)
            Case "tttl": t = 0
            Case "tamYtl": t = 1
            Case "tamXtl": t = 2
            Case "cdtl": t = 3
        End Select
        If t >= 0 Then
            If gTX(i) < xT(t) Then xT(t) = gTX(i)
        End If
    Next i
    If xT(0) > 1E+29 Then Exit Sub
    For i = 1 To gNT
        If gTS(i) = "tttl" And Abs(gTX(i) - xT(0)) < 5# Then
            If Not coY Then
                y1 = gTY(i): y2 = gTY(i): coY = True
            Else
                If gTY(i) < y1 Then y1 = gTY(i)
                If gTY(i) > y2 Then y2 = gTY(i)
            End If
        End If
    Next i
    If Not coY Or y2 - y1 <= 0# Then Exit Sub

    ' Duong ke doc cat qua it nhat nua khoang cac hang cua bang
    Dim vx() As Double, nv As Long, c() As Double, m As Long
    Dim lo As Double, hi As Double, tg As Double
    ReDim vx(1 To 2000)
    For i = 1 To gNL
        c = gLineCoords(i): m = (UBound(c) + 1) \ 2
        For j = 0 To m - 2
            If Abs(c(2 * j) - c(2 * j + 2)) < 0.5 Then
                lo = c(2 * j + 1): hi = c(2 * j + 3)
                If lo > hi Then tg = lo: lo = hi: hi = tg
                If lo < y1 Then lo = y1
                If hi > y2 Then hi = y2
                If hi - lo > 0.5 * (y2 - y1) And nv < 2000 Then
                    nv = nv + 1: vx(nv) = c(2 * j)
                End If
            End If
        Next j
    Next i
    If nv < 2 Then Exit Sub

    Dim xl As Double, xr As Double
    For t = 0 To 3
        If xT(t) < 1E+29 Then
            xl = -1E+30: xr = 1E+30
            For j = 1 To nv
                If vx(j) <= xT(t) + 1# And vx(j) > xl Then xl = vx(j)
                If vx(j) > xT(t) + 1# And vx(j) < xr Then xr = vx(j)
            Next j
            If xl > -1E+29 And xr < 1E+29 Then
                Select Case t
                    Case 0: CX_DINH = (xl + xr) / 2#: CW_DINH = xr - xl
                    Case 1: CX_X = (xl + xr) / 2#: CW_X = xr - xl
                    Case 2: CX_Y = (xl + xr) / 2#: CW_Y = xr - xl
                    Case 3: CX_CANH = (xl + xr) / 2#: CW_CANH = xr - xl
                End Select
            End If
        End If
    Next t
End Sub

' BAN VE TACH / HOP THUA (Mau 22/DK - 22tach.dgn / 22hop.dgn)
'   O so do TRUOC = level 25, o so do SAU = level 24 (dat ngay tai thua).
'   Bang "Doan | Chieu dai" = chu mau "tt" / "cd" (trang 2 roi trang 3).

Public Sub ChenKhungTachHop(ByVal cheDo As Long, ByVal goc As Collection, ByVal sau As Collection, _
                            ByVal vChinh As clsVung, ByVal mauPath As String, ByVal toBanDo As String, _
                            ByVal tenXa As String, ByVal tenHuyen As String, ByVal tenTinh As String, _
                            ByVal tyLe As Double)
    If Not NapMau(mauPath) Then
        MsgBox "Khong doc duoc file mau: " & mauPath, vbExclamation, "Tach/hop thua": Exit Sub
    End If
    Dim cxb As Double, cyb As Double, bw As Double, bh As Double
    If Not TimBox(cxb, cyb, bw, bh) Then
        MsgBox "Khong tim thay o so do SAU (level 24) trong mau.", vbExclamation, "Tach/hop thua": Exit Sub
    End If
    Dim cxt As Double, cyt As Double, bwt As Double, bht As Double
    If Not TimBoxLv(25, cxb, cyb, cxt, cyt, bwt, bht) Then
        cxt = cxb: cyt = cyb: bwt = bw: bht = bh       ' khong co o truoc -> ve chung o
    End If

    Dim lvCu As Level
    Set lvCu = ActiveSettings.Level
    ChuanBiLevel
    If Not gLvHSMot Is Nothing Then DatLv gLvHSMot

    ' Ty le khung + ty le so do (giong trich luc)
    If tyLe <= 0 Then tyLe = 500
    Dim k As Double, kp As Double, bwm As Double, bhm As Double
    k = tyLe / 20000#
    bwm = bw * k: bhm = bh * k
    Dim rg() As Double, x1 As Double, y1 As Double, x2 As Double, y2 As Double
    rg = vChinh.ring
    BaoRing rg, x1, y1, x2, y2
    Dim pw As Double, ph As Double, wMin As Double, hMin As Double
    pw = x2 - x1: ph = y2 - y1
    wMin = bw: If bwt < wMin Then wMin = bwt
    hMin = bh: If bht < hMin Then hMin = bht
    Dim buoc As Variant, bi As Long
    buoc = Array(1#, 2#, 2.5, 5#, 10#, 20#, 25#, 50#, 100#)
    kp = 1#
    If gTyLeTuDong Then
        For bi = 0 To UBound(buoc)
            kp = 1# / buoc(bi)
            If pw * kp <= wMin * k * 0.85 And ph * kp <= hMin * k * 0.85 Then Exit For
        Next bi
    End If
    Dim gx As Double, gy As Double               ' tam khung bao thua chinh
    gx = (x1 + x2) / 2#: gy = (y1 + y2) / 2#

    ' Dat o so do SAU ngay tai thua
    Dim Cpx As Double, Cpy As Double
    Cpx = gx: Cpy = gy
    gTLCx = Cpx: gTLCy = Cpy

    ' 1) Chep mau .dgn (modMauDgn thay chu mau qua modTachHop.TH_ThayChu)
    Dim chep As Boolean
    chep = (gTuDgn And gScDgn > 0 And Len(gDgnPath) > 0)
    If chep Then chep = (Len(Dir$(gDgnPath)) > 0)
    If chep Then
        DC_ChepMauDgn gDgnPath, gScDgn * k, Cpx - cxb * k, Cpy - cyb * k, goc.Item(1), _
                      toBanDo, tenXa, tenHuyen, tenTinh
    Else
        VeKhungTuCache cxb, cyb, k, Cpx, Cpy
    End If

    ' 2) So do TRUOC (o level 25) va SAU (o level 24)
    Dim cao As Double
    cao = bwm * 0.03
    Dim tx As Double, ty As Double
    tx = Cpx + (cxt - cxb) * k: ty = Cpy + (cyt - cyb) * k
    VeSoDoTH goc, tx, ty, gx, gy, kp, cao, False
    VeSoDoTH sau, Cpx, Cpy, gx, gy, kp, cao, True

    ' 3) Bang Doan / Chieu dai
    DienBangDoan sau, cxb, cyb, k, Cpx, Cpy

    Set gLvHSMot = Nothing
    Set gLvHSMap = Nothing
    DatLv lvCu
End Sub

' O so do tren level lv (lon nhat, khac o (cxb,cyb))
Private Function TimBoxLv(ByVal lv As Long, ByVal cxb As Double, ByVal cyb As Double, _
                          ByRef cx As Double, ByRef cy As Double, _
                          ByRef bw As Double, ByRef bh As Double) As Boolean
    Dim i As Long, j As Long, co() As Double, m As Long, bestA As Double
    Dim x1 As Double, y1 As Double, x2 As Double, y2 As Double
    bestA = -1
    For i = 1 To gNL
        If gLineLevel(i) = lv Then
            co = gLineCoords(i)
            m = (UBound(co) + 1) \ 2
            If m >= 4 Then
                x1 = co(0): x2 = co(0): y1 = co(1): y2 = co(1)
                For j = 0 To m - 1
                    If co(2 * j) < x1 Then x1 = co(2 * j)
                    If co(2 * j) > x2 Then x2 = co(2 * j)
                    If co(2 * j + 1) < y1 Then y1 = co(2 * j + 1)
                    If co(2 * j + 1) > y2 Then y2 = co(2 * j + 1)
                Next j
                If (x2 - x1) * (y2 - y1) > bestA Then
                    bestA = (x2 - x1) * (y2 - y1)
                    cx = (x1 + x2) / 2#: cy = (y1 + y2) / 2#: bw = x2 - x1: bh = y2 - y1
                End If
            End If
        End If
    Next i
    TimBoxLv = (bestA > 0)
End Function

' Mau .txt (khong chep duoc .dgn): ve duong + chu tu cache
Private Sub VeKhungTuCache(ByVal cxb As Double, ByVal cyb As Double, ByVal k As Double, _
                           ByVal Cpx As Double, ByVal Cpy As Double)
    Dim i As Long, j As Long, co() As Double, m As Long
    Dim ax As Double, ay As Double, bx As Double, by As Double
    For i = 1 To gNL
        co = gLineCoords(i)
        m = (UBound(co) + 1) \ 2
        For j = 0 To m - 2
            ax = (co(2 * j) - cxb) * k + Cpx: ay = (co(2 * j + 1) - cyb) * k + Cpy
            bx = (co(2 * j + 2) - cxb) * k + Cpx: by = (co(2 * j + 3) - cyb) * k + Cpy
            If Abs(bx - ax) > 0.000001 Or Abs(by - ay) > 0.000001 Then VeLine ax, ay, bx, by, -1
        Next j
    Next i
    For i = 1 To gNT
        If gTKind(i) = "TEXT" And gTCao(i) > 0 Then
            VeTextMau (gTX(i) - cxb) * k + Cpx, (gTY(i) - cyb) * k + Cpy, gTS(i), gTCao(i) * k
        End If
    Next i
End Sub

' Ve 1 so do: canh cac thua (canh chung ve 1 lan), so hieu dinh, nhan thua,
' (coKichThuoc) chieu dai canh doc theo canh.
'   Diem P (ban do) -> (ox, oy) + (P - (gx, gy)) * kp
Private Sub VeSoDoTH(ByVal ds As Collection, ByVal ox As Double, ByVal oy As Double, _
                     ByVal gx As Double, ByVal gy As Double, ByVal kp As Double, _
                     ByVal cao As Double, ByVal coKichThuoc As Boolean)
    Dim daVe As Object, daSo As Object
    Set daVe = CreateObject("Scripting.Dictionary")
    Set daSo = CreateObject("Scripting.Dictionary")
    Dim j As Long, kv As Long, i As Long, m As Long, v As clsVung, rg() As Double
    Dim ax As Double, ay As Double, bx As Double, by As Double, key As String, s As String
    Dim rMoc As Double: rMoc = cao * 0.5

    For j = 1 To ds.Count
        Set v = ds.Item(j)
        For kv = 0 To v.Dao.Count
            If kv = 0 Then rg = v.ring Else rg = v.Dao.Item(kv)
            m = UBound(rg, 1)
            For i = 0 To m - 1
                key = KhoaCanh(rg(i, 0), rg(i, 1), rg(i + 1, 0), rg(i + 1, 1))
                ax = ox + (rg(i, 0) - gx) * kp: ay = oy + (rg(i, 1) - gy) * kp
                bx = ox + (rg(i + 1, 0) - gx) * kp: by = oy + (rg(i + 1, 1) - gy) * kp
                If Not daVe.Exists(key) Then
                    daVe.Add key, True
                    DatLv mLvRanh
                    VeLine ax, ay, bx, by, -1
                    ' SUA 11/10/2026: khong ghi chieu dai canh tren so do (da co bang 3.3)
                End If
                s = TH_SoDinh(rg(i, 0), rg(i, 1), coKichThuoc)
                If Len(s) > 0 And Not daSo.Exists(s) Then
                    daSo.Add s, True
                    DatLv mLvSo
                    VeText ax + rMoc * 1.6, ay + rMoc * 1.6, s, cao * 0.84 * 0.8, -1
                End If
            Next i
        Next kv
    Next j

    ' Nhan thua: LoaiDat  SoThua/DienTich (giong trich luc)
    DatLv mLvNhan
    Dim caoN As Double, sLoai As String, sSo As String, sDt As String
    Dim wLoai As Double, wFrac As Double, gap As Double, lefX As Double, cx As Double, cy As Double
    caoN = cao * 0.84
    For j = 1 To ds.Count
        Set v = ds.Item(j)
        cx = ox + (v.cx - gx) * kp: cy = oy + (v.cy - gy) * kp
        sLoai = Trim$(v.loaiDat): sSo = Trim$(v.soThua): sDt = FNum(v.Dt, 1)
        wLoai = Len(sLoai) * caoN * 0.6
        wFrac = Len(sDt) * caoN * 0.6
        If Len(sSo) * caoN * 0.6 > wFrac Then wFrac = Len(sSo) * caoN * 0.6
        gap = caoN * 0.4
        lefX = cx - (wLoai + gap + wFrac) / 2#
        If Len(sLoai) > 0 Then VeText lefX + wLoai / 2#, cy, sLoai, caoN, -1
        VeNhanPhanSo lefX + wLoai + gap + wFrac / 2#, cy, sSo, sDt, caoN, -1
    Next j
End Sub

' Chieu dai canh: chu xoay theo canh, dat giua canh lech sang ben phai huong ve
Private Sub VeChieuDai(ByVal ax As Double, ByVal ay As Double, ByVal bx As Double, ByVal by As Double, _
                       ByVal dai As Double, ByVal h As Double)
    Dim dx As Double, dy As Double, L As Double, ang As Double, mx As Double, my As Double
    dx = bx - ax: dy = by - ay
    L = Sqr(dx * dx + dy * dy)
    If L < h * 2# Then Exit Sub                      ' canh qua ngan tren so do
    ang = Atan2(dy, dx)
    If ang > 1.5708 Then ang = ang - 3.14159265358979
    If ang <= -1.5708 Then ang = ang + 3.14159265358979
    mx = (ax + bx) / 2# + dy / L * h * 0.8          ' phap tuyen phai (ra ngoai vong CCW)
    my = (ay + by) / 2# - dx / L * h * 0.8
    On Error Resume Next
    ActiveSettings.TextStyle.Height = h
    ActiveSettings.TextStyle.Width = h * C_TILE_W
    ActiveSettings.TextStyle.Justification = msdTextJustificationCenterCenter
    Dim oT As TextElement
    Set oT = CreateTextElement1(Nothing, FNum(dai, 2), Point3dFromXYZ(mx, my, 0#), _
                                Matrix3dFromAxisAndRotationAngle(2, ang))
    If Not oT Is Nothing Then oT.LineWeight = 0: ActiveModelReference.AddElement oT
    On Error GoTo 0
End Sub

Private Function KhoaCanh(ByVal ax As Double, ByVal ay As Double, ByVal bx As Double, ByVal by As Double) As String
    Dim a As String, b As String
    a = Format$(Round(ax, 3), "0.000") & "|" & Format$(Round(ay, 3), "0.000")
    b = Format$(Round(bx, 3), "0.000") & "|" & Format$(Round(by, 3), "0.000")
    If a < b Then KhoaCanh = a & "#" & b Else KhoaCanh = b & "#" & a
End Function

' Bang "Doan | Chieu dai": moi thua sau 1 dong tieu de (khi > 1 thua) roi cac canh
' "1-2'" | "12,34". O = chu mau "tt" (cung hang voi "cd"), sap theo cot roi tren xuong.
Private Sub DienBangDoan(ByVal sau As Collection, ByVal cxb As Double, ByVal cyb As Double, _
                         ByVal k As Double, ByVal Cpx As Double, ByVal Cpy As Double)
    Dim tx() As Double, ty() As Double, ct As Long, i As Long, hT As Double, lvT As Long
    ReDim tx(1 To gNT + 1): ReDim ty(1 To gNT + 1)
    For i = 1 To gNT
        If gTS(i) = "tt" And gTKind(i) = "TAG" Then
            ct = ct + 1: tx(ct) = gTX(i): ty(ct) = TamHang(gTX(i), gTY(i))
            hT = gTCao(i): lvT = gTL(i)
        End If
    Next i
    If ct = 0 Then Exit Sub
    SapCotHang tx, ty, ct, hT * 2#
    DatLvTag lvT

    ' danh sach dong: moi canh 1 lan (ranh ngoai roi duong cat) - modTachHop
    Dim dTT As New Collection, dCD As New Collection, j As Long, sD As String, dd As Double
    For j = 1 To TH_SoDoan()
        TH_Doan j, sD, dd
        dTT.Add sD: dCD.Add FNum(dd, 2)
    Next j

    Dim w As Double, xc As Double, xT As Double, hc As Double, r As Long
    hc = hT * k
    For r = 1 To dTT.Count
        If r > ct Then
            MsgBox "Bang chieu dai canh khong du cho (" & dTT.Count & " dong, mau co " & ct & " o).", vbExclamation, "Tach/hop thua"
            Exit For
        End If
        xc = TamCot(tx(r), ty(r), w)
        DienCell xc, ty(r), dTT.Item(r), hc, cxb, cyb, k, Cpx, Cpy, w * k
        xT = TagCanhPhai("cd", tx(r), ty(r), hT)
        If xT > -1E+29 Then
            xc = TamCot(xT, ty(r), w)
            DienCell xc, ty(r), dCD.Item(r), hc, cxb, cyb, k, Cpx, Cpy, w * k
        End If
    Next r
End Sub
