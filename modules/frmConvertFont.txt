Private Sub NapLevel(ByVal cL As Object)
    Dim lv As Level, n As Long, i As Long, j As Long, t As Long, ts As String
    Dim so() As Long, ten() As String
    cL.Clear
    cL.AddItem "[T" & ChrW(7845) & "t c" & ChrW(7843) & " c" & ChrW(225) & "c l" & ChrW(7899) & "p]"
    ReDim mLv(0 To 0): mLv(0) = -1
    On Error Resume Next
    ReDim so(0 To 0): ReDim ten(0 To 0)
    For Each lv In ActiveDesignFile.Levels
        n = n + 1
        ReDim Preserve so(0 To n): ReDim Preserve ten(0 To n)
        so(n) = lv.Number: ten(n) = lv.Name
    Next lv
    ' sap xep theo so level
    For i = 1 To n - 1
        For j = i + 1 To n
            If so(j) < so(i) Then
                t = so(i): so(i) = so(j): so(j) = t
                ts = ten(i): ten(i) = ten(j): ten(j) = ts
            End If
        Next j
    Next i
    ReDim Preserve mLv(0 To n)
    For i = 1 To n
        cL.AddItem so(i) & " - " & ten(i)
        mLv(i) = so(i)
    Next i
    cL.ListIndex = 0
End Sub

Private Function Lst() As Object
    Set Lst = Me.Controls("lstFile")
End Function

Private Function ThemFile(ByVal p As String) As Boolean
    Dim k As String
    p = Trim$(p)
    If Len(p) = 0 Then Exit Function
    k = LCase$(p)
    If mDs.Exists(k) Then Exit Function
    mDs.Add k, True
    Lst.AddItem p
    Lst.List(Lst.ListCount - 1, 1) = ""
    ThemFile = True
End Function

Private Sub CapNhatDem()
    Me.Controls("lblDem").Caption = " " & Lst.ListCount & " file"
    If Lst.ListCount > 0 And Lst.ListIndex < 0 Then
        Me.Controls("lblPath").Caption = " " & Lst.List(0, 0)
    ElseIf Lst.ListCount = 0 Then
        Me.Controls("lblPath").Caption = ""
    End If
End Sub

Private Sub ThemTuDs(ByVal ds As Collection)
    Dim i As Long, n As Long
    For i = 1 To ds.Count
        If ThemFile(CStr(ds.Item(i))) Then n = n + 1
    Next i
    CapNhatDem
    Me.Controls("lblPath").Caption = " " & ChrW(272) & ChrW(227) & " th" & ChrW(234) & "m " & n & " file."
End Sub

Public Sub XuLyNut(ByVal ma As String)
    If mDangChay Then Exit Sub
    Dim ds As Collection, p As String, i As Long
    Select Case ma
    Case "FILE"
        ThemTuDs CF_ChonNhieuFile()
    Case "THUMUC", "THUMUCCON"
        If ma = "THUMUC" Then
            p = CF_ChonThuMuc("Ch" & ChrW(7885) & "n th" & ChrW(432) & " m" & ChrW(7909) & "c ch" & ChrW(7913) & "a b" & ChrW(7843) & "n " & ChrW(273) & ChrW(7891) & " (*.dgn)")
        Else
            p = CF_ChonThuMuc("Ch" & ChrW(7885) & "n th" & ChrW(432) & " m" & ChrW(7909) & "c (l" & ChrW(7845) & "y c" & ChrW(7843) & " th" & ChrW(432) & " m" & ChrW(7909) & "c con)")
        End If
        If Len(p) = 0 Then Exit Sub
        Set ds = New Collection
        Me.Controls("lblPath").Caption = " " & ChrW(272) & "ang t" & ChrW(236) & "m file trong " & p & " ..."
        DoEvents
        CF_GomFileDgn p, (ma = "THUMUCCON"), ds
        ThemTuDs ds
    Case "HIENTAI"
        On Error Resume Next
        p = ActiveDesignFile.FullName
        On Error GoTo 0
        If Len(p) > 0 Then ThemFile p
        CapNhatDem
    Case "BOCHON"
        For i = Lst.ListCount - 1 To 0 Step -1
            If Lst.Selected(i) Then
                If mDs.Exists(LCase$(Lst.List(i, 0))) Then mDs.Remove LCase$(Lst.List(i, 0))
                Lst.RemoveItem i
            End If
        Next i
        CapNhatDem
    Case "XOAHET"
        Lst.Clear
        mDs.RemoveAll
        CapNhatDem
    Case "CONVERT"
        ChayConvert
    End Select
End Sub

Private Sub ChayConvert()
    Dim n As Long, i As Long, cheDo As Long, lvSo As Long, bak As Boolean
    Dim kq As String, soChu As Long, tong As Long, nOK As Long, loi As String, nLoi As Long
    n = Lst.ListCount
    If n = 0 Then
        DC_ThongBao "Danh s" & ChrW(225) & "ch ch" & ChrW(432) & "a c" & ChrW(243) & " file n" & ChrW(224) & "o.", "Convert Font", vbExclamation
        Exit Sub
    End If
    cheDo = Me.Controls("cboKieu").ListIndex
    If cheDo < 0 Then cheDo = 0
    i = Me.Controls("cboLop").ListIndex
    If i < 0 Then i = 0
    lvSo = mLv(i)
    bak = CBool(Me.Controls("chkBak").Value)

    If DC_ThongBao("Chuy" & ChrW(7875) & "n font " & Me.Controls("cboKieu").Text & vbCrLf & _
                   "L" & ChrW(7899) & "p: " & Me.Controls("cboLop").Text & vbCrLf & _
                   "S" & ChrW(7889) & " file: " & n & IIf(bak, " (c" & ChrW(243) & " sao l" & ChrW(432) & "u .bak)", " (KH" & ChrW(212) & "NG sao l" & ChrW(432) & "u)") & vbCrLf & vbCrLf & _
                   "C" & ChrW(225) & "c file s" & ChrW(7869) & " " & ChrW(273) & ChrW(432) & ChrW(7907) & "c ghi " & ChrW(273) & ChrW(232) & ". Ti" & ChrW(7871) & "p t" & ChrW(7909) & "c?", "Convert Font", vbYesNo Or vbQuestion) <> vbYes Then Exit Sub

    DC_GhiCauHinh "CF_KIEU", CStr(cheDo)
    DC_GhiCauHinh "CF_BAK", IIf(bak, "1", "0")

    mDangChay = True
    For i = 0 To n - 1
        Lst.List(i, 1) = ""
    Next i
    For i = 0 To n - 1
        Me.Controls("lblDem").Caption = " " & (i + 1) & "/" & n
        Me.Controls("lblPath").Caption = " " & Lst.List(i, 0)
        Lst.List(i, 1) = ChrW(273) & "ang x" & ChrW(7917) & " l" & ChrW(253) & "..."
        DoEvents
        kq = CF_XuLyFile(CStr(Lst.List(i, 0)), cheDo, lvSo, bak, soChu)
        If Len(kq) = 0 Then
            Lst.List(i, 1) = soChu & " ch" & ChrW(7919)
            If Len(CF_FileMoi) > 0 Then Lst.List(i, 1) = soChu & " ch" & ChrW(7919) & " -> " & Mid$(CF_FileMoi, InStrRev(CF_FileMoi, "\") + 1)
            tong = tong + soChu
            nOK = nOK + 1
        Else
            Lst.List(i, 1) = "L" & ChrW(7895) & "i"
            nLoi = nLoi + 1
            If nLoi <= 8 Then loi = loi & vbCrLf & "- " & Lst.List(i, 0) & ": " & kq
        End If
        DoEvents
    Next i
    mDangChay = False
    On Error Resume Next
    RedrawAllViews
    On Error GoTo 0

    Me.Controls("lblDem").Caption = " " & n & "/" & n
    Me.Controls("lblPath").Caption = " Xong: " & nOK & " file, " & tong & " ch" & ChrW(7919) & "."
    kq = ChrW(272) & ChrW(227) & " x" & ChrW(7917) & " l" & ChrW(253) & " " & nOK & "/" & n & " file, " & ChrW(273) & ChrW(7893) & "i " & tong & " ch" & ChrW(7919) & "."
    If nLoi > 0 Then
        kq = kq & vbCrLf & vbCrLf & "L" & ChrW(7895) & "i " & nLoi & " file:" & loi
        If nLoi > 8 Then kq = kq & vbCrLf & "..."
    End If
    DC_ThongBao kq, "Convert Font", IIf(nLoi > 0, vbExclamation, vbInformation)
End Sub
