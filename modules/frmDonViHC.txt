' Goi tu clsComboEvent khi doi Tinh
Public Sub TinhDoi()
    FillXa CStr(Me.Controls("cboTinh").Value)
End Sub

Private Sub FillXa(ByVal tinh As String)
    Dim cboX As Object, ds As Collection, i As Long
    Set cboX = Me.Controls("cboXa")
    cboX.Clear
    If Not DaNapMaXa() Then Exit Sub
    Set ds = DsXa(tinh)
    For i = 1 To ds.Count
        cboX.AddItem ds.Item(i)
    Next i
    If ds.Count > 0 Then cboX.ListIndex = 0
End Sub

Private Sub AddLbl(ByVal nm As String, ByVal cap As String, ByVal x As Single, ByVal y As Single)
    Dim c As Object
    Set c = Me.Controls.Add("Forms.Label.1", nm, True)
    c.Caption = cap: c.Left = x: c.Top = y: c.Width = 100: c.Height = 14
    c.Font.Name = "Arial": c.Font.size = 8
End Sub

Private Sub ThemNut(ByVal ma As String, ByVal cap As String, ByVal x As Single, ByVal y As Single)
    Dim b As Object, hd As clsNutTL
    Set b = Me.Controls.Add("Forms.CommandButton.1", "cmd" & ma, True)
    b.Caption = cap: b.Left = x: b.Top = y: b.Width = 80: b.Height = 24
    b.Font.Name = "Arial": b.Font.size = 8
    Set hd = New clsNutTL
    Set hd.btn = b: Set hd.Form = Me: hd.ma = ma
    mNut.Add hd
End Sub

Public Sub XuLyNut(ByVal ma As String)
    If ma = "DONG" Then Unload Me: Exit Sub
    If ma = "LUU" Then
        If Len(Trim$(CStr(Me.Controls("cboXa").Value))) = 0 Then
            DC_ThongBao "Ch" & ChrW(432) & "a ch" & ChrW(7885) & "n x" & ChrW(227) & " / ph" & ChrW(432) & ChrW(7901) & "ng.", ChrW(272) & ChrW(417) & "n v" & ChrW(7883) & " h" & ChrW(224) & "nh ch" & ChrW(237) & "nh", vbExclamation
            Exit Sub
        End If
        DC_GhiCauHinh "TINH", Trim$(CStr(Me.Controls("cboTinh").Value))
        DC_GhiCauHinh "XA", Trim$(CStr(Me.Controls("cboXa").Value))
        DC_GhiCauHinh "HUYEN", Trim$(Me.Controls("txtHuyen").Text)
        If Val(CStr(Me.Controls("cboKTT").Value)) >= 100 Then
            DC_GhiCauHinh "KTT", CStr(Val(CStr(Me.Controls("cboKTT").Value)))
        End If
        DC_ThongBao ChrW(272) & ChrW(227) & " l" & ChrW(432) & "u " & ChrW(273) & ChrW(417) & "n v" & ChrW(7883) & " h" & ChrW(224) & "nh ch" & ChrW(237) & "nh:" & vbCrLf & _
                    Me.Controls("cboXa").Value & " - " & Me.Controls("cboTinh").Value, _
                    ChrW(272) & ChrW(417) & "n v" & ChrW(7883) & " h" & ChrW(224) & "nh ch" & ChrW(237) & "nh"
        Unload Me
    End If
End Sub
