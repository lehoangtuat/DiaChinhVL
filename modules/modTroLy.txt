Private Function DongTu() As Variant
    DongTu = Array(" tach thua", " hop thua", " tao vung", " tao topo", " tao thua", " gan ", " trich luc", " to mau", " xuat polygon", _
                   " xuat poly", " tim loi", " chieu dai", " bat lop", " tat lop", " luu ho so", " luu file ho so")
End Function

Private Function TachLenh(ByVal s As String) As Collection
    Dim c As New Collection, vt() As Long, n As Long, i As Long, j As Long, p As Long, dt As Variant, t As Long
    ReDim vt(1 To 200)
    s = Replace(Replace(s, ";", " ; "), ",", " , ")
    s = Replace(s, " roi ", " ; "): s = Replace(s, " sau do ", " ; "): s = Replace(s, " xong ", " ; ")
    For Each dt In DongTu()
        p = 1
        Do
            p = InStr(p, s, dt)
            If p = 0 Then Exit Do
            ' bo trung (vd " xuat poly" nam trong " xuat polygon", " luu ho so" trong " luu file ho so")
            For j = 1 To n
                If Abs(vt(j) - p) < 4 Then Exit For
            Next j
            If j > n And n < 200 Then n = n + 1: vt(n) = p
            p = p + Len(dt)
        Loop
    Next dt
    For i = 2 To n
        t = vt(i): j = i - 1
        Do While j >= 1
            If vt(j) <= t Then Exit Do
            vt(j + 1) = vt(j): j = j - 1
        Loop
        vt(j + 1) = t
    Next i
    For i = 1 To n
        If i < n Then c.Add Mid$(s, vt(i), vt(i + 1) - vt(i)) Else c.Add Mid$(s, vt(i))
    Next i
    Set TachLenh = c
End Function

'------------------------------------------------------------ doc so
' Cac so dung sau tu khoa (vd "lop" -> "10,12")
Private Function SoSau(ByVal s As String, ByVal tuKhoa As String) As String
    Dim p As Long, i As Long, ch As String, kq As String, so As String, dangDoc As Boolean
    p = InStr(s, " " & tuKhoa & " ")
    If p = 0 Then Exit Function
    p = p + Len(tuKhoa) + 2
    For i = p To Len(s)
        ch = Mid$(s, i, 1)
        If ch Like "[0-9]" Then
            so = so & ch: dangDoc = True
        ElseIf ch = " " Or ch = "," Or ch = "+" Or ch = "-" Then
            If Len(so) > 0 Then kq = kq & IIf(Len(kq) > 0, ",", "") & so: so = ""
        ElseIf Mid$(s, i, 2) = "va" Then
            i = i + 1
        Else
            Exit For
        End If
    Next i
    If Len(so) > 0 Then kq = kq & IIf(Len(kq) > 0, ",", "") & so
    SoSau = kq
End Function

Private Function DsLevel(ByVal so As String) As String
    Dim a() As String, i As Long, kq As String
    If Len(so) = 0 Then Exit Function
    a = Split(so, ",")
    For i = 0 To UBound(a)
        kq = kq & IIf(Len(kq) > 0, ",", "") & "Level " & a(i)
    Next i
    DsLevel = kq
End Function

Private Function DauSo(ByVal so As String) As String
    If InStr(so, ",") > 0 Then DauSo = Left$(so, InStr(so, ",") - 1) Else DauSo = so
End Function
