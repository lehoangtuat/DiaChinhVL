' Bam 1 dong -> zoom + to do vung
Public Sub DongChon()
    Dim r As Long, a As Variant, v As clsVung, rg() As Double, x1 As Double, y1 As Double, x2 As Double, y2 As Double, d As Double
    r = Me.Controls("lstLoi").ListIndex
    If r < 0 Or KV_DsLoi Is Nothing Then Exit Sub
    If r + 1 > KV_DsLoi.Count Then Exit Sub
    a = KV_DsLoi.Item(r + 1): Set v = a(0)
    On Error Resume Next
    rg = v.ring
    BaoRing rg, x1, y1, x2, y2
    d = x2 - x1: If y2 - y1 > d Then d = y2 - y1
    If d < 20 Then d = 20
    DC_ZoomVung (x1 + x2) / 2 - d * 1.5, (y1 + y2) / 2 - d * 1.5, (x1 + x2) / 2 + d * 1.5, (y1 + y2) / 2 + d * 1.5
    DanhDau rg
End Sub

Private Sub DanhDau(rg() As Double)
    Dim pts() As Point3d, i As Long, n As Long, el As Element
    TatDanhDau
    On Error Resume Next
    n = UBound(rg, 1)
    ReDim pts(0 To n)
    For i = 0 To n
        pts(i).X = rg(i, 0): pts(i).Y = rg(i, 1): pts(i).Z = 0#
    Next i
    Set el = CreateShapeElement1(Nothing, pts)
    If el Is Nothing Then Exit Sub
    el.Color = 3: el.LineWeight = 5
    mTec.Add CreateTransientElementContainer1(el, 0, 255, msdDrawingModeNormal)
End Sub

Private Sub TatDanhDau()
    Dim i As Long
    On Error Resume Next
    If mTec Is Nothing Then Set mTec = New Collection: Exit Sub
    For i = mTec.Count To 1 Step -1
        mTec.Item(i).Reset
        mTec.Remove i
    Next i
End Sub
