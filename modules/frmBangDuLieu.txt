Private Sub NapBang()
    Dim lst As Object, path As String
    Set lst = Me.Controls("lstData")
    lst.Clear
    path = Trim$(Me.Controls("txtPath").Text)
    Set mVungs = MoCSDL(path)
    If mVungs Is Nothing Then Exit Sub
    If mVungs.Count = 0 Then
        Me.Controls("lblTong").Caption = "Ch" & ChrW(432) & "a c" & ChrW(243) & " th" & ChrW(7917) & "a n" & ChrW(224) & "o trong CSDL."
        Exit Sub
    End If

    Dim i As Long, v As clsVung, r As Long, tong As Double
    For i = 1 To mVungs.Count
        Set v = mVungs.Item(i)
        lst.AddItem CStr(i)
        r = lst.ListCount - 1
        lst.List(r, 1) = v.SoTo
        lst.List(r, 2) = v.soThua
        lst.List(r, 3) = v.loaiDat
        lst.List(r, 4) = Format$(Round(v.Dt, 1), "0.0")
        lst.List(r, 5) = v.TenChu
        lst.List(r, 6) = v.DiaChi
        tong = tong + v.Dt
    Next i

    Me.Controls("lblTong").Caption = "T" & ChrW(7893) & "ng di" & ChrW(7879) & "n t" & ChrW(237) & "ch: " & _
        Format$(tong, "#,##0.0") & " m2   (" & mVungs.Count & " th" & ChrW(7917) & "a)"
End Sub

' Xuat bang hien tai ra Excel (giu lai tuy chon cu)
Private Sub XuatExcel()
    If mVungs Is Nothing Then Exit Sub
    If mVungs.Count = 0 Then Exit Sub
    Dim xl As Object, wb As Object, ws As Object
    On Error Resume Next
    Set xl = CreateObject("Excel.Application")
    On Error GoTo 0
    If xl Is Nothing Then MsgBox "Kh" & ChrW(244) & "ng m" & ChrW(7903) & " " & ChrW(273) & ChrW(432) & ChrW(7907) & "c Excel.", vbExclamation: Exit Sub
    xl.Visible = True
    Set wb = xl.Workbooks.Add
    Set ws = wb.Worksheets(1)
    ws.Cells(1, 1).Value = "STT"
    ws.Cells(1, 2).Value = ChrW(272) & ChrW(7893) & " b" & ChrW(7843) & "n " & ChrW(273) & ChrW(7891)
    ws.Cells(1, 3).Value = "Th" & ChrW(7917) & "a"
    ws.Cells(1, 4).Value = "Lo" & ChrW(7841) & "i " & ChrW(273) & ChrW(7845) & "t"
    ws.Cells(1, 5).Value = "Di" & ChrW(7879) & "n t" & ChrW(237) & "ch (m2)"
    ws.Cells(1, 6).Value = "T" & ChrW(234) & "n ch" & ChrW(7911)
    ws.Cells(1, 7).Value = ChrW(272) & ChrW(7883) & "a ch" & ChrW(7881)
    ws.Cells(1, 8).Value = "Ma (khong sua)"
    ws.Range("A1:H1").Font.Bold = True
    ws.Range("B:D,F:G").NumberFormat = "@"           ' giu nguyen "01", "0012"...
    Dim i As Long, v As clsVung, r As Long
    For i = 1 To mVungs.Count
        Set v = mVungs.Item(i): r = i + 1
        ws.Cells(r, 1).Value = i
        ws.Cells(r, 2).Value = v.SoTo
        ws.Cells(r, 3).Value = v.soThua
        ws.Cells(r, 4).Value = v.loaiDat
        ws.Cells(r, 5).Value = Round(v.Dt, 1)
        ws.Cells(r, 6).Value = v.TenChu
        ws.Cells(r, 7).Value = v.DiaChi
        ws.Cells(r, 8).Value = MaThua(v)
    Next i
    ws.Cells(mVungs.Count + 2, 4).Value = "T" & ChrW(7893) & "ng:"
    ws.Cells(mVungs.Count + 2, 5).Formula = "=SUM(E2:E" & (mVungs.Count + 1) & ")"
    ws.Columns("A:H").AutoFit
    Set mWb = wb
    MsgBox "S" & ChrW(7917) & "a trong Excel (T" & ChrW(7901) & ", Th" & ChrW(7917) & "a, Lo" & ChrW(7841) & "i " & ChrW(273) & ChrW(7845) & "t, Ch" & ChrW(7911) & ", " & ChrW(272) & ChrW(7883) & "a ch" & ChrW(7881) & "), xong b" & ChrW(7845) & "m 'N" & ChrW(7841) & "p Excel' " & ChrW(273) & ChrW(7875) & " ghi l" & ChrW(7841) & "i CSDL." & vbCrLf & _
           "(C" & ChrW(243) & " th" & ChrW(7875) & " l" & ChrW(432) & "u file Excel r" & ChrW(7891) & "i n" & ChrW(7841) & "p sau)", vbInformation, "Bang du lieu"
End Sub

' Bam 1 dong -> dua len o sua
Public Sub DongChon()
    Dim lst As Object, r As Long, v As clsVung
    Set lst = Me.Controls("lstData")
    r = lst.ListIndex
    If r < 0 Or mVungs Is Nothing Then Exit Sub
    If r + 1 > mVungs.Count Then Exit Sub
    Set v = mVungs.Item(r + 1)
    Me.Controls("txtTo").Text = v.SoTo
    Me.Controls("txtThua").Text = v.soThua
    Me.Controls("txtLoai").Text = v.loaiDat
    Me.Controls("txtChu").Text = v.TenChu
    Me.Controls("txtDc").Text = v.DiaChi
    ZoomThua v
End Sub

' SUA 13/10/2026: bam dong -> zoom view toi thua (thua o giua, rong gap ~3 lan)
Private Sub ZoomThua(ByVal v As clsVung)
    Dim x1 As Double, y1 As Double, x2 As Double, y2 As Double
    Dim rg() As Double, i As Long, d As Double
    On Error Resume Next
    x1 = v.cx: x2 = v.cx: y1 = v.cy: y2 = v.cy
    If v.HasRing Then
        rg = v.ring
        For i = LBound(rg, 1) To UBound(rg, 1)
            If rg(i, 0) < x1 Then x1 = rg(i, 0)
            If rg(i, 0) > x2 Then x2 = rg(i, 0)
            If rg(i, 1) < y1 Then y1 = rg(i, 1)
            If rg(i, 1) > y2 Then y2 = rg(i, 1)
        Next i
    End If
    d = x2 - x1: If y2 - y1 > d Then d = y2 - y1
    If d < 20 Then d = 20
    DC_ZoomVung (x1 + x2) / 2 - d * 1.5, (y1 + y2) / 2 - d * 1.5, _
                (x1 + x2) / 2 + d * 1.5, (y1 + y2) / 2 + d * 1.5
End Sub

' Ghi o sua vao thua dang chon + luu CSDL
Private Sub GhiDong()
    Dim lst As Object, r As Long, v As clsVung
    Set lst = Me.Controls("lstData")
    r = lst.ListIndex
    If r < 0 Or mVungs Is Nothing Then
        DC_ThongBao "B" & ChrW(7845) & "m ch" & ChrW(7885) & "n 1 d" & ChrW(242) & "ng trong b" & ChrW(7843) & "ng tr" & ChrW(432) & ChrW(7899) & "c.", "Bang du lieu", vbExclamation: Exit Sub
    End If
    Set v = mVungs.Item(r + 1)
    v.SoTo = Trim$(Me.Controls("txtTo").Text)
    v.soThua = Trim$(Me.Controls("txtThua").Text)
    v.loaiDat = Trim$(Me.Controls("txtLoai").Text)
    v.TenChu = Trim$(Me.Controls("txtChu").Text)
    v.DiaChi = Trim$(Me.Controls("txtDc").Text)
    If Not LuuCSDL(mVungs, Trim$(Me.Controls("txtPath").Text)) Then
        MsgBox "Kh" & ChrW(244) & "ng ghi " & ChrW(273) & ChrW(432) & ChrW(7907) & "c CSDL: " & Me.Controls("txtPath").Text, vbExclamation, "Bang du lieu": Exit Sub
    End If
    lst.List(r, 1) = v.SoTo: lst.List(r, 2) = v.soThua: lst.List(r, 3) = v.loaiDat
    lst.List(r, 5) = v.TenChu: lst.List(r, 6) = v.DiaChi
    ' sang dong ke tiep cho nhap nhanh
    If r + 1 < lst.ListCount Then lst.ListIndex = r + 1
End Sub

' Nap lai tu Excel (file vua Xuat Excel, da sua) -> ghi CSDL
'   Khop theo cot H (ma toa do tam thua); khong co thi theo STT.
Private Sub NapExcel()
    If mVungs Is Nothing Then Exit Sub
    If mVungs.Count = 0 Then Exit Sub
    Dim ws As Object, xl As Object, wbMo As Object, f As String
    On Error Resume Next
    If Not mWb Is Nothing Then Set ws = mWb.Worksheets(1)
    Err.Clear
    On Error GoTo 0
    If ws Is Nothing Then
        f = DC_ChonFile("Excel (*.xlsx;*.xls)|*.xlsx;*.xls", "Chon file Excel bang du lieu", DC_DocCauHinh("BDL_EXCEL", ""))
        If Len(f) = 0 Then Exit Sub
        DC_GhiCauHinh "BDL_EXCEL", f
        On Error Resume Next
        Set xl = CreateObject("Excel.Application")
        Set wbMo = xl.Workbooks.Open(f, , True)
        On Error GoTo 0
        If wbMo Is Nothing Then MsgBox "Kh" & ChrW(244) & "ng m" & ChrW(7903) & " " & ChrW(273) & ChrW(432) & ChrW(7907) & "c file Excel.", vbExclamation, "Bang du lieu": Exit Sub
        Set ws = wbMo.Worksheets(1)
    End If

    ' tra ma -> chi so thua
    Dim ma As Object, i As Long, v As clsVung
    Set ma = CreateObject("Scripting.Dictionary")
    For i = 1 To mVungs.Count
        Set v = mVungs.Item(i)
        ma(MaThua(v)) = i
    Next i

    Dim r As Long, idx As Long, nDoi As Long, k As String, st As Variant
    r = 2
    Do
        st = ws.Cells(r, 1).Value
        If IsEmpty(st) Or Len(Trim$(CStr(st))) = 0 Then Exit Do
        If Not IsNumeric(st) Then Exit Do              ' dong "Tong:"
        idx = 0
        k = Trim$(CStr(ws.Cells(r, 8).Value))
        If Len(k) > 0 And ma.Exists(k) Then idx = ma(k) Else idx = CLng(st)
        If idx >= 1 And idx <= mVungs.Count Then
            Set v = mVungs.Item(idx)
            If CapNhat(v, ws, r) Then nDoi = nDoi + 1
        End If
        r = r + 1
    Loop
    If Not wbMo Is Nothing Then wbMo.Close False: xl.Quit

    If nDoi = 0 Then MsgBox "Kh" & ChrW(244) & "ng c" & ChrW(243) & " th" & ChrW(7917) & "a n" & ChrW(224) & "o thay " & ChrW(273) & ChrW(7893) & "i.", vbInformation, "Bang du lieu": Exit Sub
    If Not LuuCSDL(mVungs, Trim$(Me.Controls("txtPath").Text)) Then
        MsgBox "Kh" & ChrW(244) & "ng ghi " & ChrW(273) & ChrW(432) & ChrW(7907) & "c CSDL.", vbExclamation, "Bang du lieu": Exit Sub
    End If
    NapBang
    MsgBox ChrW(272) & ChrW(227) & " n" & ChrW(7841) & "p t" & ChrW(7915) & " Excel: " & nDoi & " th" & ChrW(7917) & "a thay " & ChrW(273) & ChrW(7893) & "i, " & ChrW(273) & ChrW(227) & " ghi CSDL." & vbCrLf & _
           "(V" & ChrW(7869) & " l" & ChrW(7841) & "i nh" & ChrW(227) & "n tr" & ChrW(234) & "n b" & ChrW(7843) & "n " & ChrW(273) & ChrW(7891) & ": menu B" & ChrW(7843) & "n " & ChrW(273) & ChrW(7891) & " -> G" & ChrW(225) & "n th" & ChrW(244) & "ng tin / V" & ChrW(7869) & " nh" & ChrW(227) & "n t" & ChrW(7915) & " CSDL)", vbInformation, "Bang du lieu"
End Sub

Private Function CapNhat(ByVal v As clsVung, ByVal ws As Object, ByVal r As Long) As Boolean
    Dim a As String, b As String, c As String, d As String, e As String
    a = OChu(ws.Cells(r, 2).Value): b = OChu(ws.Cells(r, 3).Value): c = OChu(ws.Cells(r, 4).Value)
    d = OChu(ws.Cells(r, 6).Value): e = OChu(ws.Cells(r, 7).Value)
    If a <> v.SoTo Or b <> v.soThua Or c <> v.loaiDat Or d <> v.TenChu Or e <> v.DiaChi Then
        v.SoTo = a: v.soThua = b: v.loaiDat = c: v.TenChu = d: v.DiaChi = e
        CapNhat = True
    End If
End Function

Private Function OChu(ByVal x As Variant) As String
    If IsEmpty(x) Or IsNull(x) Then Exit Function
    OChu = Trim$(CStr(x))
End Function

' Ma nhan dien thua: toa do tam (khong phu thuoc dau phay/cham cua may)
Private Function MaThua(ByVal v As clsVung) As String
    MaThua = Trim$(Str$(Round(v.cx, 2))) & ";" & Trim$(Str$(Round(v.cy, 2)))
End Function
