' SUA 22/10: danh dau thua dang chon: vien do dam
'   (phan tu TAM - khong ghi vao ban ve; chon dong khac / dong bang thi mat)
Private Sub DanhDauThua(ByVal v As clsVung)
    Dim rg() As Double, pts() As Point3d, i As Long, n As Long, el As Element
    Dim x1 As Double, y1 As Double, x2 As Double, y2 As Double, r As Double
    TatDanhDau
    On Error Resume Next
    x1 = v.cx: x2 = v.cx: y1 = v.cy: y2 = v.cy
    If v.HasRing Then
        rg = v.ring: n = UBound(rg, 1)
        ReDim pts(0 To n)
        For i = 0 To n
            pts(i).X = rg(i, 0): pts(i).Y = rg(i, 1): pts(i).Z = 0#
            If rg(i, 0) < x1 Then x1 = rg(i, 0)
            If rg(i, 0) > x2 Then x2 = rg(i, 0)
            If rg(i, 1) < y1 Then y1 = rg(i, 1)
            If rg(i, 1) > y2 Then y2 = rg(i, 1)
        Next i
        Set el = CreateShapeElement1(Nothing, pts)
        If Not el Is Nothing Then
            el.Color = 3: el.LineWeight = 5
            mTec.Add CreateTransientElementContainer1(el, 0, 255, msdDrawingModeNormal)
        End If
    End If
    Err.Clear
End Sub

Private Sub TatDanhDau()
    Dim i As Long
    On Error Resume Next
    If mTec Is Nothing Then Set mTec = New Collection: Exit Sub
    For i = mTec.Count To 1 Step -1
        mTec.Item(i).Reset
        mTec.Remove i
    Next i
End Sub

Private Sub UserForm_Terminate()

End Sub

' SUA 13/10/2026: bam dong -> zoom view toi thua (thua o giua, rong gap ~3 lan)
Private Sub ZoomThua(ByVal v As clsVung)
    Dim x1 As Double, y1 As Double, x2 As Double, y2 As Double
    Dim rg() As Double, i As Long, d As Double
    On Error Resume Next
    x1 = v.cx: x2 = v.cx: y1 = v.cy: y2 = v.cy
    If v.HasRing Then
        rg = v.ring
        For i = LBound(rg, 1) To UBound(rg, 1)
            If rg(i, 0) < x1 Then x1 = rg(i, 0)
            If rg(i, 0) > x2 Then x2 = rg(i, 0)
            If rg(i, 1) < y1 Then y1 = rg(i, 1)
            If rg(i, 1) > y2 Then y2 = rg(i, 1)
        Next i
    End If
    d = x2 - x1: If y2 - y1 > d Then d = y2 - y1
    If d < 20 Then d = 20
    DC_ZoomVung (x1 + x2) / 2 - d * 1.5, (y1 + y2) / 2 - d * 1.5, _
                (x1 + x2) / 2 + d * 1.5, (y1 + y2) / 2 + d * 1.5
End Sub

Private Function CapNhat(ByVal v As clsVung, ByVal ws As Object, ByVal r As Long) As Boolean
    Dim a As String, b As String, c As String, d As String, e As String
    a = OChu(ws.Cells(r, 2).Value): b = OChu(ws.Cells(r, 3).Value): c = OChu(ws.Cells(r, 4).Value)
    d = OChu(ws.Cells(r, 6).Value): e = OChu(ws.Cells(r, 7).Value)
    If a <> v.SoTo Or b <> v.soThua Or c <> v.loaiDat Or d <> v.TenChu Or e <> v.DiaChi Then
        v.SoTo = a: v.soThua = b: v.loaiDat = c: v.TenChu = d: v.DiaChi = e
        CapNhat = True
    End If
End Function

Private Function OChu(ByVal x As Variant) As String
    If IsEmpty(x) Or IsNull(x) Then Exit Function
    OChu = Trim$(CStr(x))
End Function

' SUA 28/10: bam tieu de cot -> sap xep (bam lai = dao chieu). So truoc (1-2-3...), chu sau (A-Z).
Private Sub SapXep(ByVal cot As Long)
    Dim n As Long, i As Long, j As Long, gap As Long, tmp As Long, a() As Long, kh() As Variant
    Dim moi As Collection, giam As Boolean, lst As Object, r As Long, v As clsVung
    If mVungs Is Nothing Then Exit Sub
    n = mVungs.Count
    If n < 2 Then Exit Sub
    If mGoc Is Nothing Then Set mGoc = mVungs
    If cot = mCotSX Then giam = Not mGiamSX Else giam = False
    mCotSX = cot: mGiamSX = giam
    ReDim a(1 To n): ReDim kh(1 To n)
    Dim viTri As Object
    Set viTri = CreateObject("Scripting.Dictionary")
    For i = 1 To mGoc.Count
        viTri(ObjPtr(mGoc.Item(i))) = i
    Next i
    For i = 1 To n
        a(i) = i
        Set v = mVungs.Item(i)
        Select Case cot
            Case 0: kh(i) = viTri(ObjPtr(v))
            Case 1: kh(i) = v.SoTo
            Case 2: kh(i) = v.soThua
            Case 3: kh(i) = v.loaiDat
            Case 4: kh(i) = v.Dt
            Case 5: kh(i) = v.TenChu
            Case Else: kh(i) = v.DiaChi
        End Select
    Next i
    ' Shell sort tren chi so
    gap = n \ 2
    Do While gap > 0
        For i = gap + 1 To n
            tmp = a(i): j = i
            Do While j > gap
                If SoSanh(kh(a(j - gap)), kh(tmp), giam) <= 0 Then Exit Do
                a(j) = a(j - gap): j = j - gap
            Loop
            a(j) = tmp
        Next i
        gap = gap \ 2
    Loop
    Set moi = New Collection
    For i = 1 To n
        moi.Add mVungs.Item(a(i))
    Next i
    Set mVungs = moi
    ' hien lai bang
    Set lst = Me.Controls("lstData")
    lst.Clear
    For i = 1 To n
        Set v = mVungs.Item(i)
        lst.AddItem CStr(viTri(ObjPtr(v)))
        r = lst.ListCount - 1
        lst.List(r, 1) = v.SoTo
        lst.List(r, 2) = v.soThua
        lst.List(r, 3) = v.loaiDat
        lst.List(r, 4) = Format$(Round(v.Dt, 1), "0.0")
        lst.List(r, 5) = v.TenChu
        lst.List(r, 6) = v.DiaChi
    Next i
    DanhDauCot
End Sub

' >0 neu x dung SAU y (tang dan); giam = dao lai. So dung truoc chu.
Private Function SoSanh(ByVal x As Variant, ByVal y As Variant, ByVal giam As Boolean) As Long
    Dim sx As String, sy As String, nx As Boolean, ny As Boolean, k As Long
    sx = Trim$(CStr(x)): sy = Trim$(CStr(y))
    nx = LaSoSX(sx): ny = LaSoSX(sy)
    If Len(sx) = 0 And Len(sy) > 0 Then
        k = 1                                   ' o trong xuong cuoi
    ElseIf Len(sy) = 0 And Len(sx) > 0 Then
        k = -1
    ElseIf nx And ny Then
        If Val(Replace(sx, ",", ".")) > Val(Replace(sy, ",", ".")) Then
            k = 1
        ElseIf Val(Replace(sx, ",", ".")) < Val(Replace(sy, ",", ".")) Then
            k = -1
        End If
    ElseIf nx Then
        k = -1
    ElseIf ny Then
        k = 1
    Else
        k = StrComp(sx, sy, vbTextCompare)
    End If
    If giam Then k = -k
    SoSanh = k
End Function

Private Function LaSoSX(ByVal s As String) As Boolean
    If Len(s) = 0 Then Exit Function
    LaSoSX = IsNumeric(Replace(s, ",", "."))
End Function

        c.Caption = cap
    Next i
End Sub

    For Each c In Me.Controls
        c.Left = c.Left * k: c.Top = c.Top * k
        c.Width = c.Width * k: c.Height = c.Height * k
        c.Font.size = Round(c.Font.size * k)
    Next c
    cw = Split(Me.Controls("lstData").ColumnWidths, ";")
    For i = 0 To UBound(cw)
        If i > 0 Then s = s & ";"
        s = s & CStr(Round(Val(Replace(Replace(cw(i), "pt", ""), ",", ".")) * k, 1))
    Next i
    Me.Controls("lstData").ColumnWidths = Replace(s, ",", ".")
End Sub
