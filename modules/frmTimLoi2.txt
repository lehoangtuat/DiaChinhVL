Public Sub XuLyNut(ByVal ma As String)
    If ma = "DONG" Then
        Unload Me
        Exit Sub
    End If

    Dim lst As Object, csv As String, i As Long
    Set lst = Me.Controls("lstLevel")
    For i = 0 To lst.ListCount - 1
        If lst.Selected(i) Then
            If Len(csv) > 0 Then csv = csv & ","
            csv = csv & lst.List(i)
        End If
    Next i
    If Len(csv) = 0 Then
        MsgBox "Ch" & ChrW(432) & "a ch" & ChrW(7885) & "n level c" & ChrW(7841) & "nh n" & ChrW(224) & "o.", vbExclamation, "Tim loi"
        Exit Sub
    End If

    Dim canhNgan As Double, thuaNho As Double, snap As Double
    canhNgan = val(Me.Controls("txtCanhNgan").Text)
    thuaNho = val(Me.Controls("txtThuaNho").Text)
    snap = val(Me.Controls("txtSnap").Text)

    Me.Hide
    ChayTimLoi csv, _
               Me.Controls("chkD").Value, Me.Controls("chkS").Value, _
               Me.Controls("chkA").Value, Me.Controls("chkX").Value, _
               Me.Controls("chkO").Value, Me.Controls("chkE").Value, _
               canhNgan, thuaNho, snap, Me.Controls("txtLevelLoi").Text, _
               Me.Controls("txtMauRanh").Text
    Me.Show vbModeless
End Sub
