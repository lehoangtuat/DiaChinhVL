Public Sub KTT_ChayViec(ByVal ktt As Double, ByVal datum7 As Boolean, ByVal viec As String)
    DC_GhiCauHinh "KTT", Replace(CStr(ktt), ",", ".")
    DC_GhiCauHinh "DATUM7", IIf(datum7, "1", "0")
    If viec = "GEARTH" Then DC_MoGoogleEarth Else DC_XemVeTinh
End Sub
