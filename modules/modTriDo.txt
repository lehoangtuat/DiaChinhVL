' Phuong vi (tu Bac, theo chieu kim dong ho) dang do-phut-giay
Private Function PhuongVi(ByVal dx As Double, ByVal dy As Double) As String
    Dim a As Double, d As Long, m As Long, s As Long, t As Long
    a = Atn2(dx, dy) * 180# / Pi()          ' dx = Dong, dy = Bac
    If a < 0 Then a = a + 360#
    t = CLng(a * 3600#)
    If t >= 1296000 Then t = t - 1296000
    d = t \ 3600: m = (t Mod 3600) \ 60: s = t Mod 60
    PhuongVi = d & ChrW(176) & Format$(m, "00") & "'" & Format$(s, "00") & """"
End Function

Private Function TachDong(ByVal s As String, ByRef p() As String) As Long
    Dim t As String, a() As String, i As Long, n As Long
    t = Trim$(s)
    If Len(t) = 0 Then Exit Function
    If InStr(t, vbTab) > 0 Then
        t = Replace(t, vbTab, ";")
    ElseIf InStr(t, ";") = 0 And InStr(t, ",") > 0 And InStr(t, " ") = 0 Then
        t = Replace(t, ",", ";")
    ElseIf InStr(t, ";") = 0 Then
        Do While InStr(t, "  ") > 0: t = Replace(t, "  ", " "): Loop
        t = Replace(t, " ", ";")
    End If
    a = Split(t, ";")
    ReDim p(0 To UBound(a))
    For i = 0 To UBound(a)
        If Len(Trim$(a(i))) > 0 Then p(n) = Trim$(a(i)): n = n + 1
    Next i
    If n = 0 Then Exit Function
    ReDim Preserve p(0 To n - 1)
    TachDong = n
End Function

Private Function LaSo(ByVal s As String) As Boolean
    Dim t As String
    t = Replace(Trim$(s), ",", ".")
    If Len(t) = 0 Then Exit Function
    LaSo = (t Like "*#*") And Not (t Like "*[!0-9.+-]*")
End Function

Private Function SoVal(ByVal s As String) As Double
    SoVal = Val(Replace(Trim$(s), ",", "."))
End Function

Private Function DocChu(ByVal f As String) As String
    Dim st As Object
    On Error Resume Next
    Set st = CreateObject("ADODB.Stream")
    st.Type = 2: st.Charset = "utf-8": st.Open
    st.LoadFromFile f
    DocChu = st.ReadText(-1)
    st.Close
    If Len(DocChu) > 0 Then
        If AscW(Left$(DocChu, 1)) = &HFEFF Then DocChu = Mid$(DocChu, 2)
    End If
End Function

    Dim lvs() As Long, nLv As Long, gMin As Double, gTH As Double, lvLoi As Long, ok As Boolean
    nLv = DC_DsSoLevel(Hoi("Cac level ranh thua (vd 10,23):", "TD_LVRANH", LvHienTai(), ok), lvs): If Not ok Then Exit Sub

            For j = 0 To n - 1
                If kin Or (j > 0 And j < n - 1) Then
                    ip = j - 1: If ip < 0 Then ip = n - 1
                    inx = j + 1: If inx > n - 1 Then inx = 0

                            If Not daCo.Exists(k) Then
                                daCo.Add k, True
                                VeVong px, py, r, lv, 4
                                nTH = nTH + 1
                                If Not coF Then fx = px: fy = py: coF = True
                            End If
                        End If
                    End If
                End If
            Next j
        End If
    Next i
    RedrawAllViews
    If nNho + nTH = 0 Then
        DC_ThongBao "Kh" & ChrW(244) & "ng c" & ChrW(243) & " g" & ChrW(243) & "c nh" & ChrW(7887) & " h" & ChrW(417) & "n " & gMin & ChrW(176) & " hay " & ChrW(273) & "i" & ChrW(7875) & "m th" & ChrW(7859) & "ng h" & ChrW(224) & "ng.", TieuDe
    Else

' Goc tai B giua BA va BC (do, 0..180); -1 neu canh suy bien
Private Function GocTai(ByVal ax As Double, ByVal ay As Double, ByVal bx As Double, ByVal by As Double, _
                        ByVal cx As Double, ByVal cy As Double) As Double
    Dim ux As Double, uy As Double, vx As Double, vy As Double, lu As Double, lw As Double, cs As Double
    ux = ax - bx: uy = ay - by: vx = cx - bx: vy = cy - by
    lu = Sqr(ux * ux + uy * uy): lw = Sqr(vx * vx + vy * vy)
    GocTai = -1
    If lu < 0.0005 Or lw < 0.0005 Then Exit Function
    cs = (ux * vx + uy * vy) / (lu * lw)
    If cs > 1 Then cs = 1
    If cs < -1 Then cs = -1
    GocTai = Atn2(Sqr(1 - cs * cs), cs) * 180# / Pi()
End Function

    nLv = DC_DsSoLevel(Hoi("Cac level can lam tron (vd 10,23):", "TD_LVRANH", LvHienTai(), ok), lvs): If Not ok Then Exit Sub
    If nLv = 0 Then DC_ThongBao "Ch" & ChrW(432) & "a nh" & ChrW(7853) & "p level.", TieuDe, vbExclamation: Exit Sub
    le = CLng(Val(Hoi("So le giu lai (0-4):", "TD_LELT", "2", ok))): If Not ok Then Exit Sub
    If le < 0 Then le = 0
    If le > 4 Then le = 4
    If DC_ThongBao("L" & ChrW(224) & "m tr" & ChrW(242) & "n t" & ChrW(7885) & "a " & ChrW(273) & ChrW(7897) & " " & ChrW(273) & ChrW(7881) & "nh c" & ChrW(225) & "c " & ChrW(273) & ChrW(432) & ChrW(7901) & "ng / v" & ChrW(249) & "ng tr" & ChrW(234) & "n level " & DsLvChu(lvs, nLv) & " v" & ChrW(7873) & " " & le & " s" & ChrW(7889) & " l" & ChrW(7867) & "." & vbCrLf & _
                   "B" & ChrW(7843) & "n v" & ChrW(7869) & " s" & ChrW(7869) & " b" & ChrW(7883) & " s" & ChrW(7917) & "a v" & ChrW(224) & " KH" & ChrW(212) & "NG ho" & ChrW(224) & "n t" & ChrW(225) & "c " & ChrW(273) & ChrW(432) & ChrW(7907) & "c b" & ChrW(7857) & "ng undo mark - n" & ChrW(234) & "n l" & ChrW(432) & "u b" & ChrW(7843) & "n sao tr" & ChrW(432) & ChrW(7899) & "c." & vbCrLf & vbCrLf & "Ti" & ChrW(7871) & "p t" & ChrW(7909) & "c?", _
                   TieuDe, vbYesNo Or vbExclamation) <> vbYes Then Exit Sub

    Dim ee As ElementEnumerator, el As Element, ds As New Collection, i As Long
    Dim pts() As Point3d, j As Long, doi As Boolean, nSua As Long, nDinh As Long, nBo As Long
    Dim p As Point3d
    Set ee = QuetDuong()
    If ee Is Nothing Then Exit Sub
    Do While ee.MoveNext
        If CoLevel(ee.Current, lvs, nLv) Then ds.Add ee.Current
    Loop
    On Error Resume Next
    For i = 1 To ds.Count
        Set el = ds.Item(i)
        Select Case el.Type
        Case msdElementTypeLine, msdElementTypeLineString, msdElementTypeShape
            Err.Clear
            If el.Type = msdElementTypeShape Then pts = el.AsShapeElement.GetVertices Else pts = el.AsLineElement.GetVertices
            If Err.Number = 0 Then
                doi = False
                For j = LBound(pts) To UBound(pts)
                    p = pts(j)
                    p.X = Tron(p.X, le): p.Y = Tron(p.Y, le)
                    If p.X <> pts(j).X Or p.Y <> pts(j).Y Then
                        If el.Type = msdElementTypeShape Then
                            el.AsShapeElement.ModifyVertex j - LBound(pts), p
                        Else
                            el.AsLineElement.ModifyVertex j - LBound(pts), p
                        End If
                        doi = True: nDinh = nDinh + 1
                    End If
                Next j
                If doi Then
                    el.Rewrite
                    If Err.Number = 0 Then nSua = nSua + 1 Else nBo = nBo + 1
                End If
            Else
                nBo = nBo + 1
            End If
        Case Else
            nBo = nBo + 1

End Sub

Private Function Tron(ByVal v As Double, ByVal le As Long) As Double
    Dim f As Double
    f = 10# ^ le
    Tron = Int(v * f + 0.5) / f
End Function

Private Function DsLvChu(ByRef lvs() As Long, ByVal n As Long) As String
    Dim i As Long

        DsLvChu = DsLvChu & IIf(i > 0, ",", "") & lvs(i)

Private Function CoLevel(ByVal el As Element, ByRef lvs() As Long, ByVal n As Long) As Boolean
    Dim i As Long, s As Long
    s = DC_LevelSo(el)
    For i = 0 To n - 1
        If lvs(i) = s Then CoLevel = True: Exit Function
    Next i
End Function

Private Function QuetDuong() As ElementEnumerator
    Dim esc As ElementScanCriteria
    On Error Resume Next
    Set esc = New ElementScanCriteria
    esc.ExcludeAllTypes
    esc.IncludeType msdElementTypeLine
    esc.IncludeType msdElementTypeLineString
    esc.IncludeType msdElementTypeShape
    esc.IncludeType msdElementTypeComplexString
    esc.IncludeType msdElementTypeComplexShape
    Set QuetDuong = ActiveModelReference.Scan(esc)
End Function

' Moi phan tu tren cac level -> 1 chuoi dinh (Double phang x0,y0,x1,y1...) trong Collection
Private Function ChuoiTheoLevel(ByRef lvs() As Long, ByVal nLv As Long) As Collection
    Dim kq As New Collection, ee As ElementEnumerator, pts() As Point3d, n As Long
    Set ChuoiTheoLevel = kq
    Set ee = QuetDuong()
    If ee Is Nothing Then Exit Function
    Do While ee.MoveNext
        If CoLevel(ee.Current, lvs, nLv) Then
            n = DinhPhanTu(ee.Current, pts)
            If n >= 2 Then kq.Add Phang(pts, n)
        End If
    Loop
End Function

' Cac phan tu dang chon (chi loai duong / vung)
Private Function DsChon() As Collection
    Dim kq As New Collection, ee As ElementEnumerator, t As Long
    Set DsChon = kq
    On Error Resume Next
    Set ee = ActiveModelReference.GetSelectedElements
    If ee Is Nothing Then Exit Function
    Do While ee.MoveNext
        t = ee.Current.Type
        If t = msdElementTypeLine Or t = msdElementTypeLineString Or t = msdElementTypeShape Or _
           t = msdElementTypeComplexString Or t = msdElementTypeComplexShape Then kq.Add ee.Current
    Loop
End Function

' Dinh cua 1 phan tu (noi cac phan con cua duong phuc, bo dinh trung lien ke). Tra ve so dinh.
Private Function DinhPhanTu(ByVal el As Element, ByRef pts() As Point3d) As Long
    Dim kq() As Point3d, n As Long, subs As ElementEnumerator, p() As Point3d, i As Long
    ReDim kq(0 To 0)
    On Error Resume Next
    Select Case el.Type
    Case msdElementTypeLine, msdElementTypeLineString
        p = el.AsLineElement.GetVertices
        If Err.Number = 0 Then ThemDinh kq, n, p
    Case msdElementTypeShape
        p = el.AsShapeElement.GetVertices
        If Err.Number = 0 Then ThemDinh kq, n, p
    Case msdElementTypeComplexString, msdElementTypeComplexShape
        Set subs = el.AsComplexElement.GetSubElements
        Do While subs.MoveNext
            Err.Clear
            Select Case subs.Current.Type
            Case msdElementTypeLine, msdElementTypeLineString
                p = subs.Current.AsLineElement.GetVertices
                If Err.Number = 0 Then ThemDinh kq, n, p
            Case msdElementTypeShape
                p = subs.Current.AsShapeElement.GetVertices
                If Err.Number = 0 Then ThemDinh kq, n, p
            End Select
        Loop
    End Select
    If n = 0 Then DinhPhanTu = 0: Exit Function
    ReDim Preserve kq(0 To n - 1)
    pts = kq
    DinhPhanTu = n
End Function

Private Function Phang(ByRef pts() As Point3d, ByVal n As Long) As Double()
    Dim d() As Double, i As Long
    ReDim d(0 To 2 * n - 1)
    For i = 0 To n - 1
        d(2 * i) = pts(i).X: d(2 * i + 1) = pts(i).Y
    Next i
    Phang = d
End Function

Private Sub ThemDinh(ByRef kq() As Point3d, ByRef n As Long, ByRef p() As Point3d)
    Dim i As Long
    For i = LBound(p) To UBound(p)
        If n > 0 Then
            If Abs(kq(n - 1).X - p(i).X) < 0.0005 And Abs(kq(n - 1).Y - p(i).Y) < 0.0005 Then GoTo tiep
        End If
        If n > UBound(kq) Then ReDim Preserve kq(0 To n * 2 + 8)
        kq(n) = p(i): n = n + 1
tiep:
    Next i
End Sub

Private Function KhoaCanh(ByVal ax As Double, ByVal ay As Double, ByVal bx As Double, ByVal by As Double) As String
    Dim a As String, b As String
    a = Format$(ax, "0.000") & "|" & Format$(ay, "0.000")
    b = Format$(bx, "0.000") & "|" & Format$(by, "0.000")
    If a < b Then KhoaCanh = a & "#" & b Else KhoaCanh = b & "#" & a
End Function

Private Function SoLe(ByVal v As Double, ByVal le As Long) As String
    If le <= 0 Then SoLe = Format$(v, "0") Else SoLe = Format$(v, "0." & String$(le, "0"))
End Function

Private Function Pi() As Double
    Pi = 4# * Atn(1#)
End Function

' atan2(y, x)
Private Function Atn2(ByVal y As Double, ByVal x As Double) As Double
    If x > 0 Then
        Atn2 = Atn(y / x)
    ElseIf x < 0 Then
        If y >= 0 Then Atn2 = Atn(y / x) + Pi() Else Atn2 = Atn(y / x) - Pi()
    Else
        If y > 0 Then Atn2 = Pi() / 2 Else If y < 0 Then Atn2 = -Pi() / 2 Else Atn2 = 0
    End If
End Function

Private Function FontVN() As Object
    Dim f As Object
    If mDaTimFont Then Set FontVN = mFont: Exit Function
    mDaTimFont = True
    On Error Resume Next
    For Each f In ActiveDesignFile.Fonts
        If StrComp(f.Name, DC_FONT_VIET, vbTextCompare) = 0 Then Set mFont = f: Exit For
    Next f
    Set FontVN = mFont
End Function

' Ve chu (goc rad), level lv, mau (-1 = mac dinh), Times New Roman, net 0
Private Sub VeChu(ByVal x As Double, ByVal y As Double, ByVal s As String, ByVal cao As Double, _
                  ByVal goc As Double, ByVal lv As Level, ByVal canh As Long, Optional ByVal mau As Long = -1)
    Dim t As TextElement, ts As TextStyle, f As Object
    On Error Resume Next
    Set t = CreateTextElement1(Nothing, s, Point3dFromXYZ(x, y, 0#), Matrix3dFromAxisAndRotationAngle(2, goc))
    If t Is Nothing Then Exit Sub
    Set ts = t.TextStyle
    ts.Height = cao: ts.Width = cao
    ts.Justification = canh
    Set f = FontVN()
    If Not f Is Nothing Then Set ts.Font = f
    Set t.TextStyle = ts
    If Err.Number <> 0 Then Err.Clear: t.TextStyle = ts
    If Not lv Is Nothing Then Set t.Level = lv
    If mau >= 0 Then t.Color = mau
    t.LineWeight = 0
    ActiveModelReference.AddElement t
End Sub

Private Sub VeVong(ByVal x As Double, ByVal y As Double, ByVal r As Double, ByVal lv As Level, ByVal mau As Long)
    Dim e As EllipseElement
    On Error Resume Next
    Set e = CreateEllipseElement2(Nothing, Point3dFromXYZ(x, y, 0#), r, r, Matrix3dIdentity)
    If e Is Nothing Then Exit Sub
    If Not lv Is Nothing Then Set e.Level = lv
    If mau >= 0 Then e.Color = mau
    e.LineWeight = 0
    ActiveModelReference.AddElement e
End Sub
