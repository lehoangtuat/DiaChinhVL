' Trich luc 1 thua da co san (vung topology). Thua chua gan thong tin
' van xuat duoc: so thua = 0, loai dat = 0, dien tich tu tinh.
'============================================================
Public Sub ChayTrichLucThua(ByVal found As clsVung, _
                            ByVal mauPath As String, ByVal kieu As String, _
                            ByVal toBanDo As String, ByVal tenXa As String, _
                            ByVal tenHuyen As String, ByVal tenTinh As String, _
                            ByVal lvXuat As String, ByVal cao As Double, _
                            ByVal cheDoMau As Boolean, ByVal tyLe As Double)

    ' File mau .dgn (kieu gCadas) -> bat mau ra .txt (co cache)
    mauPath = DC_ChuanBiMau(mauPath)

    ' Bat buoc font TrueType co chu Viet (may moi chua dat font -> mat chu)
    If Not DC_DatFontViet() Then
        DC_ThongBao "Kh" & ChrW(244) & "ng " & ChrW(273) & ChrW(7863) & "t " & ChrW(273) & ChrW(432) & ChrW(7907) & "c font " & DC_FONT_VIET & " cho ch" & ChrW(7919) & "." & vbCrLf & _

    End If

    ' Thua chua gan thong tin -> gia tri mac dinh
    If Len(Trim$(found.soThua)) = 0 Then found.soThua = "0"
    If Len(Trim$(found.loaiDat)) = 0 Then found.loaiDat = "0"
    If found.Dt <= 0 Then found.Dt = DienTichVung(found)

    ' Dat level xuat (go so "50" hoac ten "Level 50")
    If Len(Trim$(lvXuat)) > 0 Then
        On Error Resume Next
        Set ActiveSettings.Level = DC_TimLevel(lvXuat)
        On Error GoTo 0
    End If

    If Len(Trim$(toBanDo)) = 0 Then toBanDo = found.SoTo
    If Len(Trim$(toBanDo)) = 0 Then toBanDo = "0"
    If cao <= 0 Then cao = 1

    ' Ty le tu dong (tyLe = 0): ty le chuan nho nhat de thua vua o so do
    Dim tuDong As Boolean
    If tyLe <= 0 And cheDoMau Then
        tyLe = TyLeTuDong(found, mauPath)
        tuDong = True
    End If

    Dim zx1 As Double, zy1 As Double, zx2 As Double, zy2 As Double
    If cheDoMau Then
        ChenKhungTuThua found, mauPath, kieu, toBanDo, tenXa, tenHuyen, tenTinh, tyLe
        ' pham vi khung vua ve (khung dat tam o so do tai tam thua)
        If DocKhungMau(mauPath) Then
            zx1 = gTLCx + (mHx1 - mBoxCX) * tyLe / 20000#
            zx2 = gTLCx + (mHx2 - mBoxCX) * tyLe / 20000#

            zy2 = gTLCy + (mHy2 - mBoxCY) * tyLe / 20000#
        End If
    Else

        VeSoDoThua found, cao, 0
        VeNhanThuaDayDu found, cao, "mau"
        VeBangToaDo found, kx2 + cao * 3#, ky2, cao, 0
    End If
    If zx2 <= zx1 Then
        ' so do tu ve / khong doc duoc mau: zoom quanh thua
        BaoThua found, zx1, zy1, zx2, zy2
        Dim r As Double
        r = (zx2 - zx1) + (zy2 - zy1)
        zx1 = zx1 - r: zx2 = zx2 + r * 2#: zy1 = zy1 - r: zy2 = zy2 + r
    End If
    gTLx1 = zx1: gTLy1 = zy1: gTLx2 = zx2: gTLy2 = zy2
    gTLTen = "TL_To" & toBanDo & "_Thua" & found.soThua
    DC_ZoomVung zx1, zy1, zx2, zy2

End Sub

'--- dien tich tu ranh (cong thuc Gauss), tru dien tich dao
Private Function DienTichVung(ByVal v As clsVung) As Double
    Dim r() As Double, s As Double, k As Long
    On Error Resume Next
    If Not v.HasRing Then Exit Function
    r = v.ring
    s = Abs(DtMang(r))
    For k = 1 To v.Dao.Count
        s = s - Abs(DtMang(v.Dao.Item(k)))
    Next k
    DienTichVung = s
End Function

'============================================================
' TY LE TU DONG: ty le chuan nho nhat de thua nam vua o so do (90%)
' o ty le that (kp = 1). O so do = level 24 cua mau, quy doi
' kich thuoc that = kich thuoc mau * tyLe / 20000 (xem modKhungMau).
'============================================================
Private Function TyLeTuDong(ByVal v As clsVung, ByVal mauPath As String) As Double
    Dim ds As Variant, i As Long, pw As Double, ph As Double
    Dim x1 As Double, y1 As Double, x2 As Double, y2 As Double
    ds = Array(200#, 500#, 1000#, 2000#, 5000#, 10000#)
    If Not DocKhungMau(mauPath) Then mBoxW = 989.8: mBoxH = 1048#
    BaoThua v, x1, y1, x2, y2
    pw = x2 - x1: ph = y2 - y1
    For i = LBound(ds) To UBound(ds)
        If pw <= mBoxW * ds(i) / 20000# * 0.9 And ph <= mBoxH * ds(i) / 20000# * 0.9 Then
            TyLeTuDong = ds(i): Exit Function
        End If
    Next i
    TyLeTuDong = 10000#      ' qua lon: giu 1:10000, so do tu thu nho
End Function

' Doc o so do (L 24) va bao ngoai (H) tu file mau
Private Function DocKhungMau(ByVal path As String) As Boolean
    Dim st As Object, txt As String, dong() As String, p() As String
    Dim i As Long, j As Long, co As Boolean
    Dim bx1 As Double, by1 As Double, bx2 As Double, by2 As Double, x As Double, y As Double
    On Error Resume Next
    Set st = CreateObject("ADODB.Stream")
    st.Type = 2: st.Charset = "utf-8": st.Open
    st.LoadFromFile path
    If Err.Number <> 0 Then Exit Function
    txt = st.ReadText(-1): st.Close
    On Error GoTo 0
    dong = Split(txt, vbLf)
    For i = LBound(dong) To UBound(dong)
        p = Split(Replace(dong(i), vbCr, ""), vbTab)
        If UBound(p) >= 5 Then
            If p(0) = "H" Then
                mHx1 = SoD(p(2)): mHy1 = SoD(p(3)): mHx2 = SoD(p(4)): mHy2 = SoD(p(5))
            ElseIf p(0) = "L" And p(1) = "24" Then
                For j = 2 To UBound(p) - 1 Step 2
                    x = SoD(p(j)): y = SoD(p(j + 1))
                    If Not co Then
                        bx1 = x: bx2 = x: by1 = y: by2 = y: co = True
                    Else
                        If x < bx1 Then bx1 = x
                        If x > bx2 Then bx2 = x
                        If y < by1 Then by1 = y
                        If y > by2 Then by2 = y
                    End If
                Next j
            End If
        End If
    Next i
    If Not co Then Exit Function
    mBoxW = bx2 - bx1: mBoxH = by2 - by1
    mBoxCX = (bx1 + bx2) / 2#: mBoxCY = (by1 + by2) / 2#
    DocKhungMau = (mHx2 > mHx1)
End Function

Private Function SoD(ByVal s As String) As Double
    SoD = Val(Replace(Trim$(s), ",", "."))
End Function

' Hinh bao ngoai cua thua
Private Sub BaoThua(ByVal v As clsVung, ByRef x1 As Double, ByRef y1 As Double, _
                    ByRef x2 As Double, ByRef y2 As Double)
    Dim r() As Double, i As Long
    r = v.ring
    x1 = r(0, 0): x2 = r(0, 0): y1 = r(0, 1): y2 = r(0, 1)
    For i = LBound(r, 1) To UBound(r, 1)
        If r(i, 0) < x1 Then x1 = r(i, 0)
        If r(i, 0) > x2 Then x2 = r(i, 0)
        If r(i, 1) < y1 Then y1 = r(i, 1)
        If r(i, 1) > y2 Then y2 = r(i, 1)
    Next i
End Sub

'============================================================
' Zoom view vao vung (x1,y1)-(x2,y2), chua le 8%, giu ti le khung view
'============================================================
Public Sub DC_ZoomVung(ByVal x1 As Double, ByVal y1 As Double, _
                       ByVal x2 As Double, ByVal y2 As Double)
    Dim vw As View, ex As Point3d, og As Point3d, w As Double, h As Double, asp As Double
    On Error Resume Next
    If x2 <= x1 Or y2 <= y1 Then Exit Sub
    Set vw = gViewTL
    If vw Is Nothing Then Set vw = CommandState.LastView
    If vw Is Nothing Then Set vw = ActiveDesignFile.Views(1)
    If vw Is Nothing Then Exit Sub
    ex = vw.Extents
    asp = 1#
    If ex.X > 0 Then asp = ex.Y / ex.X
    w = (x2 - x1) * 1.08: h = (y2 - y1) * 1.08
    If h > w * asp Then w = h / asp Else h = w * asp
    og = vw.Origin
    og.X = (x1 + x2) / 2# - w / 2#: og.Y = (y1 + y2) / 2# - h / 2#
    vw.Origin = og
    ex.X = w: ex.Y = h
    vw.Extents = ex
    vw.Redraw
    Set gViewTL = Nothing
End Sub

Private Function DtMang(ByVal r As Variant) As Double
    Dim i As Long, j As Long, n0 As Long, n1 As Long, s As Double
    n0 = LBound(r, 1): n1 = UBound(r, 1)
    For i = n0 To n1
        j = i + 1: If j > n1 Then j = n0
        s = s + r(i, 0) * r(j, 1) - r(j, 0) * r(i, 1)
    Next i
    DtMang = s / 2#
End Function

'--- Bo loc theo danh sach so level (chi level da co trong ban ve)
Private Function TaoLocLevel(ByVal dsLv As String, ByRef co As Boolean) As ElementScanCriteria
    Dim esc As New ElementScanCriteria, ds() As Long, c As Long, i As Long, lv As Level
    esc.ExcludeNonGraphical
    esc.ExcludeAllLevels
    c = DC_DsSoLevel(dsLv, ds)
    For i = 0 To c - 1
        Set lv = Nothing
        Set lv = DC_TimLevel(CStr(ds(i)))
        If Not lv Is Nothing Then esc.IncludeLevel lv: co = True
    Next i
    Set TaoLocLevel = esc
End Function

'--- Dua ket qua quet vao mang (tranh sua ban ve trong luc dang quet)
Private Function DemMang(ByVal ee As ElementEnumerator, ByRef arr() As Element) As Long
    Dim n As Long
    ReDim arr(0 To 255)
    Do While ee.MoveNext
        If n > UBound(arr) Then ReDim Preserve arr(0 To n * 2)
        Set arr(n) = ee.Current
        n = n + 1
    Loop
    DemMang = n
End Function

    End If

    ' bam Tat nhieu lan lien tiep: giu danh sach cu de Bat tra lai du
    If mLvTat Is Nothing Then Set mLvTat = New Collection
    If mAttTat Is Nothing Then Set mAttTat = New Collection
    Dim vw As View, vi As Long, lv As Level
    On Error Resume Next
    For vi = 1 To 8
        Set vw = ActiveDesignFile.Views(vi)
        If Not vw Is Nothing Then
            If vw.IsOpen Then
                For Each lv In ActiveDesignFile.Levels
                    If Not giu.Exists(CStr(lv.Number)) Then
                        If lv.IsDisplayedInView(vw) Then
                            lv.IsDisplayedInView(vw) = False
                            mLvTat.Add vi & "|" & lv.Name
                        End If
                    End If
                Next lv
            End If
        End If
        Set vw = Nothing
    Next vi
    Dim att As Attachment
    For Each att In ActiveModelReference.Attachments
        If att.DisplayFlag Then
            att.DisplayFlag = False
            att.Rewrite
            mAttTat.Add att.AttachName
        End If
    Next att
    On Error GoTo 0
    RedrawAllViews
End Sub

Public Sub DC_BatLopBanDo()
    Dim vw As View, vi As Long, lv As Level, i As Long, p As Long, it As String
    On Error Resume Next
    If mLvTat Is Nothing Then
        ' chua tat lan nao: bat het
        For vi = 1 To 8
            Set vw = ActiveDesignFile.Views(vi)
            If Not vw Is Nothing Then
                If vw.IsOpen Then
                    For Each lv In ActiveDesignFile.Levels
                        lv.IsDisplayedInView(vw) = True
                    Next lv
                End If
            End If
            Set vw = Nothing
        Next vi
    Else
        For i = 1 To mLvTat.Count
            it = mLvTat(i): p = InStr(it, "|")
            Set vw = ActiveDesignFile.Views(CLng(Left$(it, p - 1)))
            Set lv = ActiveDesignFile.Levels(Mid$(it, p + 1))
            If Not vw Is Nothing And Not lv Is Nothing Then lv.IsDisplayedInView(vw) = True
            Set vw = Nothing: Set lv = Nothing
        Next i
    End If
    Dim att As Attachment
    For Each att In ActiveModelReference.Attachments
        If Not att.DisplayFlag Then
            If mAttTat Is Nothing Then
                att.DisplayFlag = True: att.Rewrite
            ElseIf CoTrong(mAttTat, att.AttachName) Then
                att.DisplayFlag = True: att.Rewrite
            End If
        End If
    Next att
    On Error GoTo 0
    Set mLvTat = Nothing: Set mAttTat = Nothing
    RedrawAllViews
End Sub

Private Function CoTrong(ByVal col As Collection, ByVal s As String) As Boolean
    Dim i As Long
    For i = 1 To col.Count
        If col(i) = s Then CoTrong = True: Exit Function
    Next i
End Function
