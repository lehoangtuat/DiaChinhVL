'============================================================
' modTachLop - Tach lop nhan thua: chu chung 1 lop -> 3 lop
'   Co chu cai (LUK, ONT...)        -> lop loai dat
'   Chi chu so (2248)               -> lop so thua
'   So co dau phay/cham (4846,2)    -> lop dien tich
'   Con lai (12+13, -, ...)         -> giu nguyen
'   Form: frmTachLop (tu tao). Menu Tien ich -> "TACHLOP".
'============================================================
Option Explicit

Private mDs As Collection       ' phan tu se chuyen
Private mLoai As Collection     ' loai tuong ung (1/2/3)

Public Sub DC_MoTachLop()
    DC_MoFormTuTao "frmTachLop"
End Sub

' 0 = bo qua ; 1 = loai dat ; 2 = so thua ; 3 = dien tich
Public Function TL_PhanLoai(ByVal s As String) As Long
    Dim i As Long, c As Long, coChu As Boolean, nSo As Long, coPhay As Boolean, khac As Boolean
    s = Trim$(s)
    If Len(s) = 0 Then Exit Function
    For i = 1 To Len(s)
        c = AscW(Mid$(s, i, 1)) And &HFFFF&
        If (c >= 65 And c <= 90) Or (c >= 97 And c <= 122) Or c >= 128 Then
            coChu = True
        ElseIf c >= 48 And c <= 57 Then
            nSo = nSo + 1
        ElseIf c = 44 Or c = 46 Then
            coPhay = True
        ElseIf c <> 32 Then
            khac = True
        End If
    Next i
    If coChu Then TL_PhanLoai = 1: Exit Function
    If nSo = 0 Or khac Then Exit Function
    If coPhay Then TL_PhanLoai = 3 Else TL_PhanLoai = 2
End Function

Private Function NoiDung(ByVal el As Element) As String
    Dim subs As ElementEnumerator, s As String
    On Error Resume Next
    If el.Type = msdElementTypeText Then
        NoiDung = el.AsTextElement.Text
    ElseIf el.Type = msdElementTypeTextNode Then
        Set subs = el.AsComplexElement.GetSubElements
        Do While subs.MoveNext
            If subs.Current.Type = msdElementTypeText Then s = s & " " & subs.Current.AsTextElement.Text
        Loop
        NoiDung = Trim$(s)
    End If
End Function

' Buoc 1: quet + phan loai (file hien tai). Tra ve so phan tu tung loai (n(1..3)), nBo = so chu giu nguyen.
Public Function TL_Quet(ByVal lvNguon As Long, ByVal chiChon As Boolean, ByRef n() As Long, ByRef nBo As Long) As Long
    Dim ee As ElementEnumerator
    Set mDs = New Collection
    Set mLoai = New Collection
    ReDim n(0 To 3)
    nBo = 0
    On Error Resume Next
    If chiChon Then
        Set ee = ActiveModelReference.GetSelectedElements
    Else
        Set ee = QuetChu(ActiveModelReference)
    End If
    If ee Is Nothing Then Exit Function
    GomPhanLoai ee, lvNguon, mDs, mLoai, n, nBo
    TL_Quet = mDs.Count
End Function

Private Function QuetChu(ByVal m As ModelReference) As ElementEnumerator
    Dim esc As ElementScanCriteria
    On Error Resume Next
    Set esc = New ElementScanCriteria
    esc.ExcludeAllTypes
    esc.IncludeType msdElementTypeText
    esc.IncludeType msdElementTypeTextNode
    Set QuetChu = m.Scan(esc)
End Function

Private Sub GomPhanLoai(ByVal ee As ElementEnumerator, ByVal lvNguon As Long, ByVal ds As Collection, _
                        ByVal loai As Collection, ByRef n() As Long, ByRef nBo As Long)
    Dim el As Element, k As Long
    If ee Is Nothing Then Exit Sub
    On Error Resume Next
    Do While ee.MoveNext
        Set el = ee.Current
        If el.Type = msdElementTypeText Or el.Type = msdElementTypeTextNode Then
            If DC_LevelSo(el) = lvNguon Then
                k = TL_PhanLoai(NoiDung(el))
                If k > 0 Then
                    ds.Add el
                    loai.Add k
                    n(k) = n(k) + 1
                Else
                    nBo = nBo + 1
                End If
            End If
        End If
    Loop
End Sub

' Buoc 2: chuyen lop (file hien tai, sau TL_Quet). lv(1..3) = so level dich (<= 0 : khong chuyen loai do)
Public Function TL_Chuyen(ByRef lv() As Long, ByRef nDa() As Long) As Long
    ReDim nDa(0 To 3)
    If mDs Is Nothing Then Exit Function
    TL_Chuyen = ChuyenDs(ActiveDesignFile, mDs, mLoai, lv, nDa)
    Set mDs = Nothing
    Set mLoai = Nothing
    RedrawAllViews
End Function

Private Function ChuyenDs(ByVal df As DesignFile, ByVal ds As Collection, ByVal loai As Collection, _
                          ByRef lv() As Long, ByRef nDa() As Long) As Long
    Dim i As Long, k As Long, el As Element, lvObj(1 To 3) As Level
    For k = 1 To 3
        If lv(k) > 0 Then Set lvObj(k) = LvTheoSo(df, lv(k))
    Next k
    On Error Resume Next
    For i = 1 To ds.Count
        k = loai.Item(i)
        If Not lvObj(k) Is Nothing Then
            Set el = ds.Item(i)
            Err.Clear
            Set el.Level = lvObj(k)
            If Err.Number <> 0 Then Err.Clear: el.Level = lvObj(k)
            el.Rewrite
            If Err.Number = 0 Then nDa(k) = nDa(k) + 1: ChuyenDs = ChuyenDs + 1
        End If
    Next i
End Function

' Level theo SO trong 1 design file; chua co thi tao "Level n"
Private Function LvTheoSo(ByVal df As DesignFile, ByVal n As Long) As Level
    Dim lv As Level
    On Error Resume Next
    For Each lv In df.Levels
        If lv.Number = n Then Set LvTheoSo = lv: Exit Function
    Next lv
    Set lv = df.AddNewLevel("Level " & n)
    If Not lv Is Nothing Then
        lv.Number = n
        df.Levels.Rewrite
        Set LvTheoSo = lv
    End If
End Function

'============================================================
' HANG LOAT: tach lop 1 file (mo ngam -> sua -> Save -> Close)
'   Tra ve "" neu OK; nDa(1..3) = so chu da chuyen tung loai.
'============================================================
Public Function TL_XuLyFile(ByVal path As String, ByVal lvNguon As Long, ByRef lv() As Long, _
                            ByVal saoLuu As Boolean, ByRef nDa() As Long) As String
    Dim df As DesignFile, laHienTai As Boolean, m As ModelReference
    Dim ds As Collection, loai As Collection, n() As Long, nBo As Long, tong As Long, k As Long
    ReDim nDa(0 To 3)
    On Error Resume Next
    laHienTai = (StrComp(ActiveDesignFile.FullName, path, vbTextCompare) = 0)
    Err.Clear
    If saoLuu Then
        Dim fso As Object
        Set fso = CreateObject("Scripting.FileSystemObject")
        fso.CopyFile path, path & ".bak", True
        If Err.Number <> 0 Then
            TL_XuLyFile = "Kh" & ChrW(244) & "ng sao l" & ChrW(432) & "u " & ChrW(273) & ChrW(432) & ChrW(7907) & "c (.bak): " & Err.Description
            Exit Function
        End If
    End If

    On Error GoTo Loi
    If laHienTai Then
        Set df = ActiveDesignFile
    Else
        Set df = Application.OpenDesignFileForProgram(path, False)
    End If
    If df Is Nothing Then TL_XuLyFile = "Kh" & ChrW(244) & "ng m" & ChrW(7903) & " " & ChrW(273) & ChrW(432) & ChrW(7907) & "c file": Exit Function

    Dim d() As Long
    For Each m In df.Models
        Set ds = New Collection: Set loai = New Collection
        ReDim n(0 To 3): nBo = 0
        GomPhanLoai QuetChu(m), lvNguon, ds, loai, n, nBo
        If ds.Count > 0 Then
            ReDim d(0 To 3)
            tong = tong + ChuyenDs(df, ds, loai, lv, d)
            For k = 1 To 3
                nDa(k) = nDa(k) + d(k)
            Next k
        End If
    Next m
    If tong > 0 Then df.Save
    If Not laHienTai Then df.Close
    Exit Function
Loi:
    TL_XuLyFile = Err.Description
    On Error Resume Next
    If Not laHienTai Then
        If Not df Is Nothing Then df.Close
    End If
End Function

'============================================================
' PHA KHOI (drop cell) co chua chu tren lop can tach
'   Cell thuong: chep cac phan tu con (Clone) ra ngoai roi xoa cell.
'   Shared cell: chua pha duoc bang VBA -> chi dem de bao.
'   lvNguon <= 0 : khoi co chu o bat ky lop nao.
'============================================================
Public Function TL_PhaKhoiFile(ByVal path As String, ByVal lvNguon As Long, ByVal saoLuu As Boolean, _
                               ByVal chiChon As Boolean, ByRef nPha As Long, ByRef nShared As Long) As String
    Dim df As DesignFile, laHienTai As Boolean, m As ModelReference
    nPha = 0: nShared = 0
    On Error Resume Next
    laHienTai = (StrComp(ActiveDesignFile.FullName, path, vbTextCompare) = 0)
    Err.Clear
    If saoLuu And Not laHienTai Then
        Dim fso As Object
        Set fso = CreateObject("Scripting.FileSystemObject")
        fso.CopyFile path, path & ".bak", True
        If Err.Number <> 0 Then
            TL_PhaKhoiFile = "Kh" & ChrW(244) & "ng sao l" & ChrW(432) & "u " & ChrW(273) & ChrW(432) & ChrW(7907) & "c (.bak): " & Err.Description
            Exit Function
        End If
    End If

    On Error GoTo Loi
    If laHienTai Then
        If chiChon Then
            PhaKhoiDs ActiveModelReference, ActiveModelReference.GetSelectedElements, lvNguon, nPha, nShared
            RedrawAllViews
            Exit Function
        End If
        Set df = ActiveDesignFile
    Else
        Set df = Application.OpenDesignFileForProgram(path, False)
    End If
    If df Is Nothing Then TL_PhaKhoiFile = "Kh" & ChrW(244) & "ng m" & ChrW(7903) & " " & ChrW(273) & ChrW(432) & ChrW(7907) & "c file": Exit Function
    For Each m In df.Models
        PhaKhoiDs m, QuetKhoi(m), lvNguon, nPha, nShared
    Next m
    If nPha > 0 Then df.Save
    If Not laHienTai Then df.Close Else RedrawAllViews
    Exit Function
Loi:
    TL_PhaKhoiFile = Err.Description
    On Error Resume Next
    If Not laHienTai Then
        If Not df Is Nothing Then df.Close
    End If
End Function

Private Function QuetKhoi(ByVal m As ModelReference) As ElementEnumerator
    Dim esc As ElementScanCriteria
    On Error Resume Next
    Set esc = New ElementScanCriteria
    esc.ExcludeAllTypes
    esc.IncludeType msdElementTypeCellHeader
    esc.IncludeType msdElementTypeSharedCell
    Set QuetKhoi = m.Scan(esc)
End Function

Private Sub PhaKhoiDs(ByVal m As ModelReference, ByVal ee As ElementEnumerator, ByVal lvNguon As Long, _
                      ByRef nPha As Long, ByRef nShared As Long)
    Dim ds As New Collection, el As Element, i As Long, loi As Boolean
    If ee Is Nothing Then Exit Sub
    On Error Resume Next
    Do While ee.MoveNext
        Set el = ee.Current
        If el.Type = msdElementTypeSharedCell Then
            nShared = nShared + 1
        ElseIf el.Type = msdElementTypeCellHeader Then
            If KhoiCoChu(el, lvNguon) Then ds.Add el
        End If
    Loop
    For i = 1 To ds.Count
        Set el = ds.Item(i)
        loi = False
        ChepConKhoi m, el, loi
        If Not loi Then
            Err.Clear
            m.RemoveElement el
            If Err.Number = 0 Then nPha = nPha + 1
        End If
    Next i
End Sub

' Khoi co chu (text / text node) tren lop lvNguon (<= 0: lop bat ky), tim ca khoi long nhau
Private Function KhoiCoChu(ByVal el As Element, ByVal lvNguon As Long) As Boolean
    Dim subs As ElementEnumerator, c As Element
    On Error Resume Next
    Set subs = el.AsComplexElement.GetSubElements
    If subs Is Nothing Then Exit Function
    Do While subs.MoveNext
        Set c = subs.Current
        If c.Type = msdElementTypeText Or c.Type = msdElementTypeTextNode Then
            If lvNguon <= 0 Then KhoiCoChu = True: Exit Function
            If DC_LevelSo(c) = lvNguon Then KhoiCoChu = True: Exit Function
        ElseIf c.Type = msdElementTypeCellHeader Then
            If KhoiCoChu(c, lvNguon) Then KhoiCoChu = True: Exit Function
        End If
    Loop
End Function

' Chep cac phan tu con cua khoi ra model (khoi long nhau -> pha tiep)
Private Sub ChepConKhoi(ByVal m As ModelReference, ByVal el As Element, ByRef loi As Boolean)
    Dim subs As ElementEnumerator, c As Element, cp As Element, o As Object
    On Error Resume Next
    Set subs = el.AsComplexElement.GetSubElements
    If subs Is Nothing Then loi = True: Exit Sub
    Do While subs.MoveNext
        Set c = subs.Current
        If c.Type = msdElementTypeCellHeader Then
            ChepConKhoi m, c, loi
        Else
            Err.Clear
            Set cp = Nothing
            Set o = c
            Set cp = o.Clone              ' late binding: thieu Clone -> loi chay (bat duoc), khong loi bien dich
            If Err.Number = 0 And Not cp Is Nothing Then
                m.AddElement cp
                If Err.Number <> 0 Then loi = True
            Else
                loi = True
            End If
        End If
    Loop
End Sub

' Tach chuoi -> phan (noi dung, vi tri ky tu bat dau). Tra ve so phan (0 = khong phai chu gop)
Public Function TL_TachChuoi(ByVal s As String, ByRef ten() As String, ByRef viTri() As Long) As Long
    Dim i As Long, c As String, tok As String, bd As Long, n As Long
    Dim loai As String, pLoai As Long, so As String, pSo As Long, dt As String, pDt As Long
    Dim toks() As String, pos() As Long, nt As Long
    ReDim toks(1 To 20): ReDim pos(1 To 20)
    s = s & " "
    For i = 1 To Len(s)
        c = Mid$(s, i, 1)
        If c = " " Or c = "/" Or c = vbTab Then
            If Len(tok) > 0 And nt < 20 Then nt = nt + 1: toks(nt) = tok: pos(nt) = bd
            tok = ""
        Else
            If Len(tok) = 0 Then bd = i
            tok = tok & c
        End If
    Next i
    If nt < 2 Then Exit Function
    For i = 1 To nt
        tok = toks(i)
        If LaMaLoai(tok) Then
            If Len(loai) > 0 Or Len(so) > 0 Then Exit Function
            loai = tok: pLoai = pos(i)
        ElseIf LaSo(tok, False) And Len(so) = 0 Then
            so = tok: pSo = pos(i)
        ElseIf LaSo(tok, True) And Len(so) > 0 And Len(dt) = 0 Then
            dt = tok: pDt = pos(i)
        Else
            Exit Function                ' co phan la (vd ten duong) -> khong tach
        End If
    Next i
    If Len(so) = 0 Then Exit Function
    ReDim ten(1 To 3): ReDim viTri(1 To 3)
    If Len(loai) > 0 Then n = n + 1: ten(n) = loai: viTri(n) = pLoai
    n = n + 1: ten(n) = so: viTri(n) = pSo
    If Len(dt) > 0 Then n = n + 1: ten(n) = dt: viTri(n) = pDt
    If n < 2 Then n = 0
    TL_TachChuoi = n
End Function

' Ma loai dat: chu IN HOA A-Z (co the co "+"), dai 2..12
Private Function LaMaLoai(ByVal t As String) As Boolean
    Dim i As Long, c As Long, coChu As Boolean
    If Len(t) < 2 Or Len(t) > 12 Then Exit Function
    For i = 1 To Len(t)
        c = AscW(Mid$(t, i, 1))
        If c >= 65 And c <= 90 Then
            coChu = True
        ElseIf c <> 43 Then
            Exit Function
        End If
    Next i
    LaMaLoai = coChu
End Function

' So: chi chu so ; choPhay = cho phep dau , hoac . (dien tich)
Private Function LaSo(ByVal t As String, ByVal choPhay As Boolean) As Boolean
    Dim i As Long, c As Long, nSo As Long
    For i = 1 To Len(t)
        c = AscW(Mid$(t, i, 1))
        If c >= 48 And c <= 57 Then
            nSo = nSo + 1
        ElseIf choPhay And (c = 44 Or c = 46) Then
        Else
            Exit Function
        End If
    Next i
    LaSo = (nSo > 0)
End Function

Private Sub TachChuDs(ByVal m As ModelReference, ByVal ee As ElementEnumerator, ByVal lvNguon As Long, ByRef nTach As Long)
    Dim ds As New Collection, el As Element, c As Element, i As Long
    If ee Is Nothing Then Exit Sub
    On Error Resume Next
    Do While ee.MoveNext
        Set el = ee.Current
        If el.Type = msdElementTypeText Or el.Type = msdElementTypeTextNode Then
            If lvNguon <= 0 Or DC_LevelSo(el) = lvNguon Then ds.Add el
        End If
    Loop
    For i = 1 To ds.Count
        Set el = ds.Item(i)
        Set c = ChuDon(el)
        If Not c Is Nothing Then
            If TachMotChu(m, el, c) Then nTach = nTach + 1
        End If
    Next i
End Sub

' Text -> chinh no ; text node 1 dong -> chu con ; con lai Nothing
Private Function ChuDon(ByVal el As Element) As Element
    Dim sb As ElementEnumerator, dem As Long, c As Element
    On Error Resume Next
    If el.Type = msdElementTypeText Then Set ChuDon = el: Exit Function
    Set sb = el.AsComplexElement.GetSubElements
    Do While sb.MoveNext
        If sb.Current.Type = msdElementTypeText Then dem = dem + 1: Set c = sb.Current
    Loop
    If dem = 1 Then Set ChuDon = c
End Function

Private Function TachMotChu(ByVal m As ModelReference, ByVal el As Element, ByVal c As Element) As Boolean
    Dim s As String, ten() As String, vt() As Long, n As Long, i As Long
    Dim te As TextElement, rot As Matrix3d, d As Point3d, h As Double
    Dim rw As Double, rh As Double, L As Double, ca As Double, sa As Double
    Dim cx As Double, cy As Double, f As Double, nt As TextElement, ts As Object
    On Error Resume Next
    Set te = c
    s = te.Text
    n = TL_TachChuoi(s, ten, vt)
    If n < 2 Then Exit Function
    rot = te.Rotation
    d.X = rot.RowX.X: d.Y = rot.RowY.X: d.Z = 0#      ' truc X cua chu sau khi xoay
    If Err.Number <> 0 Then Err.Clear: d = Point3dFromXYZ(1#, 0#, 0#)
    L = Sqr(d.X * d.X + d.Y * d.Y): If L <= 0 Then d = Point3dFromXYZ(1#, 0#, 0#): L = 1#
    d.X = d.X / L: d.Y = d.Y / L
    h = te.TextStyle.Height
    rw = el.Range.High.X - el.Range.Low.X: rh = el.Range.High.Y - el.Range.Low.Y
    cx = (el.Range.High.X + el.Range.Low.X) / 2#: cy = (el.Range.High.Y + el.Range.Low.Y) / 2#
    ca = Abs(d.X): sa = Abs(d.Y)
    If ca >= sa Then L = (rw - h * sa) / ca Else L = (rh - h * ca) / sa
    If L <= 0 Then L = Len(s) * h * 0.6
    Dim moi As New Collection
    For i = 1 To n
        f = (vt(i) - 1 + Len(ten(i)) / 2#) / Len(s)
        Set nt = Nothing
        Set nt = CreateTextElement1(c, ten(i), Point3dFromXYZ(cx + d.X * (f - 0.5) * L, cy + d.Y * (f - 0.5) * L, 0#), rot)
        If nt Is Nothing Then Exit Function
        Set ts = nt.TextStyle
        ts.Justification = msdTextJustificationCenterCenter
        Set nt.TextStyle = ts
        Err.Clear
        nt.Color = c.Color
        moi.Add nt
    Next i
    Err.Clear
    For i = 1 To moi.Count
        m.AddElement moi.Item(i)
    Next i
    If Err.Number <> 0 Then Exit Function
    m.RemoveElement el
    TachMotChu = (Err.Number = 0)
    Err.Clear
End Function
