'------------------------------------------------------------ xu ly nut
Public Sub XuLyNut(ByVal ma As String)
    Select Case ma
    Case "CHONTEP": ChonTep
    Case "BOTEP"
        If mLstF.ListIndex >= 0 Then mLstF.RemoveItem mLstF.ListIndex
    Case "XEMDL": XemDuLieu
    Case "SANGVE": mp.Value = 2
    Case "VE": VeDiem
    End Select
End Sub

Private Function DauPhanCach() As String
    Dim s As String
    If Me.Controls("chkCach").Value Then s = s & " "
    If Me.Controls("chkPhay").Value Then s = s & ","
    If Me.Controls("chkTab").Value Then s = s & vbTab
    If Me.Controls("chkKhac").Value Then s = s & Trim$(Me.Controls("txtKhac").Text)
    If Len(s) = 0 Then s = " "
    DauPhanCach = s
End Function

Private Sub ChonTep()
    Dim f As String, d As String, a() As String, i As Long
    f = CF_ChonMotFile("T" & ChrW(7885) & "a " & ChrW(273) & ChrW(7897) & " (*.txt;*.csv;*.dat)|*.txt;*.csv;*.dat|T" & ChrW(7845) & "t c" & ChrW(7843) & " (*.*)|*.*", "Ch" & ChrW(7885) & "n t" & ChrW(7879) & "p t" & ChrW(7885) & "a " & ChrW(273) & ChrW(7897) & " " & ChrW(273) & "i" & ChrW(7875) & "m mia")
    If Len(f) = 0 Then Exit Sub
    For i = 0 To mLstF.ListCount - 1
        If StrComp(mLstF.List(i), f, vbTextCompare) = 0 Then Exit Sub
    Next i
    mLstF.AddItem f
    ' tep dau tien: tu doan thu tu cot
    If mLstF.ListCount = 1 Then
        d = DM_DoanCot(f, DauPhanCach())
        If Len(d) > 0 Then
            a = Split(d, ",")
            For i = 0 To 4
                mCot(i).ListIndex = CLng(a(i))
            Next i
        End If
    End If
End Sub

Private Sub XemDuLieu()
    Dim cot(0 To 4) As Long, i As Long, nBo As Long, n As Long, sep As String, r As Long, coX As Boolean, coY As Boolean
    If mLstF.ListCount = 0 Then
        DC_ThongBao "Ch" & ChrW(432) & "a ch" & ChrW(7885) & "n t" & ChrW(7879) & "p t" & ChrW(7885) & "a " & ChrW(273) & ChrW(7897) & ".", ChrW(272) & "i" & ChrW(7875) & "m mia", vbExclamation
        Exit Sub
    End If
    For i = 0 To 4
        cot(i) = mCot(i).ListIndex
        If cot(i) = 1 Then coX = True
        If cot(i) = 2 Then coY = True
    Next i
    If Not (coX And coY) Then
        DC_ThongBao "Ph" & ChrW(7843) & "i c" & ChrW(243) & " m" & ChrW(7897) & "t c" & ChrW(7897) & "t X v" & ChrW(224) & " m" & ChrW(7897) & "t c" & ChrW(7897) & "t Y.", ChrW(272) & "i" & ChrW(7875) & "m mia", vbExclamation
        Exit Sub
    End If
    sep = DauPhanCach()
    DC_GhiCauHinh "DM_COT", cot(0) & "," & cot(1) & "," & cot(2) & "," & cot(3) & "," & cot(4)
    DC_GhiCauHinh "DM_SEP", IIf(Me.Controls("chkCach").Value, "1", "0") & "," & IIf(Me.Controls("chkPhay").Value, "1", "0") & "," & _
                            IIf(Me.Controls("chkTab").Value, "1", "0") & "," & IIf(Me.Controls("chkKhac").Value, "1", "0")
    DC_GhiCauHinh "DM_SEPKHAC", Me.Controls("txtKhac").Text

    DM_XoaDuLieu
    For i = 0 To mLstF.ListCount - 1
        n = n + DM_DocFile(CStr(mLstF.List(i)), cot, sep, nBo)
    Next i
    mLstD.Clear
    For i = 1 To DM_N
        mLstD.AddItem DM_Ten(i)
        r = mLstD.ListCount - 1
        mLstD.List(r, 1) = Format$(DM_X(i), "0.000")
        mLstD.List(r, 2) = Format$(DM_Y(i), "0.000")
        mLstD.List(r, 3) = DM_H(i)
        mLstD.List(r, 4) = DM_Ma(i)
    Next i
    Me.Controls("lblDem").Caption = DM_N & " " & ChrW(273) & "i" & ChrW(7875) & "m" & IIf(nBo > 0, " (b" & ChrW(7887) & " qua " & nBo & " d" & ChrW(242) & "ng kh" & ChrW(244) & "ng ph" & ChrW(7843) & "i t" & ChrW(7885) & "a " & ChrW(273) & ChrW(7897) & ")", "")
    mp.Value = 1
    If DM_N = 0 Then DC_ThongBao "Kh" & ChrW(244) & "ng " & ChrW(273) & ChrW(7885) & "c " & ChrW(273) & ChrW(432) & ChrW(7907) & "c " & ChrW(273) & "i" & ChrW(7875) & "m n" & ChrW(224) & "o. Ki" & ChrW(7875) & "m tra l" & ChrW(7841) & "i " & ChrW(273) & ChrW(7883) & "nh d" & ChrW(7841) & "ng c" & ChrW(7897) & "t v" & ChrW(224) & " d" & ChrW(7845) & "u ph" & ChrW(226) & "n c" & ChrW(225) & "ch.", ChrW(272) & "i" & ChrW(7875) & "m mia", vbExclamation
End Sub

Private Sub VeDiem()
    Dim lv As Long, cao As Double, le As Long, n As Long, kh As Long
    If DM_N = 0 Then
        DC_ThongBao "Ch" & ChrW(432) & "a c" & ChrW(243) & " d" & ChrW(7919) & " li" & ChrW(7879) & "u. Ch" & ChrW(7885) & "n t" & ChrW(7879) & "p r" & ChrW(7891) & "i b" & ChrW(7845) & "m 'Xem d" & ChrW(7919) & " li" & ChrW(7879) & "u' tr" & ChrW(432) & ChrW(7899) & "c.", ChrW(272) & "i" & ChrW(7875) & "m mia", vbExclamation
        mp.Value = 0
        Exit Sub
    End If
    lv = CLng(Val(Me.Controls("txtLv").Text)): If lv <= 0 Then lv = 20
    cao = Val(Replace(Me.Controls("txtCao").Text, ",", ".")): If cao <= 0 Then cao = 1
    le = CLng(Val(Me.Controls("txtLe").Text)): If le < 0 Then le = 0
    kh = Me.Controls("cboKyHieu").ListIndex: If kh < 0 Then kh = 0
    DC_GhiCauHinh "DM_LV", CStr(lv)
    DC_GhiCauHinh "DM_CAO", Me.Controls("txtCao").Text
    DC_GhiCauHinh "DM_LE", CStr(le)
    DC_GhiCauHinh "DM_TEN", IIf(Me.Controls("chkTen").Value, "1", "0")
    DC_GhiCauHinh "DM_HH", IIf(Me.Controls("chkH").Value, "1", "0")
    DC_GhiCauHinh "DM_MA", IIf(Me.Controls("chkMa").Value, "1", "0")

    n = DM_Ve(lv, cao, CBool(Me.Controls("chkTen").Value), CBool(Me.Controls("chkH").Value), _
