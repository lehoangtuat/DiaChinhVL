Public Sub XuLyNut(ByVal ma As String)
    If ma = "DONG" Then Unload Me: Exit Sub
    Dim n As Long, nMoc As Long, loi As String
    Luu
    n = CDC_ThucHien(Me.Controls("cboFont").Text, SoCD("txtCao"), CLng(SoCD("txtMau")), SoCD("txtKc"), _
                     CLng(SoCD("txtLe")), Me.Controls("cboLvVao").Text, Me.Controls("cboLvRa").Text, _
                     SoCD("txtMin"), Me.Controls("chkFence").Value, Me.Controls("chkPhay").Value, _
                     Me.Controls("chkMoc").Value, nMoc, loi)
    If Len(loi) > 0 Then
        DC_ThongBao loi, "Chi" & ChrW(7873) & "u d" & ChrW(224) & "i c" & ChrW(7841) & "nh", vbExclamation
    Else
        DC_ThongBao ChrW(272) & ChrW(227) & " ghi " & n & " chi" & ChrW(7873) & "u d" & ChrW(224) & "i c" & ChrW(7841) & "nh" & IIf(nMoc > 0, ", " & nMoc & " m" & ChrW(7889) & "c " & ChrW(273) & ChrW(7881) & "nh", "") & "." & vbCrLf & "(Sai: key-in  undo mark)", "Chi" & ChrW(7873) & "u d" & ChrW(224) & "i c" & ChrW(7841) & "nh"
    End If
End Sub

Private Function SoCD(ByVal nm As String) As Double
    SoCD = Val(Replace(Trim$(Me.Controls(nm).Text), ",", "."))
End Function

Private Sub Luu()
    DC_GhiCauHinh "CDC_FONT", Me.Controls("cboFont").Text
    DC_GhiCauHinh "CDC_CAO", Me.Controls("txtCao").Text
    DC_GhiCauHinh "CDC_KC", Me.Controls("txtKc").Text
    DC_GhiCauHinh "CDC_LE", Me.Controls("txtLe").Text
    DC_GhiCauHinh "CDC_MIN", Me.Controls("txtMin").Text
    DC_GhiCauHinh "CDC_MAU", Me.Controls("txtMau").Text
    DC_GhiCauHinh "CDC_LVVAO", Me.Controls("cboLvVao").Text
    DC_GhiCauHinh "CDC_LVRA", Me.Controls("cboLvRa").Text
    DC_GhiCauHinh "CDC_FENCE", IIf(Me.Controls("chkFence").Value, "1", "0")
    DC_GhiCauHinh "CDC_PHAY", IIf(Me.Controls("chkPhay").Value, "1", "0")
    DC_GhiCauHinh "CDC_MOC", IIf(Me.Controls("chkMoc").Value, "1", "0")
End Sub
