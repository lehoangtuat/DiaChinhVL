'============================================================
' Ham xu ly goi tu form: gan thong tin tu nhan vao CSDL + ve nhan
'============================================================
Public Sub ChayGanTT(ByVal path As String, ByVal lvSo As String, ByVal lvTo As String, _
                     ByVal lvLoai As String, ByVal lvChu As String, ByVal lvDC As String, _
                     ByVal lvNhan As String, ByVal cao As Double, ByVal veNhan As Boolean, _
                     Optional ByVal xoaCu As Boolean = False)
    Dim vungs As Collection
    Set vungs = MoCSDL(path)
    If vungs.Count = 0 Then MsgBox "CSDL rong / khong doc duoc: " & path, vbExclamation: Exit Sub

    ' SUA 13/10/2026: khong bat buoc level so thua; bao level nhap ma khong co
    Dim thieu As String, lvX As Variant
    For Each lvX In Array(lvSo, lvTo, lvLoai, lvChu, lvDC)
        If Len(Trim$(lvX)) > 0 Then
            If DC_TimLevel(CStr(lvX)) Is Nothing Then thieu = thieu & " " & Trim$(lvX)
        End If
    Next lvX
    If Len(thieu) > 0 Then
        DC_ThongBao "Kh" & ChrW(244) & "ng t" & ChrW(236) & "m th" & ChrW(7845) & "y level:" & thieu & vbCrLf & "(G" & ChrW(245) & " s" & ChrW(7889) & " level, v" & ChrW(237) & " d" & ChrW(7909) & " 13)", "G" & ChrW(225) & "n th" & ChrW(244) & "ng tin", vbExclamation
        Exit Sub
    End If
    If xoaCu Then
        Dim k As Long, vx As clsVung
        For k = 1 To vungs.Count
            Set vx = vungs.Item(k)
            If Len(Trim$(lvSo)) > 0 Then vx.soThua = ""
            If Len(Trim$(lvTo)) > 0 Then vx.SoTo = ""
            If Len(Trim$(lvLoai)) > 0 Then vx.loaiDat = ""
            If Len(Trim$(lvChu)) > 0 Then vx.TenChu = ""
            If Len(Trim$(lvDC)) > 0 Then vx.DiaChi = ""
        Next k
    End If

    Dim nSo As Long, nTo As Long, nLoai As Long, nChu As Long, nDC As Long
    nSo = GanTruongCSDL2(vungs, lvSo, "so_thua")
    nTo = GanTruongCSDL2(vungs, lvTo, "so_to")
    nLoai = GanTruongCSDL2(vungs, lvLoai, "loai_dat")
    nChu = GanTruongCSDL2(vungs, lvChu, "ten_chu")
    nDC = GanTruongCSDL2(vungs, lvDC, "dia_chi")

    Dim ok As Boolean
    ok = LuuCSDL(vungs, path)

    If veNhan Then
        If cao <= 0 Then cao = 1
        If Len(Trim$(lvNhan)) > 0 Then
            On Error Resume Next
            Set ActiveSettings.Level = DC_TimLevel(lvNhan)
            On Error GoTo 0
        End If
        Dim i As Long, v As clsVung
        For i = 1 To vungs.Count
            Set v = vungs.Item(i)
            VeNhanTT26 v, cao        ' SUA 25/10: nhan TT26 thanh 1 khoi (cell)
        Next i
    End If

    DC_ThongBao "S" & ChrW(7889) & " th" & ChrW(7917) & "a: " & vungs.Count & vbCrLf & _
           "G" & ChrW(225) & "n s" & ChrW(7889) & " th" & ChrW(7917) & "a: " & nSo & " ; t" & ChrW(7901) & ": " & nTo & " ; lo" & ChrW(7841) & "i " & ChrW(273) & ChrW(7845) & "t: " & nLoai & vbCrLf & _
           "T" & ChrW(234) & "n ch" & ChrW(7911) & ": " & nChu & " ; " & ChrW(273) & ChrW(7883) & "a ch" & ChrW(7881) & ": " & nDC & vbCrLf & _
           IIf(ok, ChrW(272) & ChrW(227) & " c" & ChrW(7853) & "p nh" & ChrW(7853) & "t CSDL.", "KH" & ChrW(212) & "NG l" & ChrW(432) & "u " & ChrW(273) & ChrW(432) & ChrW(7907) & "c CSDL!"), "G" & ChrW(225) & "n th" & ChrW(244) & "ng tin", IIf(ok, vbInformation, vbExclamation)
End Sub

'--- gan 1 truong (dung rieng cho module nay) ---
Private Function GanTruongCSDL2(ByVal vungs As Collection, ByVal levelName As String, _
                                ByVal truong As String) As Long
    If Len(Trim$(levelName)) = 0 Then Exit Function
    Dim esc As ElementScanCriteria
    Set esc = New ElementScanCriteria
    esc.ExcludeAllTypes
    esc.IncludeType msdElementTypeText
    esc.IncludeType msdElementTypeTextNode
    Dim lv As Level, lvSo As Long
    Set lv = DC_TimLevel(levelName)          ' go "13" = level so 13
    If lv Is Nothing Then Exit Function
    lvSo = lv.Number

    Dim ee As ElementEnumerator
    Set ee = ActiveModelReference.Scan(esc)
    Dim x As Double, y As Double, s As String, v As clsVung, dem As Long
    Do While ee.MoveNext
        If DC_LevelSo(ee.Current) = lvSo Then
        If DocText2(ee.Current, x, y, s) Then
            If Len(s) > 0 Then
                Set v = TimThuaChua(vungs, x, y)
                If Not v Is Nothing Then
                    ' SUA 13/10/2026: nhan tren ban do la dung -> ghi de (truoc chi gan khi trong,
                    ' nen loai dat mac dinh LUC khong bao gio duoc thay)
                    If LayTruong2(v, truong) <> s Then DatTruong2 v, truong, s
                    dem = dem + 1
                End If
            End If
        End If
        End If
    Loop
    GanTruongCSDL2 = dem
End Function

Private Function DocText2(ByVal el As Element, ByRef x As Double, _
                          ByRef y As Double, ByRef s As String) As Boolean
    DocText2 = False: s = ""
    Dim o As Point3d
    If el.Type = msdElementTypeText Then
        Dim oT As TextElement
        Set oT = el: o = oT.Origin: x = o.x: y = o.y: s = Trim$(oT.Text): DocText2 = True
    ElseIf el.Type = msdElementTypeTextNode Then
        Dim oNode As TextNodeElement
        Set oNode = el: o = oNode.Origin: x = o.x: y = o.y
        Dim subs As ElementEnumerator
        Set subs = oNode.GetSubElements
        Do While subs.MoveNext
            If subs.Current.Type = msdElementTypeText Then
                s = Trim$(subs.Current.AsTextElement.Text)
                If Len(s) > 0 Then Exit Do
            End If
        Loop
        DocText2 = True
    End If
End Function

Private Function LayTruong2(ByVal v As clsVung, ByVal t As String) As String
    Select Case t
        Case "so_thua": LayTruong2 = v.soThua
        Case "so_to": LayTruong2 = v.SoTo
        Case "loai_dat": LayTruong2 = v.loaiDat
        Case "ten_chu": LayTruong2 = v.TenChu
        Case "dia_chi": LayTruong2 = v.DiaChi
    End Select
End Function

Private Sub DatTruong2(ByVal v As clsVung, ByVal t As String, ByVal g As String)
    Select Case t
        Case "so_thua": v.soThua = g
        Case "so_to": v.SoTo = g
        Case "loai_dat": v.loaiDat = g
        Case "ten_chu": v.TenChu = g
        Case "dia_chi": v.DiaChi = g
    End Select
End Sub

'============================================================
' SUA 25/10: NHAN THUA THEO TT26 - 1 KHOI (cell "NHANTHUA"):
'   loai dat ben trai, so thua (tren) / dien tich (duoi) co gach ngang.
'   Ve bang mau / level / font dang dung (level lay tu o "Lop nhan").
'============================================================
Public Sub VeNhanTT26(ByVal v As clsVung, ByVal cao As Double)
    Dim tu As String, duoiS As String, loai As String
    Dim wTu As Double, wDt As Double, rong As Double, nua As Double, wl As Double
    Dim els() As Element, n As Long, cel As CellElement
    If v Is Nothing Then Exit Sub
    If cao <= 0 Then cao = 1
    tu = Trim$(v.soThua)
    duoiS = Replace(Format$(Round(v.Dt, 1), "0.0"), ".", ",")
    loai = Trim$(v.loaiDat)
    If Len(tu) = 0 And v.Dt <= 0 Then Exit Sub
    wTu = Len(tu) * cao * 0.6
    wDt = Len(duoiS) * cao * 0.6
    rong = wTu: If wDt > rong Then rong = wDt
    nua = rong / 2# + cao * 0.2

    On Error Resume Next
    ReDim els(0 To 3)
    Set els(n) = CreateLineElement2(Nothing, Point3dFromXYZ(v.cx - nua, v.cy, 0#), Point3dFromXYZ(v.cx + nua, v.cy, 0#))
    If Not els(n) Is Nothing Then n = n + 1
    If Len(tu) > 0 Then Set els(n) = ChuNhan(v.cx, v.cy + cao * 0.9, tu, cao): If Not els(n) Is Nothing Then n = n + 1
    If Len(duoiS) > 0 Then Set els(n) = ChuNhan(v.cx, v.cy - cao * 0.9, duoiS, cao): If Not els(n) Is Nothing Then n = n + 1
    If Len(loai) > 0 Then
        wl = Len(loai) * cao * 0.6
        Set els(n) = ChuNhan(v.cx - nua - wl / 2# - cao * 0.3, v.cy, loai, cao)
        If Not els(n) Is Nothing Then n = n + 1
    End If
    If n = 0 Then Exit Sub
    ReDim Preserve els(0 To n - 1)
    Err.Clear
    Set cel = CreateCellElement1("NHANTHUA", els, Point3dFromXYZ(v.cx, v.cy, 0#), False)
    If Not cel Is Nothing And Err.Number = 0 Then
        ActiveModelReference.AddElement cel
    Else
        ' khong tao duoc cell -> ve rieng tung phan tu
        Err.Clear
        Dim i As Long
        For i = 0 To n - 1
            ActiveModelReference.AddElement els(i)
        Next i
    End If
    Err.Clear
End Sub

Private Function ChuNhan(ByVal x As Double, ByVal y As Double, ByVal s As String, ByVal cao As Double) As Element
    Dim t As TextElement
    On Error Resume Next
    ActiveSettings.TextStyle.Height = cao
    ActiveSettings.TextStyle.Width = cao
    ActiveSettings.TextStyle.Justification = msdTextJustificationCenterCenter
    Set t = CreateTextElement1(Nothing, s, Point3dFromXYZ(x, y, 0#), Matrix3dIdentity)
    If Not t Is Nothing Then t.LineWeight = 0
    Set ChuNhan = t
End Function
