Private Sub AddLabel(ByVal nm As String, ByVal cap As String, _
                     ByVal x As Single, ByVal y As Single, ByVal w As Single)

    Set c = Me.Controls.Add("Forms.Label.1", nm, True)
    c.Caption = cap: c.Left = x: c.Top = y: c.Width = w: c.Height = 12
    c.Font.Name = "Arial": c.Font.size = 8
End Sub

Private Sub AddPair(ByVal cap As String, ByVal nm As String, ByVal txt As String, _
                    ByVal x As Single, ByVal y As Single)
    AddLabel "lbl_" & nm, cap, x, y + 2, 70
    Dim c As Object
    Set c = Me.Controls.Add("Forms.TextBox.1", nm, True)
    c.Text = txt: c.Left = x + 72: c.Top = y: c.Width = 96: c.Height = 15
    c.Font.Name = "Arial": c.Font.size = 8
End Sub

Public Sub XuLyNut(ByVal ma As String)
    If ma = "DONG" Then Unload Me: Exit Sub
    If ma = "XOANHAN" Then
        Me.Hide
        DC_XoaNhanTheoLevel
        Me.Show vbModeless
        Exit Sub
    End If

    ' ma = "OK"
    Dim lst As Object, csv As String, i As Long
    Set lst = Me.Controls("lstLevel")
    For i = 0 To lst.ListCount - 1
        If lst.Selected(i) Then
            If Len(csv) > 0 Then csv = csv & ","
            csv = csv & lst.List(i)
        End If
    Next i
    If Len(csv) = 0 Then MsgBox "Ch" & ChrW(432) & "a ch" & ChrW(7885) & "n level ranh.", vbExclamation: Exit Sub

    ' Che do TAO THEO DIEM BAM -> giao cho cong cu bat diem, an form
    If Me.Controls("chkTheoDiem").Value Then
        Me.Hide
        DC_TaoVungTheoDiem csv, Me.Controls("txtLoaiDat").Text, Me.Controls("txtToBanDo").Text, _
                val(Me.Controls("txtSnap").Text), Me.Controls("txtLevelTam").Text, _
                CLng(val(Me.Controls("txtMau").Text)), val(Me.Controls("txtKT").Text), _
                Me.Controls("txtCSDL").Text, Me.Controls("txtMauRanh").Text, _
                val(Me.Controls("txtBatHo").Text)
        Exit Sub
    End If

    Me.Hide
    ChayTaoThua csv, Me.Controls("txtLoaiDat").Text, Me.Controls("txtToBanDo").Text, _
                CLng(val(Me.Controls("txtSoLe").Text)), val(Me.Controls("txtBoNho").Text), _
                val(Me.Controls("txtSnap").Text), Me.Controls("txtLevelTam").Text, _
                CLng(val(Me.Controls("txtMau").Text)), val(Me.Controls("txtKT").Text), _
                Me.Controls("txtCSDL").Text, Me.Controls("txtMauRanh").Text, _
                val(Me.Controls("txtBatHo").Text)
    Me.Show vbModeless
End Sub
