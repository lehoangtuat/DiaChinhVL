'============================================================
Public Sub DC_KiemTraCapNhat()

    Dim i As Long, ds As String, ten() As String, thu As String, s As String
    Dim nTai As Long, kq As String

    v = TaiChuoi(DC_URL_CAPNHAT & "version.txt")
    If Len(Trim$(v)) = 0 Then
        DC_ThongBao "Kh" & ChrW(244) & "ng k" & ChrW(7871) & "t n" & ChrW(7889) & "i " & ChrW(273) & ChrW(432) & ChrW(7907) & "c m" & ChrW(225) & "y ch" & ChrW(7911) & " c" & ChrW(7853) & "p nh" & ChrW(7853) & "t (GitHub)." & vbCrLf & _
                    "Ki" & ChrW(7875) & "m tra m" & ChrW(7841) & "ng Internet r" & ChrW(7891) & "i th" & ChrW(7917) & " l" & ChrW(7841) & "i.", "C" & ChrW(7853) & "p nh" & ChrW(7853) & "t", vbExclamation
        Exit Sub
    End If
    dong = Split(Replace(v, vbCr, ""), vbLf)
    moi = Trim$(dong(0))
    For i = 1 To UBound(dong)
        If Len(Trim$(dong(i))) > 0 Then ghiChu = ghiChu & dong(i) & vbCrLf

        Exit Sub
    End If

                   ghiChu & vbCrLf & "C" & ChrW(7853) & "p nh" & ChrW(7853) & "t ngay?", "C" & ChrW(7853) & "p nh" & ChrW(7853) & "t", vbYesNo Or vbQuestion) <> vbYes Then Exit Sub

    ds = TaiChuoi(DC_URL_CAPNHAT & "files.txt")
    If Len(Trim$(ds)) = 0 Then DC_ThongBao "Kh" & ChrW(244) & "ng t" & ChrW(7843) & "i " & ChrW(273) & ChrW(432) & ChrW(7907) & "c files.txt.", "C" & ChrW(7853) & "p nh" & ChrW(7853) & "t", vbExclamation: Exit Sub
    thu = DC_THUMUC & "\capnhat_web"
    On Error Resume Next
    If Len(Dir$(DC_THUMUC, vbDirectory)) = 0 Then MkDir DC_THUMUC
    If Len(Dir$(thu, vbDirectory)) = 0 Then MkDir thu
    On Error GoTo 0

    ' SUA 09/10/2026: chi tai module DA THAY DOI (so ma SHA cua GitHub voi lan
    ' cap nhat truoc, luu o capnhat_web\_sha.txt). Khong lay duoc SHA -> tai het.
    Dim shaMoi As Object, shaCu As Object, can() As String, nCan As Long
    Set shaMoi = DocShaGitHub()
    Set shaCu = DocShaCu(thu & "\_sha.txt")
    ten = Split(Replace(ds, vbCr, ""), vbLf)
    ReDim can(0 To UBound(ten))
    For i = LBound(ten) To UBound(ten)
        ten(i) = Trim$(ten(i))

            If shaMoi.Count = 0 Then
                can(nCan) = ten(i): nCan = nCan + 1

            ShowStatus ""

                        "Ki" & ChrW(7875) & "m tra m" & ChrW(7841) & "ng r" & ChrW(7891) & "i b" & ChrW(7845) & "m Ki" & ChrW(7875) & "m tra c" & ChrW(7853) & "p nh" & ChrW(7853) & "t l" & ChrW(7841) & "i.", "C" & ChrW(7853) & "p nh" & ChrW(7853) & "t", vbExclamation
            Exit Sub
        End If
        If Not GhiAnsi(thu & "\" & can(i) & ".txt", s) Then ShowStatus "": Exit Sub

    Next i
    ShowStatus "Cap nhat DiaChinhVL: dang nap code ..."
    DoEvents

    On Error GoTo 0
    MsgBox "Da cap nhat len ban " & moi & " (" & nTai & " module)." & vbCrLf & vbCrLf & kq & vbCrLf & _
           "Mo lai menu Dia Chinh VL de dung ban moi." & vbCrLf & _
           "(Khi dong MicroStation neu hoi luu project -> chon Yes)", vbInformation, "Cap nhat"
End Sub

'------------------------------------------------------------ nap code
Private Function ApDung(ByVal thu As String, ten() As String) As String
    Dim oVBE As Object, prj As Object, p As Object, c As Object
    Dim i As Long, kq As String, code As String, kieu As Long, loi As Long

    On Error Resume Next
    Set oVBE = Application.VBE
    On Error GoTo 0
    If oVBE Is Nothing Then ApDung = "LOI: khong truy cap duoc VBE.": Exit Function
    For Each p In oVBE.VBProjects
        If StrComp(p.Name, "DiaChinhVL", vbTextCompare) = 0 Then Set prj = p: Exit For
    Next p
    If prj Is Nothing Then ApDung = "LOI: khong thay project DiaChinhVL.": Exit Function

    ' Dong cac form dang mo
    On Error Resume Next
    For i = UserForms.Count - 1 To 0 Step -1
        Unload UserForms(i)
    Next i
    On Error GoTo 0

    For i = LBound(ten) To UBound(ten)
        If Len(ten(i)) = 0 Then GoTo Tiep
        If StrComp(ten(i), "modCapNhatWeb", vbTextCompare) = 0 Then GoTo Tiep   ' khong tu thay chinh no

        Set c = Nothing
        On Error Resume Next
        Set c = prj.VBComponents(ten(i))
        On Error GoTo 0
        If c Is Nothing Then
            Select Case LCase$(Left$(ten(i), 3))
                Case "frm": kieu = 3          ' UserForm (de trong, code tu tao control)
                Case "cls": kieu = 2
                Case Else:  kieu = 1
            End Select
            On Error Resume Next
            Set c = prj.VBComponents.Add(kieu)
            c.Name = ten(i)
            If Err.Number <> 0 Then
                kq = kq & "  [LOI] khong tao duoc " & ten(i) & vbCrLf
                Err.Clear
                On Error GoTo 0
                GoTo Tiep
            End If
            On Error GoTo 0
        End If

        code = DocFile(thu & "\" & ten(i) & ".txt")
        On Error Resume Next
        With c.CodeModule
            If .CountOfLines > 0 Then .DeleteLines 1, .CountOfLines
            .AddFromString code
        End With
        If Err.Number <> 0 Then
            kq = kq & "  [LOI] " & ten(i) & ": " & Err.Description & vbCrLf
            loi = loi + 1
        End If
        Err.Clear
        On Error GoTo 0
Tiep:
    Next i
    If loi = 0 And Len(kq) = 0 Then kq = "  Tat ca module da nap thanh cong." & vbCrLf
    ApDung = kq
End Function

'------------------------------------------------------------ mang
Private Function TaiChuoi(ByVal url As String) As String
    Dim h As Object, lan As Long
    For lan = 1 To 3
        Set h = Nothing
        On Error Resume Next
        Set h = CreateObject("MSXML2.ServerXMLHTTP.6.0")
        If h Is Nothing Then Set h = CreateObject("WinHttp.WinHttpRequest.5.1")
        If h Is Nothing Then Exit Function
        Err.Clear
        h.setTimeouts 5000, 10000, 15000, 30000      ' khong treo may khi mang cham
        h.Open "GET", url & IIf(InStr(url, "?") > 0, "&", "?") & "t=" & Format$(Now, "yyyymmddhhnnss"), False
        h.setRequestHeader "Cache-Control", "no-cache"
        h.setRequestHeader "User-Agent", "DiaChinhVL"
        h.Send
        If Err.Number = 0 Then
            If h.Status = 200 Then TaiChuoi = h.responseText: Exit Function
        End If
        Err.Clear
        On Error GoTo 0
        DoEvents
    Next lan
End Function

' SHA tung module tren GitHub (1 lan goi API): ten module -> sha
Private Function DocShaGitHub() As Object
    Dim d As Object, j As String, p As Long, q As Long, ten As String, sha As String, pSau As Long
    Set d = CreateObject("Scripting.Dictionary")
    Set DocShaGitHub = d
    j = TaiChuoi("https://api.github.com/repos/lehoangtuat/DiaChinhVL/git/trees/main?recursive=1")
    If Len(j) = 0 Then Exit Function
    j = Replace(j, " ", "")
    p = InStr(j, """path"":""modules/")
    Do While p > 0
        p = p + Len("""path"":""modules/")
        q = InStr(p, j, ".txt""")
        If q = 0 Then Exit Do
        ten = Mid$(j, p, q - p)
        pSau = InStr(q, j, """path"":""")
        q = InStr(q, j, """sha"":""")
        If q > 0 And (pSau = 0 Or q < pSau) Then
            sha = Mid$(j, q + Len("""sha"":"""), 40)
            If Len(ten) > 0 And InStr(ten, "/") = 0 Then d(ten) = sha
        End If
        p = InStr(p, j, """path"":""modules/")
    Loop
End Function

Private Function DocShaCu(ByVal f As String) As Object
    Dim d As Object, t As String, a() As String, i As Long, k As Long
    Set d = CreateObject("Scripting.Dictionary")
    Set DocShaCu = d
    If Len(Dir$(f)) = 0 Then Exit Function
    t = DocFile(f)
    a = Split(Replace(t, vbCr, ""), vbLf)
    For i = 0 To UBound(a)
        k = InStr(a(i), "=")
        If k > 1 Then d(Left$(a(i), k - 1)) = Mid$(a(i), k + 1)
    Next i
End Function

Private Sub GhiShaCu(ByVal f As String, ByVal d As Object)
    Dim k As Variant, s As String
    For Each k In d.Keys
        s = s & k & "=" & d(k) & vbCrLf
    Next k
    GhiAnsi f, s
End Sub

'------------------------------------------------------------ file
Private Function GhiAnsi(ByVal path As String, ByVal s As String) As Boolean
    Dim n As Integer
    On Error GoTo Loi
    s = Replace(Replace(s, vbCrLf, vbLf), vbLf, vbCrLf)
    n = FreeFile
    Open path For Output As #n
    Print #n, s;
    Close #n
    GhiAnsi = True
    Exit Function
Loi:
    DC_ThongBao "Kh" & ChrW(244) & "ng ghi " & ChrW(273) & ChrW(432) & ChrW(7907) & "c file: " & path & vbCrLf & Err.Description, "C" & ChrW(7853) & "p nh" & ChrW(7853) & "t", vbExclamation
End Function

Private Function DocFile(ByVal path As String) As String
    Dim n As Integer, s As String
    n = FreeFile
    Open path For Binary Access Read As #n
    s = Space$(LOF(n))
    Get #n, , s
    Close #n
    DocFile = s
End Function

' So sanh phien ban "1.10" > "1.9"
Private Function SoSanhPB(ByVal a As String, ByVal b As String) As Long
    Dim x() As String, y() As String, i As Long, va As Long, vb As Long, n As Long
    x = Split(a, "."): y = Split(b, ".")
    n = UBound(x): If UBound(y) > n Then n = UBound(y)
    For i = 0 To n
        va = 0: vb = 0
        If i <= UBound(x) Then va = Val(x(i))
        If i <= UBound(y) Then vb = Val(y(i))
        If va > vb Then SoSanhPB = 1: Exit Function
        If va < vb Then SoSanhPB = -1: Exit Function
    Next i
End Function
