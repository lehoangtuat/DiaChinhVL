' SUA 22/10: mau mac dinh - uu tien mau TT26 moi (cap nhat tu GitHub ve C:\DiaChinhVL\Mau)
Private Function MauMD(ByVal moi As String, ByVal cu As String, ByVal duoi As String) As String
    MauMD = "C:\DiaChinhVL\MAU\" & cu & "." & duoi
    On Error Resume Next
    If Len(Dir$("C:\DiaChinhVL\MAU\" & moi & "." & duoi)) > 0 Then MauMD = "C:\DiaChinhVL\MAU\" & moi & "." & duoi
End Function

' SUA 22/10 (ban 3.0): GHI CO DINH 4 mau TT26 theo thu tu danh sach (1 lan / may,
'   co MAU_TT26=1 trong DiaChinhVL.ini). Chi ghi khi file mau co trong C:\DiaChinhVL\MAU.
'   0 Trich luc ban do dia chinh -> trichluc ; 1 Tach -> PL22-tachthua ;
'   2 Hop -> PL22-hopthua ; 3 Phieu do dac chinh ly -> phieuchinhlydodac
'   Doi bo mau sau nay: tang so MAU_TT26 (vd "2") de cac may ghi lai.
Private Sub ChuyenMauTT26()
    Const PB As String = "1"
    If DC_DocCauHinh("MAU_TT26", "") = PB Then Exit Sub
    GhiMauTT26 "trichluc", "MAU_TL", "INI_TL0"
    If Len(Dir$("C:\DiaChinhVL\MAU\trichluc.ini")) > 0 Then DC_GhiCauHinh "INI_TL", "C:\DiaChinhVL\MAU\trichluc.ini"
    GhiMauTT26 "PL22-tachthua", "MAU_TACH", "INI_TACH"
    GhiMauTT26 "PL22-hopthua", "MAU_HOP", "INI_HOP"
    GhiMauTT26 "phieuchinhlydodac", "MAU_TL3", "INI_TL3"
    DC_GhiCauHinh "MAU_TT26", PB
End Sub

Private Sub GhiMauTT26(ByVal ten As String, ByVal kMau As String, ByVal kIni As String)
    Dim f As String
    On Error Resume Next
    f = "C:\DiaChinhVL\MAU\" & ten
    If Len(Dir$(f & ".dgn")) = 0 Then Exit Sub
    DC_GhiCauHinh kMau, f & ".dgn"
    If Len(Dir$(f & ".ini")) > 0 Then DC_GhiCauHinh kIni, f & ".ini"
End Sub

' Goi tu clsComboEvent khi doi "Mau ho so"
Public Sub MauHSDoi()
    Dim lo As Long
    lo = Me.Controls("cboMauHS").ListIndex
    If lo < 0 Then lo = 0
    If lo = mLoai Then Exit Sub
    If mLoai >= 0 Then LuuMau                 ' luu file mau cua loai cu
    mLoai = lo
    DC_GhiCauHinh "LOAI_HS", CStr(lo)
    Select Case lo
        Case 1
            Me.Controls("txtMau").Text = DC_DocCauHinh("MAU_TACH", MauMD("PL22-tachthua", "22tach", "dgn"))
            Me.Controls("txtIni").Text = DC_DocCauHinh("INI_TACH", MauMD("PL22-tachthua", "22tach", "ini"))
            Me.Controls("lbl_txtSoThua").Caption = "S" & ChrW(7889) & " th" & ChrW(7917) & "a c" & ChrW(7847) & "n t" & ChrW(225) & "ch:"
        Case 2
            Me.Controls("txtMau").Text = DC_DocCauHinh("MAU_HOP", MauMD("PL22-hopthua", "22hop", "dgn"))
            Me.Controls("txtIni").Text = DC_DocCauHinh("INI_HOP", MauMD("PL22-hopthua", "22hop", "ini"))
            Me.Controls("lbl_txtSoThua").Caption = "C" & ChrW(225) & "c th" & ChrW(7917) & "a h" & ChrW(7907) & "p (12,13):"
        Case 3
            ' SUA 16/10/2026: mau VL (Trich luc + Phieu do dac chinh ly), file rieng
            Me.Controls("txtMau").Text = DC_DocCauHinh("MAU_TL3", MauMD("phieuchinhlydodac", "trichluc", "dgn"))
            Me.Controls("txtIni").Text = DC_DocCauHinh("INI_TL3", MauMD("phieuchinhlydodac", "trichluc", "ini"))
            Me.Controls("lbl_txtSoThua").Caption = "S" & ChrW(7889) & " th" & ChrW(7917) & "a tr" & ChrW(237) & "ch l" & ChrW(7909) & "c:"
        Case Else
            Me.Controls("txtMau").Text = DC_DocCauHinh("MAU_TL", "C:\DiaChinhVL\TrichLucBanDo_mau.txt")
            Me.Controls("txtIni").Text = DC_DocCauHinh("INI_TL0", DC_DocCauHinh("INI_TL", ""))

    Me.Controls("txtLvCat").Enabled = (lo = 1)
    Me.Controls("lbl_lvcat").Enabled = (lo = 1)

Private Sub FillXa(ByVal tinh As String)
    Dim cboX As Object
    Set cboX = Me.Controls("cboXa")
    cboX.Clear
    Dim ds As Collection, i As Long
    Set ds = DsXa(tinh)
    For i = 1 To ds.Count
        cboX.AddItem ds.Item(i)
    Next i
    If ds.Count > 0 Then cboX.ListIndex = 0
End Sub

    DC_GhiCauHinh "LV_SD_ON", IIf(Me.Controls("chkLvSD").Value, "1", "0")
End Sub

Private Sub LuuMau()
    Dim hau As String
    Select Case mLoai
        Case 1: hau = "TACH"
        Case 2: hau = "HOP"
        Case 3: hau = "TL3"
        Case Else: hau = "TL"
    End Select
    DC_GhiCauHinh "MAU_" & hau, Trim$(Me.Controls("txtMau").Text)
    If mLoai = 0 Then
        ' INI_TL = ini DANG DUNG khi xuat trich luc (modMauDgn doc); moi mau luu rieng
        DC_GhiCauHinh "INI_TL0", Trim$(Me.Controls("txtIni").Text)
    Else
        DC_GhiCauHinh "INI_" & hau, Trim$(Me.Controls("txtIni").Text)
    End If
    If mLoai = 0 Or mLoai = 3 Then DC_GhiCauHinh "INI_TL", Trim$(Me.Controls("txtIni").Text)
    If mLoai = 1 Then DC_GhiCauHinh "TACH_LV", Trim$(Me.Controls("txtLvCat").Text)
End Sub

Private Sub LuuDonViHC()
    DC_GhiCauHinh "TINH", CStr(Me.Controls("cboTinh").Value)
    DC_GhiCauHinh "XA", CStr(Me.Controls("cboXa").Value)
    DC_GhiCauHinh "HUYEN", Me.Controls("txtHuyen").Text
End Sub

Public Sub XuLyNut(ByVal ma As String)
    If ma = "DONG" Then LuuViTri: Unload Me: Exit Sub
    If ma = "CHONMAU" Then
        Dim f As String
        f = DC_ChonFile("File m" & ChrW(7851) & "u (*.dgn;*.txt)|*.dgn;*.txt|T" & ChrW(7845) & "t c" & ChrW(7843) & "|*.*", "Chon file mau trich luc", Me.Controls("txtMau").Text)
        If Len(f) > 0 Then Me.Controls("txtMau").Text = f: LuuMau
        Exit Sub
    End If
    If ma = "CHONINI" Then
        Dim g As String
        g = DC_ChonFile("File c" & ChrW(7845) & "u h" & ChrW(236) & "nh (*.ini)|*.ini|T" & ChrW(7845) & "t c" & ChrW(7843) & "|*.*", "Chon file cau hinh", Me.Controls("txtIni").Text)
        If Len(g) > 0 Then Me.Controls("txtIni").Text = g: LuuMau
        Exit Sub
    End If
    If ma = "TATBD" Then LuuLevel: DC_TatLopBanDo Trim$(Me.Controls("txtLvHS").Text), Trim$(Me.Controls("txtLvSD").Text): Exit Sub
    If ma = "BATBD" Then DC_BatLopBanDo: Exit Sub
    If ma = "XOAHS" Then DC_XoaTheoLevel Trim$(Me.Controls("txtLvHS").Text): Exit Sub
    If ma = "XOASD" Then DC_XoaTheoLevel Trim$(Me.Controls("txtLvSD").Text): Exit Sub
    LuuLevel
    If ma = "LUUHS" Then
        DC_LuuFileHoSo Trim$(Me.Controls("txtLvHS").Text), Trim$(Me.Controls("txtLvSD").Text)
        Exit Sub
    End If
    LuuMau
    DC_GhiCauHinh "TL_TYLE", IIf(Val(Me.Controls("cboTyLe").Value) > 0, CStr(Val(Me.Controls("cboTyLe").Value)), "")
    DC_GhiCauHinh "TL_TYLE_TAY", IIf(Val(Me.Controls("txtTyLeTay").Text) > 0, CStr(Val(Me.Controls("txtTyLeTay").Text)), "")
    If mLoai = 1 Or mLoai = 2 Then XuLyTachHop ma: Exit Sub
    If ma = "CHON" Then
        If DC_DaTatLop() Then DC_BatLopBanDo     ' hien lai ban do de bam chon thua
        LuuDonViHC
        LuuViTri
        Me.Hide
        DC_ChonThuaTrichLuc Me.Controls("txtCSDL").Text, _
                 Me.Controls("txtMau").Text, Me.Controls("txtKieu").Text, _
                 Me.Controls("txtTo").Text, CStr(Me.Controls("cboXa").Value), _
                 Me.Controls("txtHuyen").Text, CStr(Me.Controls("cboTinh").Value), _
                 "", val(Me.Controls("txtCao").Text), _
                 Me.Controls("chkMau").Value, TyLeChon()
        Exit Sub
    End If
    If Len(Trim$(Me.Controls("txtSoThua").Text)) = 0 Then
        MsgBox "Nh" & ChrW(7853) & "p s" & ChrW(7889) & " th" & ChrW(7917) & "a c" & ChrW(7847) & "n tr" & ChrW(237) & "ch l" & ChrW(7909) & "c.", vbExclamation: Exit Sub
    End If
    LuuDonViHC                         ' nho lai don vi hanh chinh vua dung
    LuuViTri
    Me.Hide
    ChayTrichLuc Me.Controls("txtCSDL").Text, Me.Controls("txtSoThua").Text, _
                 Me.Controls("txtMau").Text, Me.Controls("txtKieu").Text, _
                 Me.Controls("txtTo").Text, CStr(Me.Controls("cboXa").Value), _
                 Me.Controls("txtHuyen").Text, CStr(Me.Controls("cboTinh").Value), _
                 "", val(Me.Controls("txtCao").Text), _
                 Me.Controls("chkMau").Value, TyLeChon()
    Me.Show vbModeless
End Sub

' Tach / hop thua: "Chon thua" (bam tren ban do) hoac "Chap nhan" (go so thua)
Private Sub XuLyTachHop(ByVal ma As String)
    Dim tieuDe As String
    If mLoai = 2 Then tieuDe = "H" & ChrW(7907) & "p th" & ChrW(7917) & "a" Else tieuDe = "T" & ChrW(225) & "ch th" & ChrW(7917) & "a"
    LuuDonViHC
    If ma = "CHON" Then
        If DC_DaTatLop() Then DC_BatLopBanDo     ' hien lai ban do de bam chon thua
        LuuViTri
        Me.Hide
        DC_ChonThuaTrichLuc Me.Controls("txtCSDL").Text, Me.Controls("txtMau").Text, "T", _
                 Me.Controls("txtTo").Text, CStr(Me.Controls("cboXa").Value), _
                 Me.Controls("txtHuyen").Text, CStr(Me.Controls("cboTinh").Value), _
                 "", 1, True, TyLeChon(), _
                 mLoai, Me.Controls("txtIni").Text, Me.Controls("txtLvCat").Text
        Exit Sub
    End If
    If Len(Trim$(Me.Controls("txtSoThua").Text)) = 0 Then
        DC_ThongBao IIf(mLoai = 2, "Nh" & ChrW(7853) & "p c" & ChrW(225) & "c s" & ChrW(7889) & " th" & ChrW(7917) & "a c" & ChrW(7847) & "n h" & ChrW(7907) & "p (vd 12,13) ho" & ChrW(7863) & "c b" & ChrW(7845) & "m Ch" & ChrW(7885) & "n th" & ChrW(7917) & "a.", _
                    "Nh" & ChrW(7853) & "p s" & ChrW(7889) & " th" & ChrW(7917) & "a c" & ChrW(7847) & "n t" & ChrW(225) & "ch ho" & ChrW(7863) & "c b" & ChrW(7845) & "m Ch" & ChrW(7885) & "n th" & ChrW(7917) & "a."), tieuDe, vbExclamation
        Exit Sub
    End If
    Dim goc As Collection, loi As String
    Set goc = DC_LayThuaCSDL(Me.Controls("txtCSDL").Text, Me.Controls("txtSoThua").Text, loi)
    If goc Is Nothing Then DC_ThongBao loi, tieuDe, vbExclamation: Exit Sub
    LuuViTri
    Me.Hide
    DC_ChayTachHop mLoai, goc, Me.Controls("txtMau").Text, Me.Controls("txtIni").Text, _
                   Me.Controls("txtTo").Text, CStr(Me.Controls("cboXa").Value), _
                   Me.Controls("txtHuyen").Text, CStr(Me.Controls("cboTinh").Value), _
                   TyLeChon(), Me.Controls("txtLvCat").Text
    Me.Show vbModeless
End Sub
