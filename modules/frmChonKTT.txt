' Doi tinh -> dien KTT (goi tu XuLyNut khi bam, va khi chay)
Private Function KTTTheoTinh() As Double
    Dim it As Variant, cb As Object
    Set cb = Me.Controls("cboTinh")
    If cb.ListIndex >= 0 Then
        it = mDs(cb.ListIndex + 1)
        KTTTheoTinh = it(1)
    End If
End Function

Public Sub XuLyNut(ByVal ma As String)
    Dim k As Double, kTinh As Double, tay As Double
    If ma = "DONG" Then Unload Me: Exit Sub
    kTinh = KTTTheoTinh()
    tay = Val(Replace(Me.Controls("txtKTT").Text, ",", "."))
    ' chon tinh -> dung KTT cua tinh (tru khi nguoi dung sua tay o KTT khac gia tri cu)
    If kTinh > 0 And (tay = Val(DC_DocCauHinh("KTT", "105.5")) Or tay <= 0) Then k = kTinh Else k = tay
    If k < 100 Or k > 112 Then
        DC_ThongBao "Kinh tuy" & ChrW(7871) & "n tr" & ChrW(7909) & "c kh" & ChrW(244) & "ng h" & ChrW(7907) & "p l" & ChrW(7879) & " (100 - 112 " & ChrW(273) & ChrW(7897) & ").", "V" & ChrW(7879) & " tinh", vbExclamation
        Exit Sub
    End If
    Me.Controls("txtKTT").Text = CStr(k)
    If Me.Controls("cboTinh").ListIndex >= 0 Then DC_GhiCauHinh "KTT_TINH", Me.Controls("cboTinh").Text
    KTT_ChayViec k, Me.Controls("chkD7").Value, ma
End Sub
