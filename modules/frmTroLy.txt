' Cap nhat tieu de theo mau dang chon (mo lai tu menu mau khac)
Private Sub HienMau()
    Static mCu As Long, daCo As Boolean
    On Error Resume Next
    Me.Controls("lblMau").Caption = "M" & ChrW(7851) & "u: " & TL_TenMau(gTLMau)
    If daCo And mCu <> gTLMau Then Me.Controls("txtLenh").Text = DC_DocCauHinh("AI_LENH" & gTLMau, TL_ViDu(gTLMau))
    mCu = gTLMau: daCo = True
End Sub

Public Sub XuLyNut(ByVal ma As String)
    If ma = "DONG" Then Unload Me: Exit Sub
    Dim lenh As String, kq As String
    lenh = Me.Controls("txtLenh").Text
    DC_GhiCauHinh "AI_LENH" & gTLMau, Replace(Replace(lenh, vbCr, " "), vbLf, " ")
    Me.Controls("txtKQ").Text = ChrW(272) & "ang ch" & ChrW(7841) & "y ..."
    Me.Hide
    kq = TL_ChayLenh(lenh)
    Me.Show vbModeless
    Me.Controls("txtKQ").Text = kq
End Sub

Private Sub ThemNut(ByVal ma As String, ByVal cap As String, ByVal x As Single, ByVal y As Single, ByVal w As Single)
    Dim b As Object, hd As clsNutTL
    Set b = Me.Controls.Add("Forms.CommandButton.1", "cmd" & ma, True)
    b.Caption = cap: b.Left = x: b.Top = y: b.Width = w: b.Height = 22
    b.Font.Name = "Arial": b.Font.size = 8
    Set hd = New clsNutTL
    Set hd.btn = b: Set hd.Form = Me: hd.ma = ma
    mNut.Add hd
End Sub
