' Tra ve so vung ; tomTat = cau bao tong hop (kieu he thong TW)
Public Function KV_KiemTra(ByVal ranhCSV As String, ByVal lvSo As String, ByVal lvLD As String, _
                           ByVal lvDT As String, ByVal snap As Double, ByVal chiaGiao As Boolean, _
                           ByRef tomTat As String) As Long
    Dim segs As Collection, vungs As Collection, tia As Long
    Dim minX As Double, minY As Double, maxX As Double, maxY As Double
    Set KV_DsLoi = New Collection
    tomTat = ""
    If snap <= 0 Then snap = 0.05
    ShowStatus "Kiem tra vung thua: doc ranh ..."
    Set segs = LayDoanNhieuLevel(ranhCSV, snap, minX, minY, maxX, maxY)
    If segs Is Nothing Then tomTat = "Kh" & ChrW(244) & "ng " & ChrW(273) & ChrW(7885) & "c " & ChrW(273) & ChrW(432) & ChrW(7907) & "c " & ChrW(273) & ChrW(432) & ChrW(7901) & "ng ranh tr" & ChrW(234) & "n l" & ChrW(7899) & "p " & ranhCSV: Exit Function
    If segs.Count = 0 Then tomTat = "Kh" & ChrW(244) & "ng " & ChrW(273) & ChrW(7885) & "c " & ChrW(273) & ChrW(432) & ChrW(7907) & "c " & ChrW(273) & ChrW(432) & ChrW(7901) & "ng ranh tr" & ChrW(234) & "n l" & ChrW(7899) & "p " & ranhCSV: Exit Function
    If chiaGiao Then
        ShowStatus "Kiem tra vung thua: chia doan tai giao diem ..."
        Set segs = NoiDoanGiao(segs, snap)
    End If
    ShowStatus "Kiem tra vung thua: ghep vung ..."
    gGhepVungMin = 0.000001          ' SUA 05/11: giu ca vung manh (TW bao loi ca vung rat nho)
    Set vungs = GhepVung(segs, snap, tia)
    gGhepVungMin = 0#
    If vungs Is Nothing Then tomTat = "Kh" & ChrW(244) & "ng gh" & ChrW(233) & "p " & ChrW(273) & ChrW(432) & ChrW(7907) & "c v" & ChrW(249) & "ng th" & ChrW(7917) & "a n" & ChrW(224) & "o.": Exit Function
    If vungs.Count = 0 Then tomTat = "Kh" & ChrW(244) & "ng gh" & ChrW(233) & "p " & ChrW(273) & ChrW(432) & ChrW(7907) & "c v" & ChrW(249) & "ng th" & ChrW(7917) & "a n" & ChrW(224) & "o.": Exit Function

    Dim nV As Long, i As Long, j As Long
    nV = vungs.Count
    Dim bx1() As Double, by1() As Double, bx2() As Double, by2() As Double, rg() As Double
    Dim dem() As Long, soTim() As String
    ReDim bx1(1 To nV): ReDim by1(1 To nV): ReDim bx2(1 To nV): ReDim by2(1 To nV)
    ReDim dem(1 To nV, 1 To 3): ReDim soTim(1 To nV)
    For i = 1 To nV
        rg = vungs.Item(i).ring
        BaoRing rg, bx1(i), by1(i), bx2(i), by2(i)
    Next i

    ' --- gom chu tren 3 lop
    Dim lv(1 To 3) As Long, o As Level
    Set o = DC_TimLevel(lvSo): If Not o Is Nothing Then lv(1) = o.Number Else lv(1) = -99
    Set o = DC_TimLevel(lvLD): If Not o Is Nothing Then lv(2) = o.Number Else lv(2) = -99
    Set o = DC_TimLevel(lvDT): If Not o Is Nothing Then lv(3) = o.Number Else lv(3) = -99
    ShowStatus "Kiem tra vung thua: doc chu ..."
    Dim cx() As Double, cy() As Double, ck() As Long, cs() As String, nC As Long
    ReDim cx(1 To 1000): ReDim cy(1 To 1000): ReDim ck(1 To 1000): ReDim cs(1 To 1000)
    Dim esc As ElementScanCriteria, ee As ElementEnumerator
    Set esc = New ElementScanCriteria
    esc.ExcludeAllTypes
    esc.IncludeType msdElementTypeText
    esc.IncludeType msdElementTypeTextNode
    esc.IncludeType msdElementTypeCellHeader
    Set ee = ActiveModelReference.Scan(esc)
    Do While ee.MoveNext
        GomChu ee.Current, lv, cx, cy, ck, cs, nC
    Loop

    ' --- gan chu vao vung
    ShowStatus "Kiem tra vung thua: doi chieu " & nC & " chu voi " & nV & " vung ..."
    Dim v As clsVung, k As Long, coVung() As Boolean
    ReDim coVung(1 To nC + 1)
    For j = 1 To nC
        For i = 1 To nV
            If cx(j) >= bx1(i) And cx(j) <= bx2(i) And cy(j) >= by1(i) And cy(j) <= by2(i) Then
                Set v = vungs.Item(i)
                If TrongVung(v, cx(j), cy(j)) Then
                    dem(i, ck(j)) = dem(i, ck(j)) + 1
                    coVung(j) = True
                    If ck(j) = 1 Then
                        If Len(soTim(i)) > 0 Then soTim(i) = soTim(i) & ", "
                        soTim(i) = soTim(i) & cs(j)
                    End If
                    Exit For
                End If
            End If
        Next i
        If j Mod 200 = 0 Then DoEvents
    Next j

    ' --- tong hop loi
    Dim ten(1 To 3) As String, mo As String, nThieu(1 To 3) As Long, nThua(1 To 3, 2 To 9) As Long
    ten(1) = "s" & ChrW(7889) & " hi" & ChrW(7879) & "u th" & ChrW(7917) & "a": ten(2) = "lo" & ChrW(7841) & "i " & ChrW(273) & ChrW(7845) & "t": ten(3) = "di" & ChrW(7879) & "n t" & ChrW(237) & "ch"
    For i = 1 To nV
        mo = ""
        For k = 1 To 3
            If dem(i, k) = 0 Then
                mo = mo & IIf(Len(mo) > 0, "; ", "") & "thi" & ChrW(7871) & "u " & ten(k)
                nThieu(k) = nThieu(k) + 1
            ElseIf dem(i, k) > 1 Then
                mo = mo & IIf(Len(mo) > 0, "; ", "") & dem(i, k) & " " & ten(k)
                If dem(i, k) <= 9 Then nThua(k, dem(i, k)) = nThua(k, dem(i, k)) + 1 Else nThua(k, 9) = nThua(k, 9) + 1
            End If
        Next k
        If Len(mo) > 0 Then
            If vungs.Item(i).Dt < 1# And dem(i, 1) = 0 Then
                mo = "V" & ChrW(217) & "NG M" & ChrW(7842) & "NH " & Format$(vungs.Item(i).Dt, "0.000") & " m2 (ranh ch" & ChrW(7891) & "ng/h" & ChrW(7903) & "): " & mo
            End If
            KV_DsLoi.Add Array(vungs.Item(i), mo, soTim(i))
        End If
    Next i

    ' SUA 06/11: so thua nam NGOAI moi vung (ranh ho / vung khong ghep duoc) -> bao loi
    Dim nNgoai As Long, vN As clsVung, rN() As Double, hN As Double
    For j = 1 To nC
        If ck(j) = 1 And Not coVung(j) Then
            nNgoai = nNgoai + 1
            hN = 2#
            ReDim rN(0 To 4, 0 To 1)
            rN(0, 0) = cx(j) - hN: rN(0, 1) = cy(j) - hN
            rN(1, 0) = cx(j) + hN: rN(1, 1) = cy(j) - hN
            rN(2, 0) = cx(j) + hN: rN(2, 1) = cy(j) + hN
            rN(3, 0) = cx(j) - hN: rN(3, 1) = cy(j) + hN
            rN(4, 0) = rN(0, 0): rN(4, 1) = rN(0, 1)
            Set vN = New clsVung
            vN.ring = rN: vN.cx = cx(j): vN.cy = cy(j): vN.Dt = -1#: vN.HasRing = True
            KV_DsLoi.Add Array(vN, "S" & ChrW(7889) & " th" & ChrW(7917) & "a n" & ChrW(7857) & "m NGO" & ChrW(192) & "I v" & ChrW(249) & "ng kh" & ChrW(233) & "p k" & ChrW(237) & "n (ranh h" & ChrW(7903) & " / thi" & ChrW(7871) & "u ranh chia th" & ChrW(7917) & "a)", cs(j))
        End If
    Next j

    ' SUA 07/11: TRUNG SO THUA tren cung to (TW khong chap nhan)
    Dim dTr As Object, kTr As String, nTrung As Long, sKey As Variant, idx() As String, q As Long, c As Long
    Set dTr = CreateObject("Scripting.Dictionary")
    For j = 1 To nC
        If ck(j) = 1 Then
            kTr = KV_ChuanSo(cs(j))
            If Len(kTr) > 0 Then
                If dTr.Exists(kTr) Then dTr(kTr) = dTr(kTr) & "," & j Else dTr.Add kTr, CStr(j)
            End If
        End If
    Next j
    For Each sKey In dTr.Keys
        idx = Split(dTr(sKey), ",")
        If UBound(idx) >= 1 Then
            nTrung = nTrung + 1
            For q = 0 To UBound(idx)
                c = CLng(idx(q))
                hN = 2#
                ReDim rN(0 To 4, 0 To 1)
                rN(0, 0) = cx(c) - hN: rN(0, 1) = cy(c) - hN
                rN(1, 0) = cx(c) + hN: rN(1, 1) = cy(c) - hN
                rN(2, 0) = cx(c) + hN: rN(2, 1) = cy(c) + hN
                rN(3, 0) = cx(c) - hN: rN(3, 1) = cy(c) + hN
                rN(4, 0) = rN(0, 0): rN(4, 1) = rN(0, 1)
                Set vN = New clsVung
                vN.ring = rN: vN.cx = cx(c): vN.cy = cy(c): vN.Dt = -1#: vN.HasRing = True
                KV_DsLoi.Add Array(vN, "TR" & ChrW(217) & "NG s" & ChrW(7889) & " th" & ChrW(7917) & "a " & cs(c) & " (" & (q + 1) & "/" & (UBound(idx) + 1) & " tr" & ChrW(234) & "n t" & ChrW(7901) & ") - TW kh" & ChrW(244) & "ng ch" & ChrW(7845) & "p nh" & ChrW(7853) & "n", cs(c))
            Next q
        End If
    Next sKey

    Dim s As String, m As Long
    For m = 2 To 9
        s = ""
        For k = 1 To 3
            If nThua(k, m) > 0 Then s = s & IIf(Len(s) > 0, "; ", "") & "c" & ChrW(243) & " " & m & " " & ten(k)
        Next k
        If Len(s) > 0 Then tomTat = tomTat & "- v" & ChrW(249) & "ng th" & ChrW(7917) & "a " & s & " (ch" & ChrW(7881) & " " & ChrW(273) & ChrW(432) & ChrW(7907) & "c 1)" & vbCrLf
    Next m
    For k = 1 To 3
        If nThieu(k) > 0 Then tomTat = tomTat & "- " & nThieu(k) & " v" & ChrW(249) & "ng th" & ChrW(7917) & "a thi" & ChrW(7871) & "u " & ten(k) & vbCrLf
    Next k
    If nTrung > 0 Then tomTat = tomTat & "- " & nTrung & " s" & ChrW(7889) & " th" & ChrW(7917) & "a b" & ChrW(7883) & " TR" & ChrW(217) & "NG tr" & ChrW(234) & "n t" & ChrW(7901) & " (TW kh" & ChrW(244) & "ng ch" & ChrW(7845) & "p nh" & ChrW(7853) & "n)" & vbCrLf
    If nNgoai > 0 Then tomTat = tomTat & "- " & nNgoai & " s" & ChrW(7889) & " th" & ChrW(7917) & "a n" & ChrW(7857) & "m ngo" & ChrW(224) & "i v" & ChrW(249) & "ng kh" & ChrW(233) & "p k" & ChrW(237) & "n (ki" & ChrW(7875) & "m tra ranh h" & ChrW(7903) & ")" & vbCrLf
    If Len(tomTat) = 0 Then
        tomTat = ChrW(272) & ChrW(7841) & "t: " & nV & " v" & ChrW(249) & "ng th" & ChrW(7917) & "a " & ChrW(273) & ChrW(7873) & "u c" & ChrW(243) & " " & ChrW(273) & ChrW(250) & "ng 1 s" & ChrW(7889) & " hi" & ChrW(7879) & "u th" & ChrW(7917) & "a, 1 lo" & ChrW(7841) & "i " & ChrW(273) & ChrW(7845) & "t, 1 di" & ChrW(7879) & "n t" & ChrW(237) & "ch."
    Else
        tomTat = KV_DsLoi.Count & "/" & nV & " v" & ChrW(249) & "ng th" & ChrW(7917) & "a c" & ChrW(243) & " l" & ChrW(7895) & "i:" & vbCrLf & tomTat
    End If
    ShowStatus ""
    KV_KiemTra = nV
End Function

' Diem trong vung (trong ranh ngoai, ngoai cac dao)
Private Function TrongVung(ByVal v As clsVung, ByVal x As Double, ByVal y As Double) As Boolean
    Dim rg() As Double, k As Long
    rg = v.ring
    If Not DiemTrongPolygon(rg, x, y) Then Exit Function
    For k = 1 To v.Dao.Count
        rg = v.Dao.Item(k)
        If DiemTrongPolygon(rg, x, y) Then Exit Function
    Next k
    TrongVung = True
End Function

' Gom chu (text / text node / chu trong cell) tren 3 lop dich
Private Sub GomChu(ByVal el As Element, lv() As Long, cx() As Double, cy() As Double, _
                   ck() As Long, cs() As String, ByRef n As Long)
    Dim k As Long, s As String, sb As ElementEnumerator, lvE As Long
    On Error Resume Next
    If el.Type = msdElementTypeCellHeader Then
        Set sb = el.AsCellElement.GetSubElements
        If sb Is Nothing Then Exit Sub
        Do While sb.MoveNext
            GomChu sb.Current, lv, cx, cy, ck, cs, n
        Loop
        Exit Sub
    End If
    If el.Type <> msdElementTypeText And el.Type <> msdElementTypeTextNode Then Exit Sub
    lvE = DC_LevelSo(el)
    For k = 1 To 3
        If lvE = lv(k) Then Exit For
    Next k
    If k > 3 Then Exit Sub
    If el.Type = msdElementTypeText Then
        s = Trim$(el.AsTextElement.Text)
    Else
        Set sb = el.AsTextNodeElement.GetSubElements
        Do While sb.MoveNext
            If sb.Current.Type = msdElementTypeText Then s = s & sb.Current.AsTextElement.Text
        Loop
        s = Trim$(s)
    End If
    If Len(s) = 0 Then Exit Sub
    n = n + 1
    If n > UBound(cx) Then
        ReDim Preserve cx(1 To n * 2): ReDim Preserve cy(1 To n * 2)
        ReDim Preserve ck(1 To n * 2): ReDim Preserve cs(1 To n * 2)
    End If
    ' SUA 05/11: dung DIEM CHEN (origin) cua chu nhu he thong TW (khong dung tam khung chu)
    Dim o As Point3d
    If el.Type = msdElementTypeText Then o = el.AsTextElement.Origin Else o = el.AsTextNodeElement.Origin
    cx(n) = o.X: cy(n) = o.Y
    ' SUA 07/11: theo thuc te TW - chu chi can KHONG de len ranh -> xet theo TAM khung chu
    cx(n) = (el.Range.Low.X + el.Range.High.X) / 2#
    cy(n) = (el.Range.Low.Y + el.Range.High.Y) / 2#
    ck(n) = k: cs(n) = s
    Err.Clear
End Sub

' Ve dau loi (cell "LOIVUNG": vong tron + chu) tai tam cac vung loi. Tra ve so dau.
Public Function KV_VeDauLoi(ByVal lvLoi As String, ByVal cao As Double) As Long
    Dim i As Long, a As Variant, v As clsVung, els(0 To 1) As Element, cel As Element, lv As Level
    If KV_DsLoi Is Nothing Then Exit Function
    If cao <= 0 Then cao = 1
    Set lv = DC_TimLevel(lvLoi)
    On Error Resume Next
    ActiveSettings.TextStyle.Height = cao: ActiveSettings.TextStyle.Width = cao
    ActiveSettings.TextStyle.Justification = msdTextJustificationLeftCenter
    For i = 1 To KV_DsLoi.Count
        a = KV_DsLoi.Item(i): Set v = a(0)
        Set els(0) = CreateEllipseElement2(Nothing, Point3dFromXYZ(v.cx, v.cy, 0#), cao * 1.5, cao * 1.5, Matrix3dIdentity)
        Set els(1) = CreateTextElement1(Nothing, CStr(a(1)), Point3dFromXYZ(v.cx + cao * 2, v.cy, 0#), Matrix3dIdentity)
        Set cel = CreateCellElement1("LOIVUNG", els, Point3dFromXYZ(v.cx, v.cy, 0#), False)
        If Not cel Is Nothing Then
            If Not lv Is Nothing Then Set cel.Level = lv
            cel.Color = 3
            ActiveModelReference.AddElement cel
            If Err.Number = 0 Then KV_VeDauLoi = KV_VeDauLoi + 1
        End If
        Err.Clear
    Next i
End Function

' Xoa cac dau loi "LOIVUNG"
Public Function KV_XoaDauLoi() As Long
    Dim esc As ElementScanCriteria, ee As ElementEnumerator, ds As New Collection, i As Long
    Set esc = New ElementScanCriteria
    esc.ExcludeAllTypes
    esc.IncludeType msdElementTypeCellHeader
    Set ee = ActiveModelReference.Scan(esc)
    On Error Resume Next
    Do While ee.MoveNext
        If UCase$(ee.Current.AsCellElement.Name) = "LOIVUNG" Then ds.Add ee.Current
        Err.Clear
    Loop
    For i = 1 To ds.Count
        ActiveModelReference.RemoveElement ds.Item(i)
        If Err.Number = 0 Then KV_XoaDauLoi = KV_XoaDauLoi + 1
        Err.Clear
    Next i
End Function

'============================================================
' SUA 06/11: DO DAU RANH LECH NHO (TW coi la HO, MicroStation van tao vung duoc)
'   Dau mut cua doan ranh khong daTrung DUNG 1 dinh / khong nam DUNG tren 1 canh
'   cua doi tuong ranh khac, nhung cach < tolLech (mac dinh 1 mm).
'   Ket qua dua vao KV_DsLoi (dung chung danh sach / zoom / ve dau loi).
'============================================================

Public Function KV_DoDauLech(ByVal ranhCSV As String, ByVal tolLech As Double, ByRef tomTat As String) As Long
    Const CELL As Double = 2#
    Const SAI_TRUNG As Double = 0.000000005     ' <= 5e-9 m coi nhu daTrung khit
    Dim arr() As String, i As Long, j As Long, k As Long, lvN() As Long, nL As Long, o As Level
    Set KV_DsLoi = New Collection
    tomTat = ""
    If tolLech <= 0 Then tolLech = 0.001
    arr = Split(ranhCSV, ",")
    ReDim lvN(0 To UBound(arr) + 1)
    For i = LBound(arr) To UBound(arr)
        If Len(Trim$(arr(i))) > 0 Then
            Set o = DC_TimLevel(Trim$(arr(i)))
            If Not o Is Nothing Then lvN(nL) = o.Number: nL = nL + 1
        End If
    Next i
    If nL = 0 Then tomTat = "Kh" & ChrW(244) & "ng t" & ChrW(236) & "m th" & ChrW(7845) & "y l" & ChrW(7899) & "p ranh " & ranhCSV: Exit Function

    dlNV = 0: dlNS = 0: dlNE = 0
    ReDim dlVX(1 To 1000): ReDim dlVY(1 To 1000): ReDim dlVE(1 To 1000)
    ReDim dlSX1(1 To 1000): ReDim dlSY1(1 To 1000): ReDim dlSX2(1 To 1000): ReDim dlSY2(1 To 1000): ReDim dlSE(1 To 1000)
    ReDim dlEX(1 To 1000): ReDim dlEY(1 To 1000): ReDim dlEE(1 To 1000)

    ShowStatus "Do dau ranh lech: doc ranh ..."
    Dim esc As ElementScanCriteria, ee As ElementEnumerator, el As Element, lvE As Long, ok As Boolean
    Dim nEl As Long, pts() As Point3d, sb As ElementEnumerator, kin As Boolean
    Set esc = New ElementScanCriteria
    esc.ExcludeAllTypes
    esc.IncludeType msdElementTypeLine
    esc.IncludeType msdElementTypeLineString
    esc.IncludeType msdElementTypeShape
    esc.IncludeType msdElementTypeComplexString
    esc.IncludeType msdElementTypeComplexShape
    Set ee = ActiveModelReference.Scan(esc)
    On Error Resume Next
    Do While ee.MoveNext
        Set el = ee.Current
        lvE = el.Level.Number
        ok = False
        For k = 0 To nL - 1
            If lvE = lvN(k) Then ok = True: Exit For
        Next k
        If ok Then
            nEl = nEl + 1
            Erase pts
            kin = (el.Type = msdElementTypeShape Or el.Type = msdElementTypeComplexShape)
            If el.Type = msdElementTypeLine Or el.Type = msdElementTypeLineString Then
                pts = el.AsLineElement.GetVertices()
                DL_ThemDay pts, nEl, Not kin
            ElseIf el.Type = msdElementTypeShape Then
                pts = el.AsShapeElement.GetVertices()
                DL_ThemDay pts, nEl, False
            Else
                Set sb = Nothing
                If el.Type = msdElementTypeComplexString Then Set sb = el.AsComplexStringElement.GetSubElements Else Set sb = el.AsComplexShapeElement.GetSubElements
                Dim ax() As Double, ay() As Double, na As Long, q As Long
                na = 0: ReDim ax(1 To 64): ReDim ay(1 To 64)
                Err.Clear
                If Not sb Is Nothing Then
                Do While sb.MoveNext
                    Erase pts
                    pts = sb.Current.AsLineElement.GetVertices()
                    If Err.Number = 0 Then
                        For q = LBound(pts) To UBound(pts)
                            na = na + 1
                            If na > UBound(ax) Then ReDim Preserve ax(1 To na * 2): ReDim Preserve ay(1 To na * 2)
                            ax(na) = pts(q).X: ay(na) = pts(q).Y
                        Next q
                    End If
                    Err.Clear
                Loop
                End If
                If na >= 2 Then
                    ReDim pts(0 To na - 1)
                    For q = 1 To na
                        pts(q - 1).X = ax(q): pts(q - 1).Y = ay(q)
                    Next q
                    DL_ThemDay pts, nEl, Not kin
                End If
            End If
        End If
        Err.Clear
    Loop
    On Error GoTo 0
    If dlNE = 0 Then tomTat = "Kh" & ChrW(244) & "ng c" & ChrW(243) & " " & ChrW(273) & ChrW(7847) & "u m" & ChrW(250) & "t ranh n" & ChrW(224) & "o tr" & ChrW(234) & "n l" & ChrW(7899) & "p " & ranhCSV: Exit Function

    ' --- luoi o vuong cho dinh va canh
    ShowStatus "Do dau ranh lech: lap luoi ..."
    Dim dV As Object, dS As Object, key As String, ix As Long, iy As Long
    Set dV = CreateObject("Scripting.Dictionary")
    Set dS = CreateObject("Scripting.Dictionary")
    For i = 1 To dlNV
        key = CStr(Int(dlVX(i) / CELL)) & "_" & CStr(Int(dlVY(i) / CELL))
        If dV.Exists(key) Then dV(key) = dV(key) & "," & i Else dV.Add key, CStr(i)
    Next i
    Dim nb As Long, sL As Double, tt As Double, px As Double, py As Double, lastK As String
    For i = 1 To dlNS
        sL = Sqr((dlSX2(i) - dlSX1(i)) ^ 2 + (dlSY2(i) - dlSY1(i)) ^ 2)
        nb = Int(sL / (CELL * 0.5)) + 1
        lastK = ""
        For k = 0 To nb
            tt = k / nb
            px = dlSX1(i) + (dlSX2(i) - dlSX1(i)) * tt: py = dlSY1(i) + (dlSY2(i) - dlSY1(i)) * tt
            key = CStr(Int(px / CELL)) & "_" & CStr(Int(py / CELL))
            If key <> lastK Then
                If dS.Exists(key) Then dS(key) = dS(key) & "," & i Else dS.Add key, CStr(i)
                lastK = key
            End If
        Next k
        If i Mod 500 = 0 Then DoEvents
    Next i

    ' --- xet tung dau mut
    ShowStatus "Do dau ranh lech: xet " & dlNE & " dau mut ..."
    Dim ex As Double, ey As Double, g As Long, dmin As Double, d As Double, daTrung As Boolean
    Dim dsTach() As String, m As Long, dx As Long, dy As Long, nLech As Long
    Dim kq() As Double, kx() As Double, ky() As Double
    ReDim kq(1 To 16): ReDim kx(1 To 16): ReDim ky(1 To 16)
    For i = 1 To dlNE
        ex = dlEX(i): ey = dlEY(i): g = dlEE(i)
        ix = Int(ex / CELL): iy = Int(ey / CELL)
        daTrung = False
        For dx = -1 To 1
            For dy = -1 To 1
                key = CStr(ix + dx) & "_" & CStr(iy + dy)
                If dV.Exists(key) Then
                    dsTach = Split(dV(key), ",")
                    For m = 0 To UBound(dsTach)
                        j = CLng(dsTach(m))
                        If dlVE(j) <> g Then
                            If Abs(dlVX(j) - ex) <= SAI_TRUNG And Abs(dlVY(j) - ey) <= SAI_TRUNG Then daTrung = True: Exit For
                        End If
                    Next m
                End If
                If daTrung Then Exit For
            Next dy
            If daTrung Then Exit For
        Next dx
        If Not daTrung Then
            dmin = 1E+30
            For dx = -1 To 1
                For dy = -1 To 1
                    key = CStr(ix + dx) & "_" & CStr(iy + dy)
                    If dS.Exists(key) Then
                        dsTach = Split(dS(key), ",")

                            End If
                        Next m
                    End If
                Next dy
            Next dx
            If dmin > SAI_TRUNG And dmin < tolLech Then
                ' bo daTrung vi tri (2 dau mut cung cho)
                ok = True
                For k = 1 To nLech
                    If Abs(kx(k) - ex) < 0.000001 And Abs(ky(k) - ey) < 0.000001 Then ok = False: Exit For
                Next k
                If ok Then
                    nLech = nLech + 1
                    If nLech > UBound(kq) Then ReDim Preserve kq(1 To nLech * 2): ReDim Preserve kx(1 To nLech * 2): ReDim Preserve ky(1 To nLech * 2)
                    kq(nLech) = dmin: kx(nLech) = ex: ky(nLech) = ey
                End If
            End If
        End If
        If i Mod 300 = 0 Then DoEvents
    Next i

    ' --- sap xep tang dan (lech cang nho cang kho thay, xep truoc)
    Dim t As Double
    For i = 1 To nLech - 1
        For j = i + 1 To nLech
            If kq(j) < kq(i) Then
                t = kq(i): kq(i) = kq(j): kq(j) = t
                t = kx(i): kx(i) = kx(j): kx(j) = t
                t = ky(i): ky(i) = ky(j): ky(j) = t
            End If
        Next j
    Next i
    Dim v As clsVung, rN() As Double, h As Double, mo As String
    h = 0.5
    For i = 1 To nLech
        ReDim rN(0 To 4, 0 To 1)
        rN(0, 0) = kx(i) - h: rN(0, 1) = ky(i) - h
        rN(1, 0) = kx(i) + h: rN(1, 1) = ky(i) - h
        rN(2, 0) = kx(i) + h: rN(2, 1) = ky(i) + h
        rN(3, 0) = kx(i) - h: rN(3, 1) = ky(i) + h
        rN(4, 0) = rN(0, 0): rN(4, 1) = rN(0, 1)
        Set v = New clsVung
        v.ring = rN: v.cx = kx(i): v.cy = ky(i): v.Dt = -1#: v.HasRing = True
        If kq(i) * 1000# < 0.001 Then mo = Format$(kq(i) * 1000#, "0.0E+00") Else mo = Format$(kq(i) * 1000#, "0.000")
        mo = ChrW(272) & ChrW(7847) & "u ranh l" & ChrW(7879) & "ch " & mo & " mm (TW coi l" & ChrW(224) & " h" & ChrW(7903) & ") - b" & ChrW(7855) & "t Keypoint l" & ChrW(7841) & "i"
        KV_DsLoi.Add Array(v, mo, "X=" & Format$(kx(i), "0.000") & " Y=" & Format$(ky(i), "0.000"))
    Next i
    If nLech = 0 Then
        tomTat = ChrW(272) & ChrW(7841) & "t: " & dlNE & " " & ChrW(273) & ChrW(7847) & "u m" & ChrW(250) & "t ranh " & ChrW(273) & ChrW(7873) & "u kh" & ChrW(7899) & "p " & ChrW(273) & ChrW(250) & "ng " & ChrW(273) & ChrW(7881) & "nh/c" & ChrW(7841) & "nh (kh" & ChrW(244) & "ng c" & ChrW(243) & " " & ChrW(273) & ChrW(7847) & "u l" & ChrW(7879) & "ch d" & ChrW(432) & ChrW(7899) & "i " & Format$(tolLech * 1000#, "0.###") & " mm)."
    Else
        tomTat = nLech & " " & ChrW(273) & ChrW(7847) & "u ranh l" & ChrW(7879) & "ch nh" & ChrW(7887) & " (d" & ChrW(432) & ChrW(7899) & "i " & Format$(tolLech * 1000#, "0.###") & " mm): MicroStation v" & ChrW(7851) & "n t" & ChrW(7841) & "o v" & ChrW(249) & "ng nh" & ChrW(432) & "ng TW coi l" & ChrW(224) & " H" & ChrW(7902) & " -> 2 th" & ChrW(7917) & "a th" & ChrW(224) & "nh 1 v" & ChrW(249) & "ng." & vbCrLf & "B" & ChrW(7845) & "m t" & ChrW(7915) & "ng d" & ChrW(242) & "ng, d" & ChrW(249) & "ng Modify Element b" & ChrW(7855) & "t Keypoint (K) " & ChrW(273) & ChrW(432) & "a " & ChrW(273) & ChrW(7847) & "u m" & ChrW(250) & "t v" & ChrW(224) & "o " & ChrW(273) & ChrW(250) & "ng " & ChrW(273) & ChrW(7881) & "nh ranh b" & ChrW(234) & "n c" & ChrW(7841) & "nh."
    End If
    ShowStatus ""
    KV_DoDauLech = nLech
End Function

' Them 1 day dinh (1 doi tuong) vao bang dinh / canh / dau mut
Private Sub DL_ThemDay(pts() As Point3d, ByVal g As Long, ByVal coDauMut As Boolean)
    Dim i As Long, lo As Long, hi As Long
    On Error Resume Next
    lo = LBound(pts): hi = UBound(pts)
    If Err.Number <> 0 Then Err.Clear: Exit Sub
    On Error GoTo 0
    If hi < lo Then Exit Sub
    If Not DC_DinhHopLe(pts) Then Exit Sub      ' SUA 08/11: bo net hong toa do NaN
    For i = lo To hi
        dlNV = dlNV + 1
        If dlNV > UBound(dlVX) Then ReDim Preserve dlVX(1 To dlNV * 2): ReDim Preserve dlVY(1 To dlNV * 2): ReDim Preserve dlVE(1 To dlNV * 2)
        dlVX(dlNV) = pts(i).X: dlVY(dlNV) = pts(i).Y: dlVE(dlNV) = g
        If i > lo Then
            If pts(i).X <> pts(i - 1).X Or pts(i).Y <> pts(i - 1).Y Then
                dlNS = dlNS + 1
                If dlNS > UBound(dlSX1) Then
                    ReDim Preserve dlSX1(1 To dlNS * 2): ReDim Preserve dlSY1(1 To dlNS * 2)
                    ReDim Preserve dlSX2(1 To dlNS * 2): ReDim Preserve dlSY2(1 To dlNS * 2): ReDim Preserve dlSE(1 To dlNS * 2)
                End If
                dlSX1(dlNS) = pts(i - 1).X: dlSY1(dlNS) = pts(i - 1).Y
                dlSX2(dlNS) = pts(i).X: dlSY2(dlNS) = pts(i).Y: dlSE(dlNS) = g
            End If
        End If
    Next i
    If coDauMut And hi > lo Then
        If pts(lo).X = pts(hi).X And pts(lo).Y = pts(hi).Y Then Exit Sub   ' day tu khep kin
        DL_ThemDau pts(lo).X, pts(lo).Y, g
        DL_ThemDau pts(hi).X, pts(hi).Y, g
    End If
End Sub

Private Sub DL_ThemDau(ByVal x As Double, ByVal y As Double, ByVal g As Long)
    dlNE = dlNE + 1
    If dlNE > UBound(dlEX) Then ReDim Preserve dlEX(1 To dlNE * 2): ReDim Preserve dlEY(1 To dlNE * 2): ReDim Preserve dlEE(1 To dlNE * 2)
    dlEX(dlNE) = x: dlEY(dlNE) = y: dlEE(dlNE) = g
End Sub

' Khoang cach diem (px,py) toi doan AB - tinh theo toa do tuong doi de giu do chinh xac
Private Function DL_KcDoan(ByVal px As Double, ByVal py As Double, ByVal ax As Double, ByVal ay As Double, _
                           ByVal bx As Double, ByVal by As Double) As Double
    Dim ux As Double, uy As Double, wx As Double, wy As Double, L2 As Double, t As Double
    ux = bx - ax: uy = by - ay
    wx = px - ax: wy = py - ay
    L2 = ux * ux + uy * uy
    If L2 <= 0# Then DL_KcDoan = Sqr(wx * wx + wy * wy): Exit Function
    t = (wx * ux + wy * uy) / L2
    If t < 0# Then t = 0#
    If t > 1# Then t = 1#
    wx = wx - t * ux: wy = wy - t * uy
    DL_KcDoan = Sqr(wx * wx + wy * wy)
End Function

' Chuan hoa so thua de so trung: bo khoang trang, ky tu dieu khien

    Dim i As Long, ch As String, r As String
    For i = 1 To Len(t)
        ch = Mid$(t, i, 1)
        If AscW(ch) > 32 Then r = r & ch
    Next i
    KV_ChuanSo = UCase$(r)
End Function
