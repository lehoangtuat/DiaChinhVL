'============================================================
' modCuonChuot - LAN CON CHUOT de cuon ListBox tren form (MSForms khong tu ho tro)
'   Moi form: Set mCuon = DC_GanCuon(lst, Me) sau khi tao ListBox ;
'             UserForm_Terminate -> DC_CuonTat
'   Hook chuot (WH_MOUSE_LL) chi bat khi chuot vao ListBox, tu go khi chuot
'   ra khoi form (khong an mat cuon chuot de zoom trong MicroStation).
'============================================================
Option Explicit

Private Type POINTAPI
    X As Long
    Y As Long
End Type

Private Declare Function SetWindowsHookEx Lib "user32" Alias "SetWindowsHookExA" (ByVal idHook As Long, ByVal lpfn As Long, ByVal hmod As Long, ByVal dwThreadId As Long) As Long
Private Declare Function CallNextHookEx Lib "user32" (ByVal hHook As Long, ByVal nCode As Long, ByVal wParam As Long, ByVal lParam As Long) As Long
Private Declare Function UnhookWindowsHookEx Lib "user32" (ByVal hHook As Long) As Long
Private Declare Function GetModuleHandle Lib "kernel32" Alias "GetModuleHandleA" (ByVal lpModuleName As String) As Long
Private Declare Sub CopyMemory Lib "kernel32" Alias "RtlMoveMemory" (Destination As Any, ByVal Source As Long, ByVal Length As Long)
Private Declare Function WindowFromPoint Lib "user32" (ByVal xPoint As Long, ByVal yPoint As Long) As Long
Private Declare Function GetAncestor Lib "user32" (ByVal hwnd As Long, ByVal gaFlags As Long) As Long
Private Declare Function FindWindow Lib "user32" Alias "FindWindowA" (ByVal lpClassName As String, ByVal lpWindowName As String) As Long

Private Const WH_MOUSE_LL As Long = 14
Private Const WM_MOUSEWHEEL As Long = &H20A
Private Const GA_ROOT As Long = 2

Private mHook As Long
Private mLst As Object
Private mHwnd As Long

' Gan cuon chuot cho 1 ListBox cua form. Giu doi tuong tra ve trong form.
Public Function DC_GanCuon(ByVal lst As Object, ByVal frm As Object) As Object
    Dim c As clsCuonList
    On Error Resume Next
    Set c = New clsCuonList
    Set c.lst = lst
    Set c.Form = frm
    Set DC_GanCuon = c
End Function

' Goi khi chuot di vao ListBox
Public Sub DC_CuonBat(ByVal lst As Object, ByVal frm As Object)
    On Error Resume Next
    Set mLst = lst
    If mHook <> 0 Then Exit Sub
    mHwnd = FindWindow("ThunderDFrame", frm.Caption)
    If mHwnd = 0 Then mHwnd = FindWindow(vbNullString, frm.Caption)
    If mHwnd = 0 Then Exit Sub
    mHook = SetWindowsHookEx(WH_MOUSE_LL, AddressOf CuonProc, GetModuleHandle(vbNullString), 0)
End Sub

' Go hook (dong form / chuot ra ngoai form)
Public Sub DC_CuonTat()
    On Error Resume Next
    If mHook <> 0 Then UnhookWindowsHookEx mHook
    mHook = 0
    Set mLst = Nothing
End Sub

Private Function CuonProc(ByVal nCode As Long, ByVal wParam As Long, ByVal lParam As Long) As Long
    Dim pt As POINTAPI, md As Long, h As Long, hk As Long
    On Error Resume Next
    hk = mHook
    If nCode = 0 And wParam = WM_MOUSEWHEEL Then
        CopyMemory pt, lParam, 8
        CopyMemory md, lParam + 8, 4
        h = WindowFromPoint(pt.X, pt.Y)
        If h <> 0 Then h = GetAncestor(h, GA_ROOT)
        If h = mHwnd And mHwnd <> 0 And Not mLst Is Nothing Then
            CuonList mLst, md
            CuonProc = 1
            Exit Function
        End If
        ' chuot da ra ngoai form -> go hook, tra lai cuon chuot cho MicroStation
        CuonProc = CallNextHookEx(hk, nCode, wParam, lParam)
        DC_CuonTat
        Exit Function
    End If
    CuonProc = CallNextHookEx(hk, nCode, wParam, lParam)
End Function

Private Sub CuonList(ByVal lst As Object, ByVal md As Long)
    Dim d As Long, b As Long, ti As Long, n As Long
    On Error Resume Next
    d = Fix(md / 65536#)
    If d = 0 Then Exit Sub
    b = Abs(d) \ 120: If b < 1 Then b = 1
    n = lst.ListCount
    If n = 0 Then Exit Sub
    ti = lst.TopIndex - Sgn(d) * 3 * b
    If ti < 0 Then ti = 0
    If ti > n - 1 Then ti = n - 1
    lst.TopIndex = ti
End Sub
