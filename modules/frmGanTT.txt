Public Sub XuLyNut(ByVal ma As String)
    If ma = "DONG" Then Unload Me: Exit Sub
    Dim i As Long, coLv As Boolean, nm As Variant
    For Each nm In Array("txtLvSo", "txtLvTo", "txtLvLoai", "txtLvChu", "txtLvDC")
        If Len(Trim$(Me.Controls(nm).Text)) > 0 Then coLv = True
    Next nm
    If Not coLv Then
        DC_ThongBao "Ch" & ChrW(432) & "a nh" & ChrW(7853) & "p level n" & ChrW(224) & "o (s" & ChrW(7889) & " hi" & ChrW(7879) & "u th" & ChrW(7917) & "a / t" & ChrW(7901) & " / lo" & ChrW(7841) & "i " & ChrW(273) & ChrW(7845) & "t / t" & ChrW(234) & "n ch" & ChrW(7911) & " / " & ChrW(273) & ChrW(7883) & "a ch" & ChrW(7881) & ").", "G" & ChrW(225) & "n th" & ChrW(244) & "ng tin", vbExclamation
        Exit Sub
    End If
    Me.Hide
    On Error GoTo Loi
    ChayGanTT Me.Controls("txtPath").Text, Me.Controls("txtLvSo").Text, _
              Me.Controls("txtLvTo").Text, Me.Controls("txtLvLoai").Text, _
              Me.Controls("txtLvChu").Text, Me.Controls("txtLvDC").Text, _
              Me.Controls("txtLvNhan").Text, Val(Me.Controls("txtCao").Text), _
              Me.Controls("chkVe").Value, Me.Controls("chkXoaCu").Value
    Me.Show vbModeless
    Exit Sub
Loi:
    Me.Show vbModeless
    DC_ThongBao "L" & ChrW(7895) & "i khi g" & ChrW(225) & "n th" & ChrW(244) & "ng tin: " & Err.Description, "G" & ChrW(225) & "n th" & ChrW(244) & "ng tin", vbCritical
End Sub
