'------------------------------------------------------------ TY LE BAN DO
Public Function DC_TyLeBD() As Long
    DC_TyLeBD = CLng(Val(DC_DocCauHinh("TYLE_BD", "1000")))
    If DC_TyLeBD <= 0 Then DC_TyLeBD = 1000
End Function

Public Sub DC_DatTyLeBD(ByVal tl As Long)
    If tl <= 0 Then Exit Sub
    DC_GhiCauHinh "TYLE_BD", CStr(tl)
    On Error Resume Next
    ShowStatus "Ty le ban do 1:" & tl
End Sub

'------------------------------------------------------------ BANG QUY DINH LOP
' Tra ve Collection, moi phan tu = mang String(0..8)
Public Function BT_DocBang() As Collection
    Dim kq As New Collection, txt As String, dong() As String, i As Long, p() As String, j As Long
    Set BT_DocBang = kq
    If Len(Dir$(BT_FILE)) = 0 Then GhiBangMacDinh
    txt = DocUtf8(BT_FILE)
    If Len(txt) = 0 Then txt = BangMacDinh()
    dong = Split(Replace(txt, vbCr, ""), vbLf)
    For i = 0 To UBound(dong)
        If Len(Trim$(dong(i))) > 0 And Left$(Trim$(dong(i)), 1) <> "#" Then
            p = Split(dong(i), "|")
            If UBound(p) >= 3 Then
                ReDim Preserve p(0 To 8)
                For j = 0 To 8
                    p(j) = Trim$(p(j))
                Next j
                kq.Add p
            End If
        End If
    Next i
End Function

Private Function TimMa(ByVal ma As String) As Variant
    Dim ds As Collection, i As Long
    Set ds = BT_DocBang()
    For i = 1 To ds.Count
        If StrComp(ds.Item(i)(1), ma, vbTextCompare) = 0 Then TimMa = ds.Item(i): Exit Function
    Next i
    TimMa = Empty
End Function

'------------------------------------------------------------ CHAY 1 MUC
Public Sub DC_BienTap(ByVal ma As String)
    Dim r As Variant, lv As Level, cao As Double, f As Object
    r = TimMa(ma)
    If IsEmpty(r) Then
        DC_ThongBao "Ch" & ChrW(432) & "a c" & ChrW(243) & " m" & ChrW(7909) & "c '" & ma & "' trong " & BT_FILE, TieuDe, vbExclamation
        Exit Sub
    End If
    On Error Resume Next
    If Val(r(4)) > 0 Then
        Set lv = DC_LevelTheoSo(CLng(Val(r(4))))
        If Not lv Is Nothing Then Set ActiveSettings.Level = lv
    End If
    ActiveSettings.Color = CLng(Val(r(5)))
    ActiveSettings.LineWeight = CLng(Val(r(7)))
    CadInputQueue.SendKeyin "lc=" & CLng(Val(r(6)))
    Select Case UCase$(r(3))
    Case "L"
        CadInputQueue.SendKeyin "place smartline"
        ShowPrompt "DiaChinhVL: " & ma
    Case "T"
        cao = Val(Replace(r(8), ",", ".")) * DC_TyLeBD() / 1000#
        If cao <= 0 Then cao = 2 * DC_TyLeBD() / 1000#
        ActiveSettings.TextStyle.Height = cao
        ActiveSettings.TextStyle.Width = cao
        Set f = FontVN()
        If Not f Is Nothing Then Set ActiveSettings.TextStyle.Font = f
        CadInputQueue.SendKeyin "place text"
        ShowPrompt "DiaChinhVL: " & ma & " (co chu " & Format$(cao, "0.00") & " m)"
    Case Else
        DC_ThongBao "M" & ChrW(7909) & "c k" & ChrW(253) & " hi" & ChrW(7879) & "u n" & ChrW(224) & "y ch" & ChrW(432) & "a l" & ChrW(224) & "m (c" & ChrW(7847) & "n th" & ChrW(432) & " vi" & ChrW(7879) & "n cell).", TieuDe, vbExclamation
    End Select
End Sub

Private Function FontVN() As Object
    Dim f As Object
    If Not mFont Is Nothing Then Set FontVN = mFont: Exit Function
    On Error Resume Next
    For Each f In ActiveDesignFile.Fonts
        If StrComp(f.Name, DC_FONT_VIET, vbTextCompare) = 0 Then Set mFont = f: Exit For
    Next f
    Set FontVN = mFont
End Function

'------------------------------------------------------------ FILE UTF-8
Private Function DocUtf8(ByVal f As String) As String
    Dim st As Object
    On Error Resume Next
    Set st = CreateObject("ADODB.Stream")
    st.Type = 2: st.Charset = "utf-8": st.Open
    st.LoadFromFile f
    DocUtf8 = st.ReadText(-1)
    st.Close
    If Len(DocUtf8) > 0 Then
        If AscW(Left$(DocUtf8, 1)) = &HFEFF Then DocUtf8 = Mid$(DocUtf8, 2)
    End If
End Function

Private Sub GhiBangMacDinh()
    Dim st As Object
    On Error Resume Next
    If Len(Dir$(DC_THUMUC, vbDirectory)) = 0 Then MkDir DC_THUMUC
    Set st = CreateObject("ADODB.Stream")
    st.Type = 2: st.Charset = "utf-8": st.Open
    st.WriteText BangMacDinh()
    st.SaveToFile BT_FILE, 2
    st.Close
End Sub
