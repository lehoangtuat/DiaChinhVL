'============================================================
' Hien popup menu ngay duoi nhan menu tren form.
'   x, y     : toa do chuot trong nhan (point, tu su kien MouseDown)
'   caoNhan  : chieu cao nhan (point)
' Tra ve ma lenh da chon ("" neu bam ra ngoai)
'============================================================
Public Function DC_HienMenu(ByVal ten As String, ByVal tieuDe As String, _
                            ByVal x As Single, ByVal y As Single, _
                            ByVal caoNhan As Single) As String
    Dim h As Long, hw As Long, id As Long, k As Double
    Dim p As POINTAPI

    h = TaoMenu(ten)
    If h = 0 Then Exit Function

    GetCursorPos p
    k = HeSoDpi()
    hw = FindWindowA("ThunderDFrame", tieuDe)
    If hw = 0 Then hw = GetForegroundWindow()
    SetForegroundWindow hw

    id = TrackPopupMenu(h, TPM_RETURNCMD Or TPM_NONOTIFY Or TPM_LEFTBUTTON, _
                        CLng(p.x - x * k), CLng(p.y + (caoNhan - y) * k), 0, hw, 0)
    DestroyMenu h

    If id >= 1 And id <= mN Then DC_HienMenu = mMa(id)
End Function

'------------------------------------------------------------
Private Sub Muc(ByVal h As Long, ByVal cap As String, ByVal ma As String)
    mN = mN + 1
    ReDim Preserve mMa(1 To mN)
    mMa(mN) = ma
    If Len(ma) = 0 Then
        AppendMenuW h, MF_STRING Or MF_GRAYED, mN, StrPtr(cap)
    Else
        AppendMenuW h, MF_STRING, mN, StrPtr(cap)
    End If
End Sub

Private Sub MucTick(ByVal h As Long, ByVal cap As String, ByVal ma As String, ByVal tick As Boolean)
    mN = mN + 1
    ReDim Preserve mMa(1 To mN)
    mMa(mN) = ma
    AppendMenuW h, MF_STRING Or IIf(tick, MF_CHECKED, 0), mN, StrPtr(cap)
End Sub

' Menu con theo nhom trong bang Quy dinh lop (modBienTap). K = ky hieu chua lam -> mo.
Private Sub NhomBT(ByVal h As Long, ByVal nhom As String, ByVal cap As String)
    Dim ds As Collection, i As Long, hc As Long, r As Variant, loaiTruoc As String, coMuc As Boolean
    hc = MenuCon(h, cap)
    If hc = 0 Then Exit Sub
    Set ds = BT_DocBang()
    For i = 1 To ds.Count
        r = ds.Item(i)
        If StrComp(r(0), nhom, vbTextCompare) = 0 Then
            If coMuc And UCase$(r(3)) <> loaiTruoc And loaiTruoc = "L" Then Gach hc
            If UCase$(r(3)) = "K" Then
                Muc hc, r(2), ""
            Else
                Muc hc, r(2), "BT:" & r(1)
            End If
            loaiTruoc = UCase$(r(3)): coMuc = True
        End If
    Next i
    If Not coMuc Then Muc hc, "(ch" & ChrW(432) & "a c" & ChrW(243) & " trong Quy " & ChrW(273) & ChrW(7883) & "nh l" & ChrW(7899) & "p)", ""
End Sub

' Menu con (submenu); id cac muc trong menu con dung chung day so mN
Private Function MenuCon(ByVal h As Long, ByVal cap As String) As Long
    Dim hc As Long
    hc = CreatePopupMenu()
    If hc = 0 Then Exit Function
    AppendMenuW h, MF_POPUP, hc, StrPtr(cap)
    MenuCon = hc
End Function

Private Sub Gach(ByVal h As Long)
    AppendMenuW h, MF_SEPARATOR, 0, 0
End Sub

Private Function HeSoDpi() As Double
    Dim hdc As Long, dpi As Long
    hdc = GetDC(0)
    dpi = GetDeviceCaps(hdc, LOGPIXELSX)
    ReleaseDC 0, hdc
    If dpi <= 0 Then dpi = 96
    HeSoDpi = dpi / 72#
End Function

'============================================================
' Gan logo (file .ico) len thanh tieu de cua form.
'   Goi trong UserForm_Activate (luc do cua so da ton tai).
'   Khong co file icon -> bo qua, khong bao loi.
'============================================================
Public Sub DC_GanIcon(ByVal tieuDe As String)
    On Error Resume Next
    Dim hw As Long, hNho As Long, hLon As Long, ex As Long
    If Len(Dir$(DC_ICON)) = 0 Then Exit Sub
    hw = FindWindowA("ThunderDFrame", tieuDe)
    If hw = 0 Then Exit Sub

    ' bo kieu khung hop thoai -> Windows moi ve icon o goc trai
    ex = GetWindowLongA(hw, GWL_EXSTYLE)
    SetWindowLongA hw, GWL_EXSTYLE, ex And Not WS_EX_DLGMODALFRAME

    hNho = LoadImageA(0, DC_ICON, IMAGE_ICON, 16, 16, LR_LOADFROMFILE)
    hLon = LoadImageA(0, DC_ICON, IMAGE_ICON, 32, 32, LR_LOADFROMFILE)
    If hNho <> 0 Then SendMessageA hw, WM_SETICON, ICON_SMALL, hNho
    If hLon <> 0 Then SendMessageA hw, WM_SETICON, ICON_BIG, hLon

    SetWindowPos hw, 0, 0, 0, 0, 0, SWP_FLAGS
End Sub
